"""副次の観察のための形: 表を関数の中でなく module の直下で 1 度だけ読む(読み込みの時に dict を作る)。"""
import sys
from pathlib import Path
check = Path(sys.argv[1]) / "static_check.py"
text = check.read_text(encoding="utf-8")
old = "    severities = finding_severities()\n"
new = "    severities = _SEVERITIES\n"
assert text.count(old) == 1
text = text.replace(old, new)
anchor = "\n\ndef _finding_diagnostic("
assert text.count(anchor) == 1
text = text.replace(anchor, "\n\n_SEVERITIES = json.loads(Path(__file__).with_name(FINDING_SEVERITIES).read_text(encoding=\"utf-8\"))[\"severity\"]\n" + anchor, 1)
check.write_text(text, encoding="utf-8")
