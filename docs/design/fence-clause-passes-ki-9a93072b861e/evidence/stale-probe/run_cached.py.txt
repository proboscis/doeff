"""結果の cache を使って through_gate を解析する(同じ cache の dir を 2 つの木の analyzer が使う)。

使い方: DOEFF_EFFECT_ANALYZER_CACHE=<dir> uv run python run_cached.py <analyzer の python の dir>
"""
import sys
from pathlib import Path

import hy  # noqa: F401 - .hy import hook

here = Path(__file__).resolve().parent
sys.path.insert(0, sys.argv[1])
sys.path.insert(0, str(here))

from doeff_effect_analyzer.program_effects import analyze_program
from doeff_effect_analyzer.result_cache import cached_result

computed = []


def compute():
    computed.append(True)
    report = analyze_program("fxstale.programs:through_gate")
    return sorted(t.__name__ for t in report.effect_types)


answer = cached_result(("stale-probe", "through_gate"), compute)
print("ANSWER", answer, "FROM", "analysis" if computed else "stored answer")
