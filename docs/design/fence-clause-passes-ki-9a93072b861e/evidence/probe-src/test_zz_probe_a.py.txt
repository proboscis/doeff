import pytest
from doeff_effect_analyzer.handler_effects import check_coverage, analyze_handler
from doeff_effect_analyzer.program_effects import analyze_program

pytest.importorskip("hy")
pytest.importorskip("doeff_hy")

EFFECTS_PY = """\
from dataclasses import dataclass

from doeff_vm import EffectBase


@dataclass(frozen=True)
class Tick(EffectBase):
    pass


@dataclass(frozen=True)
class Nap(EffectBase):
    seconds: float


@dataclass(frozen=True)
class ReadFlag(EffectBase):
    name: str
"""

PROGRAMS_HY = """\
(require doeff-hy.macros [defk <-])
(require doeff-hy.handle [handle defhandler with-handler])
(import doeff [EffectBase])
(import {pkg}.effects [Tick Nap ReadFlag])
(defk nap-only [] {:pre [] :post [(: % int)]} (<- (Nap 1.0)) 1)
(defhandler maintenance-gate
  (session val closed (! (ReadFlag "maintenance")))
  (EffectBase []
    :when (and closed (not (isinstance effect Nap)))
    (raise (ValueError "closed for maintenance"))))
(defk through-gate [] {:pre [] :post [(: % int)]}
  (<- r (with-handler [maintenance-gate] (nap-only)))
  r)
"""


def test_what_leaves_the_maintenance_gate(make_package):
    pkg = make_package({"effects.py": EFFECTS_PY, "programs.hy": PROGRAMS_HY})
    report = analyze_program(f"{pkg}.programs:through_gate")
    print("LEAVES", sorted(t.__name__ for t in report.effect_types), "unresolved", report.unresolved)
    [scope] = report.handled
    for handler in scope.handlers:
        print("HANDLER", handler.name, handler.basis)
        for clause in handler.clauses:
            print("  CLAUSE", clause.handles.__name__, "passes=", clause.passes, "emits=", sorted(clause.emits.effect_names))
