"""S2: 番に ``issubclass(type(effect), X)`` という書き方が増えた(使い捨ての入力)。"""

import pytest
from doeff_effect_analyzer.program_effects import analyze_program

pytest.importorskip("hy")
pytest.importorskip("doeff_hy")

EFFECTS_PY = """\
from dataclasses import dataclass

from doeff_vm import EffectBase


@dataclass(frozen=True)
class Tick(EffectBase):
    pass


@dataclass(frozen=True)
class Nap(EffectBase):
    seconds: float
"""

PROGRAMS_HY = """\
(require doeff-hy.macros [defk <-])
(require doeff-hy.handle [defhandler with-handler])
(import doeff [EffectBase])
(import {pkg}.effects [Tick Nap])
(defk both [] {:pre [] :post [(: % int)]} (<- (Nap 1.0)) (<- (Tick)) 1)
(defhandler fence-subclass
  (EffectBase [] :when (not (issubclass (type effect) Nap)) (raise (ValueError "fenced"))))
(defk through-fence-subclass [] {:pre [] :post [(: % int)]}
  (<- r (with-handler [fence-subclass] (both)))
  r)
"""


def test_issubclass_of_the_type_is_read_as_a_class_test(make_package):
    pkg = make_package({"effects.py": EFFECTS_PY, "programs.hy": PROGRAMS_HY})
    report = analyze_program(f"{pkg}.programs:through_fence_subclass")
    assert {t.__name__ for t in report.effect_types} == {"Nap"}
