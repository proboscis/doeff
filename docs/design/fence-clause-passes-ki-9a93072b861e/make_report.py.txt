def applicable(sid, axis, change, claim, expected, actual, changed, unchanged, reason, revision):
    return {
        "id": sid, "axis": axis, "applicability": "applicable", "change": change, "claim": claim,
        "expected_scope": expected, "actual_scope": actual, "changed_modules": changed,
        "unchanged_modules": unchanged, "reason": reason, "revision": revision,
    }


def excluded(sid, axis, reason):
    return {"id": sid, "axis": axis, "applicability": "excluded", "reason": reason}
