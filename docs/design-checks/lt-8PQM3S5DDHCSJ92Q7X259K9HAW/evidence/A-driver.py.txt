"""worker_like.py を systemd の StandardOutput=append: と StandardError=append: の持ち方(同じ path を 2 度開く)で走らせ、見張りが
止めるべき子が止まったかと、見張りの行が file に在るかを読む。argv: <case 名> <lifeline の file> <送り先 on|off> <終わり方 exit|kill> <待つ秒>"""

import json
import os
import signal
import subprocess
import sys
import time
from pathlib import Path

here = Path(__file__).parent
case, lifeline_file, shipping, ending, settle = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4], float(sys.argv[5])
work = here / case
work.mkdir()
log = work / "worker.log"
report = work / "report.json"
env = {k: v for k, v in os.environ.items() if not k.startswith("DOEFF_JOB_LOG_")}
if shipping == "on":
    env |= {"DOEFF_JOB_LOG_OTLP": "http://127.0.0.1:9", "DOEFF_JOB_LOG_ATTRIBUTES": "agora.tenant=personal"}
with open(log, "ab") as out, open(log, "ab") as err:
    worker = subprocess.Popen(
        [sys.executable, "-B", str(here / "worker_like.py"), lifeline_file, str(report), ending],
        stdin=subprocess.DEVNULL, stdout=out, stderr=err, env=env,
    )
code = worker.wait(timeout=30)
ended_at = time.monotonic()
facts = json.loads(report.read_text())
time.sleep(settle)


def alive_group(pgid: int) -> bool:
    try:
        os.killpg(pgid, 0)
    except ProcessLookupError:
        return False
    return True


def alive_pid(pid: int) -> bool:
    try:
        os.kill(pid, 0)
    except ProcessLookupError:
        return False
    return True


child_alive = alive_group(facts["child_pgid"])
guard_alive = alive_pid(facts["guard_pid"])
text = log.read_text(errors="replace")
print(json.dumps({
    "case": case,
    "worker_exit": code,
    **facts,
    "seconds_after_worker_end": round(time.monotonic() - ended_at, 1),
    "child_group_still_alive": child_alive,
    "guard_still_alive": guard_alive,
    "lifeline_line_in_file": "lifeline:" in text,
}, ensure_ascii=False, indent=1))
print("--- worker.log ---")
print(text)
if child_alive:
    os.killpg(facts["child_pgid"], signal.SIGKILL)
