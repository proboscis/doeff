# S1(機体を足す)と S2(口座を替える)の名乗りの確かめ。走らせ方: 木の dir で `uv run --no-sync python -I <この file を .py に写した物>`
from doeff_cluster.worker.entry.line_stamp import JobLogTarget, worker_log_plan

def resource(plan):
    assert isinstance(plan, JobLogTarget), plan
    return dict((a.key, a.value) for a in plan.resource)

# C-S1-P: 新しい機体の worker eos-native — 配備の値(名と送り先)だけで job の名が worker/eos-native になる
r = resource(worker_log_plan({"DOEFF_JOB_LOG_OTLP": "http://collector:4318", "DOEFF_JOB_LOG_ATTRIBUTES": "agora.tenant=personal"}, "eos-native", "b1"))
print("C-S1-P", r)
assert r["doeff.job"] == "worker/eos-native" and r["doeff.worker"] == "eos-native" and r["doeff.attempt"] == "0" and r["doeff.instance"] == "b1"

# C-S2-P: 会社の口座へ — 配備の属性 agora.tenant=company がそのまま resource に載る(doeff は口座の語彙を持たない)
r = resource(worker_log_plan({"DOEFF_JOB_LOG_OTLP": "http://collector:4318", "DOEFF_JOB_LOG_ATTRIBUTES": "agora.tenant=company"}, "atlas-native", "b2"))
print("C-S2-P", r)
assert r["agora.tenant"] == "company" and r["doeff.job"] == "worker/atlas-native"

# C-S2-N: 配備の属性が doeff の名乗り(doeff.job・service.name・doeff.attempt)を書き換えようとしても doeff の値が勝つ
r = resource(worker_log_plan({"DOEFF_JOB_LOG_OTLP": "http://collector:4318",
                              "DOEFF_JOB_LOG_ATTRIBUTES": "agora.tenant=company,doeff.job=turn-host-x,service.name=other,doeff.attempt=7"}, "atlas-native", "b2"))
print("C-S2-N", r)
assert r["doeff.job"] == "worker/atlas-native" and r["service.name"] == "doeff.job-log" and r["doeff.attempt"] == "0"
print("all asserted")
