"""機体の worker の代わり(検の材料): main.hy と同じ順で process_lines を呼んだ後、StartProcess の本物の答え手と同じ持ち方で子を立て、
見張り(lifeline)へ知らせて終わる。argv: <lifeline の file> <report の path> <終わり方 exit|kill>。"""

import json
import os
import signal
import subprocess
import sys

from doeff_cluster.worker.entry.line_stamp import process_lines, worker_log_plan

lifeline_file, report_path, ending = sys.argv[1], sys.argv[2], sys.argv[3]

lines = process_lines(worker_log_plan(os.environ, "zeus-native", "b0"))
print(f"worker: 起動します lines={type(lines).__name__}", file=sys.stderr, flush=True)

# StartProcess(stdout-path・stderr-path = None → DEVNULL・process-group = True)と同じ子: TERM を無視する子(猶予の後の KILL でしか止まらない)。
child = subprocess.Popen(
    ["sh", "-c", 'trap "" TERM; sleep 30'],
    stdin=subprocess.DEVNULL,
    stdout=subprocess.DEVNULL,
    stderr=subprocess.DEVNULL,
    start_new_session=True,
)
# doeff_lifeline.Lifeline._written と同じ起こし方(file だけを引数で替える)。
guard = subprocess.Popen(
    [sys._base_executable, "-I", "-S", "-B", lifeline_file],
    stdin=subprocess.PIPE,
    stdout=subprocess.DEVNULL,
    start_new_session=True,
)
guard.stdin.write(f"group {child.pid}\n".encode("ascii"))
guard.stdin.flush()

report = {
    "lines": type(lines).__name__,
    "child_pgid": child.pid,
    "guard_pid": guard.pid,
    "guard_fd2": os.readlink(f"/proc/{guard.pid}/fd/2"),
    "worker_fd2_now": os.readlink("/proc/self/fd/2"),
}
with open(report_path, "w") as handle:
    json.dump(report, handle)
print("worker: 全 job を回収しました", file=sys.stderr, flush=True)
if ending == "kill":
    os.kill(os.getpid(), signal.SIGKILL)
sys.exit(0)
