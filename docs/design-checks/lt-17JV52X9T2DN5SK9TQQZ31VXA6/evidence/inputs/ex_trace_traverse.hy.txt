;;; Example 10: trace the Traverse effects a pipeline performs.
;;;
;;; A tracing handler sits outside the strategy and prints every Traverse it sees, then forwards it.

(require doeff-hy.macros [defk <- traverse defhandler])
(import doeff [do :as _doeff-do])
(import doeff [run])

(import doeff_core_effects [try-handler :as try_handler])
(import doeff_core_effects.scheduler [scheduled])

(import doeff_traverse [Traverse])
(import doeff_traverse [Traverse :as _doeff_traverse_Traverse])
(import doeff_traverse.collection [Collection])
(import doeff_traverse.handlers [sequential fail-handler :as fail_handler])


(defhandler trace-traverse
  "Print each Traverse that passes by, then forward it to the strategy."
  (Traverse []
    (print "trace:" (. (type effect) __name__)
           "step-is-do:" (hasattr effect.f "__wrapped__")
           "items:" (. (type effect.items) __name__))
    (reperform effect)))


(defk double [x]
  {:pre [(: x int)]
   :post [(: % int)]}
  (* x 2))

(defk pipeline [items]
  {:pre [(: items list)]
   :post [(: % Collection)]}
  (<- results
    (traverse
      (<- x (Iterate items))
      (<- y (double x))
      y))
  results)

(setv body (pipeline [1 2 3]))
(setv body (try_handler body))
(setv body (fail_handler body))
(setv body (trace-traverse body))
(setv body ((sequential) body))
(setv body (scheduled body))

(setv result (run body))
(print "Valid:" (. result valid_values))
(print "Failed:" (len (. result failed_items)))
