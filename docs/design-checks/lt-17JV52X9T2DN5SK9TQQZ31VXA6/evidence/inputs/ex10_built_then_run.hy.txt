;;; Example 10: for/do builds a value — run executes it.
;;;
;;; (for/do …) does not process any item: it builds a value (data) that
;;; describes the batch. Items are processed only when run() hands that value
;;; to the handlers (here: sequential, one by one).
;;;
;;; The example prints what for/do built, then the answers after run.

(require doeff-hy.macros [<- for/do])
(import doeff [do :as _doeff-do])
(import doeff [run])

(import doeff_core_effects [try-handler :as try_handler])
(import doeff_core_effects.scheduler [scheduled])

(import doeff_traverse [Traverse :as _doeff_traverse_Traverse])
(import doeff_traverse.handlers [sequential fail-handler :as fail_handler])


;; One generator: one item per element
(setv single
  (for/do
    (<- x (From [1 2 3] :label "scale"))
    (* x 10)))

;; Two generators: one item per (x, y) pair
(setv pairs
  (for/do
    (<- x (From [1 2] :label "outer"))
    (<- y (From [10 20] :label "inner"))
    (+ (* x 10) y)))

(print "built:" (. (type single) __name__) "/" (. (type pairs) __name__))
(print "single:" (. (run (scheduled ((sequential) (fail_handler (try_handler single))))) valid_values))
(print "pairs:" (. (run (scheduled ((sequential) (fail_handler (try_handler pairs))))) valid_values))
