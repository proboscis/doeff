;;; Example 11: a stage built by a shared module, described and then run.

(require doeff-hy.macros [defk <-])
(import doeff [run])

(import doeff_core_effects [try-handler :as try_handler])
(import doeff_core_effects.scheduler [scheduled])

(import doeff_traverse.handlers [sequential fail-handler :as fail_handler])

(import support.stages [doubling-stage])


(setv stage (doubling-stage [1 2 3]))
(print "stage:" (. (type stage) __name__))

(setv result (run (scheduled ((sequential) (fail_handler (try_handler stage))))))
(print "Valid:" (. result valid_values))
(print "Failed:" (len (. result failed_items)))
