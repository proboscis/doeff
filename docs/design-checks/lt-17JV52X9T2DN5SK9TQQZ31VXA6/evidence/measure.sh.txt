# 設計検証 lt-17JV52X9T2DN5SK9TQQZ31VXA6 の確かめ c0〜c6 を、直した版の作業木で走らせ、出力を <evidence>/checks/ へ書く。
# 使い方: bash measure.sh.txt <作業木> <この記録の evidence の dir> <確かめの名 …>(例: c0 c1)
# 材料: <evidence>/inputs/ の *.txt(設計検証の盲検 B の ex10・独立の評価者の R2a/R2b の patch と例 — 例の拡張子 .hy の後ろに .txt を
#       足してある。使う時に外す)
# 変異と足した例は作業木の file を書き戻して・消して戻す(git の checkout・reset は使わない)。終わりに作業木の git status を印字する。
set -u
W=$1
OUT=$2
shift 2
EX=$W/packages/doeff-traverse/examples
TEST=packages/doeff-traverse/tests/test_examples_run.py
MACROS=$W/packages/doeff-hy/src/doeff_hy/macros.hy
IN=$OUT/inputs
SAVE=$(mktemp -d)
cp "$MACROS" "$SAVE/macros.hy"

restore() {
  cp "$SAVE/macros.hy" "$MACROS"
  rm -f "$EX/ex10_built_then_run.hy" "$EX/expected/ex10_built_then_run.txt" "$EX/ex11_uses_do_bang.hy" "$EX/expected/ex11_uses_do_bang.txt"
  rm -f "$EX/ex12_trace_traverse.hy" "$EX/expected/ex12_trace_traverse.txt" "$EX/ex13_shared_stage.hy" "$EX/expected/ex13_shared_stage.txt"
  rm -rf "$EX/support"
  rm -f "$EX/ex14_program_is_data.hy" "$EX/expected/ex14_program_is_data.txt" "$EX/ex15_copy_of_simple.hy" "$EX/expected/ex15_copy_of_simple.txt"
}
trap restore EXIT

pytest_run() {
  echo "\$ uv run pytest $TEST -q --tb=short -rf $*"
  (cd "$W" && uv run pytest "$TEST" -q --tb=short -rf "$@" 2>&1 \
    | grep -vE '^check-slot|PytestWarning|warnings.warn|^ +- docs/adr|^  - docs/adr|Add their directories|^deadline scale|^test-admission|^$' \
    | cut -c1-700
   echo "exit=${PIPESTATUS[0]}")
}

head_lines() {
  echo "# 確かめ $1 — $2"
  echo "# 版: $(git -C "$W" rev-parse HEAD)・作業木 ${W}・$(date -u +%FT%TZ)"
}

add_ex10() {
  cp "$IN/ex10_built_then_run.hy.txt" "$EX/ex10_built_then_run.hy"
  cp "$IN/ex10_built_then_run.expected.txt" "$EX/expected/ex10_built_then_run.txt"
}

for name in "$@"; do
  f="$OUT/checks/$name.txt"
  mkdir -p "$OUT/checks"
  case $name in
    c0)
      { head_lines c0 "直した版で例の検が全部緑(例 9 本・期待の出力は書き換えない)"; pytest_run; } >"$f" 2>&1 ;;
    c1)
      { head_lines c1 "展開が作った値の class 名を印字する例 ex10 を足す(盲検 B の形) — 変異の下の検がその例だけを名指して赤"
        add_ex10
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c2)
      { head_lines c2 "答えを変えない macro の変更(From 1 つの一番外も Program で包む)の下、例は今の 9 本 — 例と検は変わらず全部緑"
        (cd "$W" && git apply "$IN/macro_shape_change.patch.txt" && git diff --stat -- packages/doeff-hy)
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c3)
      { head_lines c3 "変異の表に無い macro(do!)を初めて使う例を足す — 表の検が名指して赤"
        cp "$IN/uses_do_bang.hy.txt" "$EX/ex11_uses_do_bang.hy"
        printf '%s\n' '3' >"$EX/expected/ex11_uses_do_bang.txt"
        pytest_run -k "mutated_or_named"
        restore; } >"$f" 2>&1 ;;
    c4)
      { head_lines c4 "変異で包む macro の名が今の doeff-hy に無い — 入口が 0 以外で終わり、表を合わせると名指す"
        (cd "$W/packages/doeff-traverse" && TZ=UTC0 uv run python tests/run_example_mutated_expansion.py examples/ex04_simple.hy traverse nosuchmacro; echo "exit=$?"); } >"$f" 2>&1 ;;
    c5)
      { head_lines c5 "変異の走りは bytecode の保存先へ、通常の走りと違う物を書かない(空の保存先で比べる): (1) 今の例 ex09 (2) 例が import する lib が defk・traverse を使う場合"
        MUT="defk fnk traverse for/do"
        T=$(mktemp -d)
        mkdir -p "$T/lib"
        cat >"$T/lib/built_by_a_library.hy" <<'HY'
(require doeff-hy.macros [<- defk traverse])
(import doeff [do :as _doeff-do])
(import doeff_traverse [Traverse :as _doeff_traverse_Traverse])
(defk twice [x] {:pre [(: x int)] :post [(: % int)]} (* x 2))
(setv library_value (traverse (<- x (Iterate [1 2 3])) (* x 2)))
HY
        printf '%s\n' '(import built_by_a_library)' '(print "library:" (. (type built_by_a_library.library_value) __name__) (. (type (built_by_a_library.twice 21)) __name__))' >"$T/lib/uses_library.hy"
        for scenario in ex09 lib; do
          for kind in plain mutated; do
            d=$(mktemp -d)
            if [ $scenario = ex09 ]; then target="$W/packages/doeff-traverse/examples/ex09_for_do_when.hy"; else target="$T/lib/uses_library.hy"; fi
            if [ $kind = plain ]; then
              (cd "$W/packages/doeff-traverse" && DOEFF_HY_CODE_STORE=$d TZ=UTC0 uv run python -m hy "$target" >"$T/out.$scenario.$kind" 2>&1)
            else
              (cd "$W/packages/doeff-traverse" && DOEFF_HY_CODE_STORE=$d TZ=UTC0 uv run python tests/run_example_mutated_expansion.py "$target" $MUT >"$T/out.$scenario.$kind" 2>&1)
            fi
            echo "$scenario $kind exit=$? files=$(find "$d" -type f | wc -l) content-hash=$(cd "$d" && find . -type f | sort | xargs md5sum | md5sum | cut -c1-32) output-last=$(tail -1 "$T/out.$scenario.$kind" | cut -c1-80)"
            rm -rf "$d"
          done
        done
        rm -rf "$T"; } >"$f" 2>&1 ;;
    c6)
      { head_lines c6 "検の file の所要(通常 3 回)— 基の版の 62 passed は 19.6 秒"
        for i in 1 2 3; do pytest_run | tail -3; done; } >"$f" 2>&1 ;;
    c7)
      { head_lines c7 "答えを変えない macro の変更: traverse が compiler を受ける(第 1 引数 _hy-compiler)— 独立の評価者の R2a。変異の入口は compiler を渡して包む"
        (cd "$W" && git apply "$IN/R2a-traverse-takes-compiler.patch.txt" && git diff --stat -- packages/doeff-hy)
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c8)
      { head_lines c8 "答えを変えない macro の変更: traverse の展開を恒久に何もしない handler の下で走らせる(変異と同じ形)— 独立の評価者の R2b"
        (cd "$W" && git apply "$IN/R2b-traverse-always-under-handler.patch.txt" && git diff --stat -- packages/doeff-hy)
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c9)
      { head_lines c9 "例の補助の lib を examples/support/ に置き、その lib が traverse で作った値の class 名を例が印字する — 独立の評価者の ex11。population の検が名指して赤"
        cp "$IN/ex_shared_stage.hy.txt" "$EX/ex13_shared_stage.hy"
        cp "$IN/ex_shared_stage.expected.txt" "$EX/expected/ex13_shared_stage.txt"
        mkdir -p "$EX/support"
        cp "$IN/stages.hy.txt" "$EX/support/stages.hy"
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c10)
      { head_lines c10 "(限界の実測)内側の defhandler が受けた Traverse の中身(件の関数の形)を印字する例 — 独立の評価者の ex10。変異は一番外の値だけを包むので、検は全部緑のまま"
        cp "$IN/ex_trace_traverse.hy.txt" "$EX/ex12_trace_traverse.hy"
        cp "$IN/ex_trace_traverse.expected.txt" "$EX/expected/ex12_trace_traverse.txt"
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c11)
      { head_lines c11 "defk の関数を呼んだ値の class を印字する例 ex14(独立の評価者 B の例: 「Program is DATA」を見せる)を足す — 変異の下の検がその例だけを名指して赤"
        cp "$IN/ex_program_is_data.hy.txt" "$EX/ex14_program_is_data.hy"
        cp "$IN/ex_program_is_data.expected.txt" "$EX/expected/ex14_program_is_data.txt"
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c12)
      { head_lines c12 "答えを変えない macro の変更: traverse の中身を内部の macro へ移し、公開の macro は hy.R で呼ぶだけにする — 独立の評価者 A。今回足した検は変わらず、前の card の「1 段の展開に印が在る」検だけが名指して赤"
        (cd "$W" && git apply "$IN/A-traverse-delegates-to-inner-macro.patch.txt" && git diff --stat -- packages/doeff-hy)
        pytest_run
        restore; } >"$f" 2>&1 ;;
    c13)
      { head_lines c13 "traverse の例を 1 本足す(ex04_simple の写し)— 例の検は緑だが、変更箇所の品質検査が private-symbol(別名 _doeff_traverse_Traverse の import)で赤: 前の card の所見 F1"
        cp "$EX/ex04_simple.hy" "$EX/ex15_copy_of_simple.hy"
        cp "$EX/expected/ex04_simple.txt" "$EX/expected/ex15_copy_of_simple.txt"
        pytest_run -k "ex15"
        (cd "$W" && git add -N "$EX/ex15_copy_of_simple.hy" "$EX/expected/ex15_copy_of_simple.txt" \
          && sh ~/dotfiles/agent/venv-run.sh ~/dotfiles/agentcli code-quality --repo "$PWD" --scope changed --base "$(git merge-base HEAD origin/main)" 2>&1 \
          | grep -vE '^check-slot|code-quality-timing|code-quality-coverage|code-quality-policy|code-quality-stage|code-quality-baseline'
          git rm -q --cached "$EX/ex15_copy_of_simple.hy" "$EX/expected/ex15_copy_of_simple.txt")
        restore; } >"$f" 2>&1 ;;
    c14)
      { head_lines c14 "同じ保存先で通常の走りと変異の走りを交互に走らせる(lib が defk・traverse を使う場合・どちらを先にしても)— 保存先は膨らまず、2 回目以降は作り直さない"
        MUT="defk fnk traverse for/do"
        T=$(mktemp -d)
        printf '%s\n' '(require doeff-hy.macros [<- defk traverse])' '(import doeff [do :as _doeff-do])' '(import doeff_traverse [Traverse :as _doeff_traverse_Traverse])' \
          '(defk twice [x] {:pre [(: x int)] :post [(: % int)]} (* x 2))' '(setv library_value (traverse (<- x (Iterate [1 2 3])) (* x 2)))' >"$T/built_by_a_library.hy"
        printf '%s\n' '(import built_by_a_library)' '(print "library:" (. (type built_by_a_library.library_value) __name__) (. (type (built_by_a_library.twice 21)) __name__))' >"$T/uses_library.hy"
        for first in plain mutated; do
          echo "## 冷えた保存先で $first を先に"
          S=$(mktemp -d)
          for step in $first $( [ $first = plain ] && echo mutated plain mutated plain || echo plain mutated plain mutated ); do
            start=$(date +%s%N)
            if [ $step = plain ]; then
              (cd "$W/packages/doeff-traverse" && DOEFF_HY_CODE_STORE=$S TZ=UTC0 uv run python -m hy "$T/uses_library.hy" >"$T/out" 2>&1)
            else
              (cd "$W/packages/doeff-traverse" && DOEFF_HY_CODE_STORE=$S TZ=UTC0 uv run python tests/run_example_mutated_expansion.py "$T/uses_library.hy" $MUT >"$T/out" 2>&1)
            fi
            rc=$?
            end=$(date +%s%N)
            echo "$step exit=$rc $(( (end - start) / 1000000 ))ms files=$(find "$S" -type f | wc -l) code=$(find "$S" -name '*.code' | wc -l) hash=$(cd "$S" && find . -type f | sort | xargs md5sum | md5sum | cut -c1-12) out=$(tail -1 "$T/out" | cut -c1-60)"
          done
          rm -rf "$S"
        done
        rm -rf "$T"; } >"$f" 2>&1 ;;
    *) echo "unknown check $name" >&2 ;;
  esac
done
echo "# 作業木の状態"; git -C "$W" status --short
