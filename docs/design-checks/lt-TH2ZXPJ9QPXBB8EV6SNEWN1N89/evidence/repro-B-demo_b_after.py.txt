"""盲検 B の実験: 写しの正本(K = 1)を検体の HOME に置き、別 process が正本の run で唯一の枠を持つ間に、変更後の doeff-hy-check を撃つ。

case = declared | knob | knob-renamed | renamed | knob-namedtuple
  declared        : 宣言 waitLimitSeconds=W・knob 無し(基準)
  knob            : 宣言 W・DOEFF_HY_CHECK_SLOT_WAIT=1
  knob-renamed    : 正本の写しの Policy の内部の欄 wait_limit_s を wait_limit_seconds へ改名(公開の名は不変)・knob=30
  renamed         : 同じ改名・knob 無し
  knob-namedtuple : 正本の写しの Policy を dataclass から NamedTuple へ(公開の名は不変)・knob=30
"""

import json
import os
import shlex
import shutil
import subprocess
import sys
import tempfile
import time
from pathlib import Path

TREE = Path("/home/ubuntu/.worktrees/doeff-wt-ki-26eb-mut")
PY = str(TREE / ".venv" / "bin" / "python")
CANON = Path.home() / "dotfiles" / "agent" / "tests" / "machine_check_slots.py"


def main(case: str, wait_s: float) -> None:
    base = Path(tempfile.mkdtemp(prefix=f"demo-{case}-", dir="/tmp/blind-B-scratch"))
    home = base / "home"
    canon = home / "dotfiles" / "agent" / "tests" / "machine_check_slots.py"
    canon.parent.mkdir(parents=True)
    text = CANON.read_text(encoding="utf-8")
    if "renamed" in case:
        text = text.replace("wait_limit_s", "wait_limit_seconds")
    if "namedtuple" in case:
        text = text.replace(
            "@dataclass(frozen=True)\nclass Policy:",
            "class Policy(NamedTuple):",
        ).replace("from typing import Callable,", "from typing import NamedTuple, Callable,")
    canon.write_text(text, encoding="utf-8")
    canon.with_name("machine_check_slots.json").write_text(
        json.dumps({"slots": 1, "dir": str(base / "slots"), "waitLimitSeconds": wait_s}), encoding="utf-8"
    )
    root = base / "repo"
    root.mkdir()
    (root / "probe.hy").write_text("(setv answer 1)\n", encoding="utf-8")
    started = base / "pyright-started.log"
    pyright = base / "pyright"
    pyright.write_text(
        f"#!/bin/sh\necho started >> {shlex.quote(str(started))}\necho '{{\"generalDiagnostics\": []}}'\n",
        encoding="utf-8",
    )
    pyright.chmod(0o755)
    env = dict(os.environ)
    env.pop("AI_CHECK_SLOT_HOLDER", None)
    env.pop("DOEFF_HY_CHECK_SLOT_WAIT", None)
    env["XDG_CACHE_HOME"] = env.get("XDG_CACHE_HOME") or str(Path.home() / ".cache")
    env["HOME"] = str(home)
    if "knob" in case:
        env["DOEFF_HY_CHECK_SLOT_WAIT"] = "1" if case == "knob" else "30"
    blocker = subprocess.Popen(
        [PY, str(canon), "run", "--label", "blocker-holds", "--", PY, "-c",
         "import sys; print('held', flush=True); sys.stdin.read()"],
        stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, env=env,
    )
    assert blocker.stdout is not None
    line = blocker.stdout.readline()
    print(f"[{case}] blocker (正本の写しの run) の枠: {line.strip()!r}")
    t0 = time.monotonic()
    check = [PY, "-m", "doeff_hy.static_check", "--root", str(root), "--no-cache", "--pyright", str(pyright),
             str(root / "probe.hy")]
    if case.startswith("nested"):
        # gate の子・pytest の中と同じ — 枠を持つ process(正本の写しの run)の子として撃つ。blocker を解いてから撃つ。
        assert blocker.stdin is not None
        blocker.stdin.close()
        blocker.wait(timeout=15)
        check = [PY, str(canon), "run", "--label", "outer-holds", "--", *check]
    try:
        done = subprocess.run(
            check,
            capture_output=True, text=True, timeout=90, env=env, cwd=str(TREE), check=False,
        )
        elapsed = time.monotonic() - t0
        runs = started.read_text().splitlines() if started.exists() else []
        print(f"[{case}] 宣言 waitLimitSeconds={wait_s}・knob={env.get('DOEFF_HY_CHECK_SLOT_WAIT')}")
        print(f"[{case}] exit={done.returncode}・{elapsed:.1f} 秒・枠が埋まっている間の pyright の起動={runs}")
        tail = [l for l in done.stderr.splitlines() if l.strip()][-4:]
        print(f"[{case}] stderr の末尾:\n    " + "\n    ".join(tail))
    except subprocess.TimeoutExpired:
        print(f"[{case}] 90 秒で終わらない(待ち続けた)")
    finally:
        assert blocker.stdin is not None
        blocker.stdin.close()
        blocker.wait(timeout=15)
        err = blocker.stderr.read() if blocker.stderr else ""
        print(f"[{case}] blocker の終了={blocker.returncode}" + (f"・stderr={err.strip()[:200]!r}" if err.strip() else ""))
        shutil.rmtree(base, ignore_errors=True)


if __name__ == "__main__":
    main(sys.argv[1], float(sys.argv[2]))
