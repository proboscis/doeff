def write(name: str, header: str, done: subprocess.CompletedProcess[str], elapsed: float) -> None:
    command = " ".join(str(part) for part in done.args)
    text = (f"{header}\n$ (cwd {MUT}) {command}\nexit={done.returncode}・{elapsed:.1f} 秒\n"
            f"--- stdout ---\n{done.stdout}\n--- stderr ---\n{done.stderr}\n")
    (EVID / f"{name}.log.txt").write_text(text, encoding="utf-8")
    print(f"{name}: exit={done.returncode}・{elapsed:.1f} 秒")


def shoot(command: list[str], cwd: Path, timeout: int = 900) -> tuple[subprocess.CompletedProcess[str], float]:
    started = time.monotonic()
    done = subprocess.run(command, cwd=cwd, capture_output=True, text=True, timeout=timeout, check=False)
    return done, time.monotonic() - started


def version() -> str:
    head = subprocess.run(["git", "-C", str(MUT), "rev-parse", "HEAD"], capture_output=True, text=True, check=True)
    return head.stdout.strip()


def restore() -> None:
    subprocess.run(["git", "-C", str(MUT), "checkout", "--", "packages/doeff-hy/src"], check=True)
    dirty = subprocess.run(["git", "-C", str(MUT), "status", "--porcelain", "--", "packages"],
                           capture_output=True, text=True, check=True).stdout
    assert dirty == "", dirty


def positive() -> None:
    done, elapsed = shoot([*PYTEST, TEST, "packages/doeff-hy/tests/test_env_places_read.py"], MUT)
    write("P-slot-tests", f"正常例: 変異なしの木 {version()} で枠の検 7 本と env_places の検 4 本", done, elapsed)


def mutants() -> None:
    for name, scenario, edits, tests, expectation in MUTANTS:
        restore()
        for relative, old, new in edits:
            path = MUT / relative
            text = path.read_text(encoding="utf-8")
            assert text.count(old) == 1, (name, relative, old[:50])
            path.write_text(text.replace(old, new), encoding="utf-8")
        diff = subprocess.run(["git", "-C", str(MUT), "diff", "--", "packages/doeff-hy/src"],
                              capture_output=True, text=True, check=True).stdout
        (EVID / f"{name}.diff.txt").write_text(diff, encoding="utf-8")
        done, elapsed = shoot([*PYTEST, *tests], MUT)
        write(name, f"失敗例(シナリオ {scenario}): 木 {version()} に {name}.diff.txt を当てて撃つ。期待 = {expectation}",
              done, elapsed)
    restore()
