#!/bin/bash
# 設計検証 lt-4Z5WD2EKQ4ARF9CPCHPTQG00TF の測り直し。使い方: measure.sh setup | <検の名>
# 写しの木 $M/tree(git archive $REV)の中だけを変え、検の file 1 本を走らせ、記録を $REC/<名>.txt に書く。
set -u
DC=/home/ubuntu/.worktrees/doeff-wt-dc-4z5w
REC=$DC/docs/design-checks/lt-4Z5WD2EKQ4ARF9CPCHPTQG00TF/evidence/checks
M=/tmp/exnc/measure
REV=${REV:-7d56f9307}
T=packages/doeff-traverse/tests/test_examples_run.py
X=packages/doeff-traverse/examples
MAC=packages/doeff-hy/src/doeff_hy/macros.hy
BLIND_B=$DC/docs/design-checks/lt-4Z5WD2EKQ4ARF9CPCHPTQG00TF/blind/b-files

# 変える所(doeff-traverse と macros.hy)を $REV の形へ戻す
fresh() {
  rm -rf "$M/tree/packages/doeff-traverse"
  git -C $DC archive $REV packages/doeff-traverse $MAC | tar -x -C "$M/tree"
  rm -rf "$M/pristine" && mkdir -p "$M/pristine"
  git -C $DC archive $REV packages/doeff-traverse $MAC | tar -x -C "$M/pristine"
}

# 記録: 見出し・版・差分・pytest の出力・終了コード
record() {
  local name=$1 desc=$2; shift 2
  {
    echo "# $name: $desc"
    echo "対象の版: $REV(写しの木 $M/tree)+ 下の差分"
    (cd "$M" && diff -ruN -x __pycache__ -x '*.egg-info' pristine/packages/doeff-traverse tree/packages/doeff-traverse; diff -u pristine/$MAC tree/$MAC)
    echo
    echo "\$ uv run pytest $* -q"
    (cd "$M/tree" && uv run pytest "$@" -q -W ignore::pytest.PytestWarning 2>&1)
    echo "pytest rc=$?"
  } > "$REC/$name.txt" 2>&1
  grep -E "^(FAILED|E +AssertionError)|passed|failed in|pytest rc" "$REC/$name.txt"
}

helper_refactor() {
  mkdir -p "$M/tree/$X/common"
  printf '%s\n' '(import doeff_core_effects.scheduler [scheduled])' '' '(defn with-stack [stack program]' '  (setv body program)' '  (for [h stack]' '    (setv body (h body)))' '  (scheduled body))' > "$M/tree/$X/common/stack.hy"
  for f in ex05_parallel_timing ex06_mediagen_rewrite ex07_proboscis_ema_rewrite; do
    sed -i '/^(defn with-stack \[stack program\]$/,/^  (scheduled body))$/c\(import common.stack [with-stack])' "$M/tree/$X/$f.hy"
  done
}

case $1 in
  setup)
    rm -rf "$M" && mkdir -p "$M/tree" "$REC"
    git -C $DC archive $REV | tar -x -C "$M/tree"
    (cd "$M/tree" && make sync > "$M/sync.log" 2>&1); echo "sync rc=$?" ;;
  c0)
    fresh; record c0-all-green "変えていない木で検の file が全部緑(別の path の木 — S1・S2・S3 の正)" $T ;;
  c1)
    fresh
    sed -i 's/(resume (\* x 10))/(resume (\/\/ 10 (- x 3)))/' "$M/tree/$X/ex05_parallel_timing.hy"
    sed -i 's/"MSFT" 300.0/"MSFX" 300.0/' "$M/tree/$X/ex07_proboscis_ema_rewrite.hy"
    record c1-case-failure-red "ex05・ex07 の件の中だけで失敗させる(終了コード 0 のまま)と test_example_runs が赤(S1 の負)" "$T::test_example_runs" ;;
  c2)
    fresh
    sed -i 's/550 in …s/… in …s/' "$M/tree/$X/expected/ex05_parallel_timing.txt"
    sed -i 's/^Success: 3$/Success: …/; s/^Failed: 2$/Failed: …/' "$M/tree/$X/expected/ex07_proboscis_ema_rewrite.txt"
    sed -i '/^=== item history ===$/q' "$M/tree/$X/expected/ex07_proboscis_ema_rewrite.txt" && printf '…\n' >> "$M/tree/$X/expected/ex07_proboscis_ema_rewrite.txt"
    record c2-broad-any-red "… を件の結果まで広げると、test_example_runs は通るが変異の検が赤(S2 の負)" "$T::test_example_runs" "$T::test_case_failure_shows_in_compared_output" ;;
  c3)
    fresh
    sed -i '/# mock が MSFT の価格を知らない/,+1d' "$M/tree/$T"
    record c3-missing-mutant-red "期待の出力に … を置いた ex07 の変異の行を外すと表の検が赤(S3 の負)" "$T::test_tables_name_examples_as_they_are" ;;
  c4)
    fresh
    sed -i '/# sync-backend が x = 3 の件で/,+1d' "$M/tree/$T"
    sed -i 's|^OUTPUT_NOT_COMPARED: dict\[str, str\] = {$|&\n    "ex05_parallel_timing.hy": "実時間の経過(秒)を印字する",|' "$M/tree/$T"
    rm "$M/tree/$X/expected/ex05_parallel_timing.txt"
    record c4-not-compared-runs-red "件を走らせる ex05 を比べない表へ戻すと test_not_compared_example_runs_no_program が赤(S3 の負)" "$T::test_tables_name_examples_as_they_are" "$T::test_not_compared_example_runs_no_program" ;;
  c5)
    fresh
    cp "$M/tree/$X/ex05_parallel_timing.hy" "$M/tree/$X/ex10_timing_copy.hy"
    cp "$M/tree/$X/expected/ex05_parallel_timing.txt" "$M/tree/$X/expected/ex10_timing_copy.txt"
    sed -i 's|^CASE_FAILURE_MUTANTS: dict\[str, tuple\[str, str\]\] = {$|&\n    "ex10_timing_copy.hy": ("(resume (* x 10))", "(resume (// 10 (- x 3)))"),|' "$M/tree/$T"
    record c5-new-varying-example-green "走るたびに印字が変わる新しい例を、例 1 file + 期待の出力 1 file + 変異の表の 1 行で足すと緑(S3 の正 — 判じの code は変えない)" "$T::test_tables_name_examples_as_they_are" "$T::test_example_runs[ex10_timing_copy.hy]" "$T::test_case_failure_shows_in_compared_output[ex10_timing_copy.hy]" ;;
  c6)
    fresh; helper_refactor
    record c6-blind-a-helper-green "盲検 A の変更(with-stack を examples/common/stack.hy へ寄せる)を当てても、直した版では全部緑" $T ;;
  c7)
    fresh; helper_refactor
    sed -i 's/550 in …s/… in …s/' "$M/tree/$X/expected/ex05_parallel_timing.txt"
    record c7-blind-a-helper-broad-any-red "盲検 A の変更の上で ex05 の … を合計まで広げると、dir ごとの写しの変異の検が赤(写しの場を変えても検が効く)" "$T::test_case_failure_shows_in_compared_output" ;;
  c8)
    fresh
    (cd "$M/tree" && patch -p1 -s < "$BLIND_B/feature-ex07.patch.txt")
    record c8-blind-b-hy-names-red "盲検 B の機能の差分(ex07 に raised in: _hy_anon_17:129 … を印字)を当てると、表の検が Hy の付ける名を名指して赤" "$T::test_tables_name_examples_as_they_are" "$T::test_example_runs[ex07_proboscis_ema_rewrite.hy]" ;;
  c9)
    fresh
    (cd "$M/tree" && patch -p1 -s < "$BLIND_B/feature-ex07.patch.txt")
    sed -i 's/raised in: _hy_anon_17:129 -> fetch_price:95 -> _hy_anon_28:/raised in: …:129 -> fetch_price:95 -> …:/' "$M/tree/$X/expected/ex07_proboscis_ema_rewrite.txt"
    (cd "$M/tree" && patch -p1 -s < "$BLIND_B/macros-pre2871.patch.txt")
    record c9-blind-b-any-names-macro-green "盲検 B の機能で Hy の名の所を … にすると、答えを変えない macro の変更(072dce983 の前の展開)を当てても緑" "$T::test_tables_name_examples_as_they_are" "$T::test_example_runs[ex07_proboscis_ema_rewrite.hy]" "$T::test_case_failure_shows_in_compared_output[ex07_proboscis_ema_rewrite.hy]" ;;
esac
