(require doeff-hy.macros [defk defmcp-tool <-])

(defk scale [price]
  {:pre [(: price float)] :post [(: % float)]}
  (* price 2.0))

(defmcp-tool double-tool "double" [{:name "qty" :type "integer" :description "q"}]
  (* qty 2))

(defmcp-tool price-tool "price" [{:name "price" :type "number" :description "p"}]
  (<- r (scale price))
  r)

(defmcp-tool side-tool "side" [{:name "side" :type "string" :enum ["buy" "sell"] :description "s"}]
  side)

(defmcp-tool default-tool "default" [{:name "limit" :type "integer" :description "l" :default 10}]
  limit)
