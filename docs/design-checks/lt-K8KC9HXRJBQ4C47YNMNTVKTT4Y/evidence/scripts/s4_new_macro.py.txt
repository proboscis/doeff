"""S4(実行時の do を引く新しい macro を足す)の正常例と失敗例。

使い方(doeff の作業木の根で): uv run --no-project python s4_new_macro.py <作業木>
- 正常例: macros.hy に `_runtime-do-import` を通す新しい macro を足す → 綴りの検は通り、`_runtime_do` を点を含む置き場へ
  差し替えて展開すると、新しい macro の import は差し替えた置き場を指す。
- 失敗例: `doeff.do` を自分で綴る新しい macro を足す → 綴りの検が落ち、その macro の名を出す。
どちらも macros.hy を元へ戻す。
"""
import subprocess
import sys
from pathlib import Path

GOOD = '\n\n(defmacro probe-new-macro [f]\n  `(do ~(_runtime-do-import (quote _probe_do)) (_probe_do ~f)))\n'
BAD = '\n\n(defmacro probe-new-macro [f]\n  `(do (import doeff.do [do :as _probe_do]) (_probe_do ~f)))\n'
EXPAND = """
import ast, hy, doeff_hy
from doeff_hy import macros
macros._runtime_do = lambda: macros.RuntimeDo("doeff_hy_table_probe.table", "do")
tree = hy.compiler.hy_compile(hy.read_many('(require doeff-hy.macros [probe-new-macro])\\n(defn g [] 1)\\n(setv wrapped (probe-new-macro g))'), "__main__")
print("  差し替えた置き場の下の展開の import:", [ast.unparse(n) for n in ast.walk(tree) if isinstance(n, ast.ImportFrom) and any(a.name == "do" for a in n.names)])
"""


def spelling(tree: Path) -> str:
    result = subprocess.run(["uv", "run", "pytest", "packages/doeff-hy/tests/test_runtime_do_spelling.py", "-q"],
                            cwd=tree, capture_output=True, text=True, timeout=300)
    lines = [l for l in result.stdout.splitlines() if l.startswith(("E ", "FAILED")) or " passed" in l or " failed" in l]
    return f"rc={result.returncode}\n" + "\n".join(f"  {l}" for l in lines[-4:])


def main(tree: Path) -> None:
    path = tree / "packages/doeff-hy/src/doeff_hy/macros.hy"
    original = path.read_text(encoding="utf-8")
    head = subprocess.run(["git", "-C", str(tree), "rev-parse", "HEAD"], capture_output=True, text=True, check=True)
    print(f"# 版 {head.stdout.strip()}")
    try:
        path.write_text(original + GOOD, encoding="utf-8")
        print("## 正常例: `_runtime-do-import` を通す新しい macro\n  綴りの検:", spelling(tree))
        expand = subprocess.run(["uv", "run", "python", "-c", EXPAND], cwd=tree, capture_output=True, text=True, timeout=300)
        print(expand.stdout.rstrip() or expand.stderr[-600:])
        path.write_text(original + BAD, encoding="utf-8")
        print("## 失敗例: `doeff.do` を自分で綴る新しい macro\n  綴りの検:", spelling(tree))
    finally:
        path.write_text(original, encoding="utf-8")
    status = subprocess.run(["git", "-C", str(tree), "status", "--short", "--", str(path)], capture_output=True, text=True, check=True)
    print(f"# 戻した後の macros.hy の git status: {status.stdout.strip() or '(空)'}")


if __name__ == "__main__":
    main(Path(sys.argv[1]))
