(require doeff-hy.macros [defk defmcp-tool <-])


(defk greet [name times]
  {:pre [(: name str) (: times int)] :post [(: % str)]}
  (* name times))

(defmcp-tool say-hello
  "Say hello"
  [{:name "name" :type "string" :description "who"}
   {:name "times" :type "integer" :description "how many"}
   {:name "loud" :type "boolean" :description "loud?" :required False}]
  (<- r (greet name times))
  r)
