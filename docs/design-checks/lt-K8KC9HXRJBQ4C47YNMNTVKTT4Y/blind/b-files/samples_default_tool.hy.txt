(require doeff-hy.macros [defk defmcp-tool <-])

(defk recent-orders [limit]
  {:pre [(: limit int)] :post [(: % int)]}
  "直近の注文を limit 件まで読む(検体では件数をそのまま返す)"
  limit)

(defmcp-tool list-orders
  "List recent orders"
  [{:name "limit" :type "integer" :description "max rows (default 10)" :required False :default 10}]
  (<- n (recent-orders limit))
  n)
