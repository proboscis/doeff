"""M8(mcp_server_loop の `_tool-result-with-stack`)を本物のまま 1 回呼び、tool の答え #("tool" ok 値) を出す。

使い方: uv run python run_tool_via_m8.py <検体の dir> <module 名> <tool の値の名> '<JSON の arguments>'
"""
import json
import sys
import threading

import doeff_hy  # noqa: F401 — .hy の import を有効にする

samples, module_name, attr, arguments_json = sys.argv[1:5]
sys.path.insert(0, samples)

import importlib

from doeff_agents.handlers.mcp_server_loop import _tool_result_with_stack
from doeff_agents.mcp_server import McpToolRequest, McpToolServer

from doeff import run

tool = getattr(importlib.import_module(module_name), attr)
server = McpToolServer((tool,), port=0)
try:
    req = McpToolRequest(
        tool_name=tool.name,
        arguments=json.loads(arguments_json),  # HTTP の thread が JSON を解いた物と同じ形
        event=threading.Event(),
        holder=[],
    )
    print(arguments_json, "->", run(_tool_result_with_stack(server, [], req)))
finally:
    server.server_close()
