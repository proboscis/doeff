(require doeff-hy.macros [defk defmcp-tool <-])

(defk take [s i n b xs o maybe]
  {:pre [(: s str) (: i int) (: n float) (: b bool) (: xs (get list object))
         (: o (get dict #(str object))) (: maybe (| str None))]
   :post [(: % str)]}
  s)

(defmcp-tool typed-tool
  "typed"
  [{:name "s" :type "string" :description "s"}
   {:name "i" :type "integer" :description "i"}
   {:name "n" :type "number" :description "n"}
   {:name "b" :type "boolean" :description "b"}
   {:name "xs" :type "array" :description "xs"}
   {:name "o" :type "object" :description "o"}
   {:name "maybe" :type "string" :description "maybe" :required False}]
  (<- r (take s i n b xs o maybe))
  r)
