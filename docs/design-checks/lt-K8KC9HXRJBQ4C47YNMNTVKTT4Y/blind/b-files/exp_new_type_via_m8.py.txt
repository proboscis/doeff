"""S3 で M6 に新しい型(例 "null")を足した後に defmcp-tool が組む McpToolDef と同じ物を Python で組み、M8 に通す。
M6 の表は変えていない(この file は M6 の変更の代わりに、展開が作るのと同じ McpParamSchema を直に作る)。"""
import threading

from doeff_agents.handlers.mcp_server_loop import _tool_result_with_stack
from doeff_agents.mcp_server import McpToolRequest, McpToolServer

from doeff import do, run
from doeff.mcp import McpParamSchema, McpToolDef


@do
def handler(cursor: None) -> str:
    return f"ok {cursor!r}"


tool = McpToolDef(
    name="page",
    description="page",
    params=(McpParamSchema(name="cursor", type="null", description="cursor", required=False),),
    handler=handler,
)
server = McpToolServer((tool,), port=0)
try:
    for arguments in ({}, {"cursor": None}):
        req = McpToolRequest(tool_name="page", arguments=arguments, event=threading.Event(), holder=[])
        try:
            print(arguments, "->", run(_tool_result_with_stack(server, [], req)))
        except BaseException as e:  # noqa: BLE001
            print(arguments, "-> raised", type(e).__name__, e)
    req = McpToolRequest(tool_name="page", arguments={"cursor": "abc"}, event=threading.Event(), holder=[])
    try:
        print({"cursor": "abc"}, "->", run(_tool_result_with_stack(server, [], req)))
    except BaseException as e:  # noqa: BLE001
        print({"cursor": "abc"}, "-> raised", type(e).__name__, repr(e))
finally:
    server.server_close()
