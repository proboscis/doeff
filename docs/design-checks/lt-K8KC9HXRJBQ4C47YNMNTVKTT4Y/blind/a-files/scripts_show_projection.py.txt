"""doeff-hy-check の投影(型検査のための展開の Python)の頭と handler の def の行を出す。
使い方: uv run python show_projection.py <root> <.hy>"""
import sys
from pathlib import Path

from doeff_hy.static_check import project

root, source = Path(sys.argv[1]), Path(sys.argv[2])
projection = project(root, [root], source)
text = getattr(projection, "text", None)
if text is None:
    print(projection)
    sys.exit(1)
for line in text.splitlines():
    if line.startswith(("from ", "import ")) or "_mcp_handler(" in line:
        print(line)
