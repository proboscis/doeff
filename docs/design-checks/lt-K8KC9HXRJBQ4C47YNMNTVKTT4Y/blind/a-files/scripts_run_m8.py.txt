"""M8(mcp_server_loop の _tool-result-with-stack)そのものに、agent が送った JSON の引数を渡して tool を走らせる。

使い方: uv run python /tmp/ki973b-blind-a/scripts/run_m8.py <検体の dir> <検体の module 名> <tool の変数名> '<JSON の arguments>' [...]
"""
import importlib
import json
import sys
import threading

import doeff_hy  # noqa: F401
import hy  # noqa: F401
from doeff_agents.handlers.mcp_server_loop import _tool_result_with_stack
from doeff_agents.mcp_server import McpToolRequest, McpToolServer
from doeff_core_effects.scheduler import scheduled

from doeff import run

probe_dir, module_name, tool_var = sys.argv[1:4]
sys.path.insert(0, probe_dir)
module = importlib.import_module(module_name)
tool = getattr(module, tool_var)
import inspect
print("handler の注記:", {n: p.annotation for n, p in inspect.signature(tool.handler).parameters.items()})
print("inputSchema:", json.dumps(tool.input_schema(), ensure_ascii=False))
server = McpToolServer(tools=(tool,))
try:
    for raw in sys.argv[4:]:
        arguments = json.loads(raw)
        req = McpToolRequest(tool_name=tool.name, arguments=arguments, event=threading.Event(), holder=[])
        outcome = run(scheduled(_tool_result_with_stack(server, [], req)))
        print(f"arguments={raw} -> {outcome!r}")
finally:
    server.server_close()
