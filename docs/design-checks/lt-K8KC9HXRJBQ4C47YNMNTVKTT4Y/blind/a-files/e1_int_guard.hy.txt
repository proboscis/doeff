(require doeff-hy.macros [defmcp-tool <-])

;; 書き手の守り: agent が "100" のような文字を送っても本体へ入れない
(defmcp-tool double-qty-guarded
  "Double the quantity (guarded)"
  [{:name "qty" :type "integer" :description "Quantity"}]
  (when (not (isinstance qty int))
    (raise (TypeError "qty must be an integer")))
  (* qty 2))

;; 守りの無い本体: 型検査は qty を int と読む
(defmcp-tool double-qty
  "Double the quantity"
  [{:name "qty" :type "integer" :description "Quantity"}]
  (* qty 2))
