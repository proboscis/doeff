(require doeff-hy.macros [defmcp-tool <-])

;; :enum の値で分ける本体(Literal なら match の網羅を型検査が見る)
(defmcp-tool place-order
  "Place an order"
  [{:name "side" :type "string" :enum ["buy" "sell"] :description "Order side"}
   {:name "qty" :type "integer" :description "Quantity"}]
  (match side
    "buy" qty
    "sell" (- qty)))

;; 書き手の守り: agent が大文字や範囲外の値を送った時に正す・断る
(defmcp-tool place-order-guarded
  "Place an order (guarded)"
  [{:name "side" :type "string" :enum ["buy" "sell"] :description "Order side"}
   {:name "qty" :type "integer" :description "Quantity"}]
  (when (= side "BUY")
    (return qty))
  (match side
    "buy" qty
    "sell" (- qty)
    _ (raise (ValueError (+ "unknown side: " side)))))
