import sys

# A の module の写し(最後の行 contrast() だけを除いた物 — import の時に走らせない)
sys.path.insert(0, "/tmp/kie3a9-dc-a-after")
import exp_consumers as a  # noqa: E402

from doeff_traverse.effects import ItemOutcomeNotPassedError  # noqa: E402

for strategy_name in a.STRATEGIES:
    for name, build in a.CASES:
        bars, bars2 = a.Bars(), a.Bars()
        ledger_r, ledger_p = a.FailureLedger("resume"), a.FailureLedger("pass")
        inside = build(bars, bars2, ledger_r, ledger_p)
        try:
            _, second = a.run_under(a.STRATEGIES[strategy_name](), inside, a._pipeline([1, 2, 3, 4, 5, 6]))
            result = f"valid={second.valid_values}"
        except ItemOutcomeNotPassedError as error:
            result = f"ItemOutcomeNotPassedError(cause={error.cause!r}) — {error.report!r}"
        print(f"[{strategy_name:11}] {name}")
        print(f"    結果   {result}")
        print(f"    bars   {bars.states()}")
        if bars2.made or "2 本目" in name:
            print(f"    bars2  {bars2.states()}")
        print(f"    ledger {ledger_r.rows or ledger_p.rows}")
