"""doeff-linter の基点の比べを、数を見せて行う(hook の check と同じ関数を呼ぶ)。repo の根で走らせる。"""
import sys
from pathlib import Path

top = Path.cwd()
sys.path.insert(0, str(top / "scripts"))
import hook_finding_baseline as h  # noqa: E402

files = sys.argv[1:]
baseline = h.read_baseline(top / h.BASELINE_DIR / "doeff-linter.json")
instrument = h.linter_for_checking(top, baseline)
print("instrument:", instrument)
current = h.measure(top, "doeff-linter", instrument, files)
for path in files:
    now = {rule: by[path] for rule, by in current.items() if path in by}
    base = {rule: by[path] for rule, by in baseline.counts.items() if path in by}
    print(path, "\n   now:", now, "\n   base:", base)
comparison = h.compare(baseline.counts, current, frozenset(files))
print("grown:", comparison.grown, "stale:", comparison.stale)
