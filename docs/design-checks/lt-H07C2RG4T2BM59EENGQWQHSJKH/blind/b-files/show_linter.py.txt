"""doeff-linter の所見を、--force-exclude の有無で見せる(repo の根で走らせる)。"""
import json
import subprocess
import sys

BINARY = "/home/ubuntu/.cache/doeff-linter-snapshots/2de2bbe4d93862069b1e9d501cacc10616ec3ca7/doeff-linter"
files = sys.argv[1:]
for extra in (["--force-exclude"], []):
    out = subprocess.run([BINARY, "--no-log", *extra, "--output-format", "json", *files], capture_output=True, text=True, check=False)
    data = json.loads(out.stdout or "[]")
    items = data if isinstance(data, list) else data.get("violations", data)
    print("== args:", extra, "count:", len(items))
    for i in items:
        print("  ", i.get("severity"), i.get("rule_id") or i.get("rule"), i.get("file_path") or i.get("file"), i.get("line"), (i.get("message") or "")[:120])
