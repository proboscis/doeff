"""行の returns の宣言と実行の時の検め: 続きの後に節へ戻る node(ResumeThrow)の行を returns=False と誤って宣言すると、節は実行の時に
RuntimeError で止まる(黙って通らない)。returns=True なら handle.hy が「再開した」の印を立てて答えが返る。走らせ方: 作業木で
uv run python -I <この file>"""

import sys
import types

import doeff_hy  # noqa: F401
import hy
from doeff_hy import clause_ops
from doeff_hy.clause_ops import ClauseOp, Ending
from hy.models import Expression, Symbol

from doeff import run

SOURCE = """
(require doeff-hy.macros [defk <- handle])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Bump [EffectBase] #^ int n)
(defk guarded [n] {:pre [(: n int)] :post [(: % int)]}
  (try (do (<- y (Bump n)) y) (except [e ValueError] -1)))
(defk subject [] {:pre [] :post [(: % int)]}
  (<- r (handle (guarded 1) (Bump [n] (setv answer (throw (ValueError "boom"))) answer)))
  r)
"""


def rewrite(args, recur):
    return Expression([Symbol("yield"), Expression([Symbol("ResumeThrow"), Symbol("k"), recur(args[0])])])


original = clause_ops.CLAUSE_OPS
for returns in (False, True):
    row = ClauseOp(name="throw", ending=Ending.RESUMES, arity=1, returns=returns, tail=None,
                   nodes=("ResumeThrow",), static_nodes=("ResumeThrow",), rewrite=rewrite)
    clause_ops.CLAUSE_OPS = (*original, row)
    mod = types.ModuleType(f"returns_flag_{returns}")
    sys.modules[mod.__name__] = mod
    try:
        for form in hy.read_many(SOURCE):
            hy.eval(form, mod.__dict__, module=mod)
        print(f"returns={returns}: 答え {run(mod.subject())!r}")
    except Exception as error:  # noqa: BLE001 — 記録の表示だけ
        print(f"returns={returns}: {type(error).__name__}: {str(error)[:120]}")
clause_ops.CLAUSE_OPS = original
