"""S4(storage): defhandler を使う module の展開の記録(doeff_hy_bytecode_guard)が、節の操作の表の行の変更で古いと判じられ、表と関係の
無い行の変更では今に合うままかを見る。doeff_hy を一時の dir へ写し、写しの clause_ops.hy を書き換えて、別の process で照らす。
走らせ方: 作業木で uv run python -I <この file> <作業木>"""

import json
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

CHILD = """
import importlib.machinery, json, sys
from pathlib import Path
root, mode, stored = sys.argv[1], sys.argv[2], Path(sys.argv[3])
sys.path.insert(0, root)
import hy
import doeff_hy
from doeff_hy_bytecode_guard import record_from_json, record_is_current_here, record_to_json
from doeff_hy_bytecode_guard import records, source_to_code_as_import
if mode == "record":
    path = root + "/clause_ops_probe/user.hy"
    loader = importlib.machinery.SourceFileLoader("clause_ops_probe.user", path)
    record = records.record_of(source_to_code_as_import(loader, open(path, "rb").read(), path))
    stored.write_text(json.dumps(record_to_json(record)))
    print(json.dumps({"macros": sorted(row.module + ":" + row.name for row in record.macros), "doeff_hy": doeff_hy.__file__}))
else:
    record = record_from_json(json.loads(stored.read_text()))
    print(json.dumps({"current": record_is_current_here(record), "doeff_hy": doeff_hy.__file__}))
"""

USER = """(require doeff-hy.macros [defhandler])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Bump [EffectBase] #^ int n)
(defhandler bump-handler (Bump [n] (resume (+ n 1))))
"""

worktree = Path(sys.argv[1])
source = worktree / "packages/doeff-hy/src/doeff_hy"


def child(root: Path, mode: str, stored: Path) -> dict:
    done = subprocess.run([sys.executable, "-I", "-c", CHILD, str(root), mode, str(stored)],
                          capture_output=True, text=True, timeout=240, check=False)
    if done.returncode != 0:
        return {"error": done.stderr[-600:]}
    return json.loads(done.stdout.strip().splitlines()[-1])


CASES = {
    "変えない": lambda text: text,
    "表と関係の無い行(module の末尾の註)": lambda text: text + "\n;; 表と関係の無い註\n",
    "表の行(resume の行の nodes に Pass を足す)": lambda text: text.replace(
        ':nodes #("Resume") :static-nodes #() :rewrite _resume)',
        ':nodes #("Resume" "Pass") :static-nodes #() :rewrite _resume)'),
    "表の行の書き換え(transfer の node の綴り)": lambda text: text.replace(
        "`(yield (Transfer k ~value))", "`(yield (Transfer k (do ~value)))"),
}
for label, change in CASES.items():
    with tempfile.TemporaryDirectory() as tmp:
        root = Path(tmp)
        shutil.copytree(source, root / "doeff_hy", ignore=shutil.ignore_patterns("__pycache__"))
        (root / "clause_ops_probe").mkdir()
        (root / "clause_ops_probe/__init__.py").write_text("")
        (root / "clause_ops_probe/user.hy").write_text(USER)
        stored = root / "record.json"
        recorded = child(root, "record", stored)
        table = root / "doeff_hy/clause_ops.hy"
        before = table.read_text()
        after = change(before)
        assert label == "変えない" or after != before, f"{label}: 書き換えが当たらない"
        table.write_text(after)
        checked = child(root, "check", stored)
        print(f"{label}: 記録 {recorded} → 照らし {checked}")
