"""失敗例の確かめ: 新しい one-shot の検めが表を介さずに名を持つ事を、S1(throw を表に足す)と S3(pass を退役)で見る。"""
import importlib.util, re, sys, types
import hy
import doeff_hy
from doeff_hy import clause_ops, handle, clause_endings
from doeff import run

spec = importlib.util.spec_from_file_location(
    "t_clause_ops", "/home/ubuntu/.worktrees/doeff-wt-blind-b-b4c0/tests/test_handle_clause_ops.py")
T = importlib.util.module_from_spec(spec); spec.loader.exec_module(T)

ORIG = clause_ops.CLAUSE_OPS

def refusal(code):
    msg = T._expansion_refusal(code)
    if not msg:
        return "(通った)"
    m = re.search(r"(続き k を 2 度使う[^。]*|Raise の節は \\S+ できない|Absent の節は \\S+ できない)", msg)
    return "断った: " + (m.group(1)[:70] if m else msg.strip().splitlines()[-1][:100])

def subject(clause):
    return f"""
    (defk subject [] {{:pre [] :post [(: % int)]}}
      (<- r (handle (guarded 1) (Bump [n] {clause})))
      r)"""

print("== D2 今の表: 機能は動く(展開の時に断る)")
for c in ["(resume 10) (resume 20)", "(resume 10) (transfer 20)", "(resume 10) (reperform effect)"]:
    print(f"  {c:40s} ->", refusal(subject(c)))

print("== D3 S1: 表に throw の行を足す(tests の _THROW と同じ行・読み手は直さない)")
clause_ops.CLAUSE_OPS = (*ORIG, T._THROW)
try:
    c = '(resume 10) (throw (ValueError "x"))'
    print(f"  {c:40s} ->", refusal(subject(c)))
    print("  比較 (Raise [reason] (throw …))          ->",
          refusal('(setv h (handle (raises) (Raise [reason] (throw (ValueError "x")))))'))
    try:
        mod = T._eval_module(subject(c))
        print("  実行 ->", run(mod.subject()))
    except BaseException as e:
        print("  実行 ->", type(e).__name__, str(e)[:120])
finally:
    clause_ops.CLAUSE_OPS = ORIG

print("== D4 S3: 表から pass の行を消す(退役)")
clause_ops.CLAUSE_OPS = tuple(op for op in ORIG if op.name != "pass")
try:
    print("  named-op('pass') =", clause_ops.named_op("pass"))
    T.test_readers_name_operations_only_through_the_table()
    print("  test_readers_name_operations_only_through_the_table: 通った")
    form = hy.read("(pass)")
    print("  _terminates((pass)) =", handle._terminates(form), "/ clause_ops =", set(clause_endings.clause_ops([form])))
    try:
        handle._check_one_shot("probe", list(hy.read_many("(resume 10) (pass)")))
        print("  _check_one_shot((resume 10) (pass)) -> 通った")
    except SyntaxError as e:
        print("  _check_one_shot((resume 10) (pass)) -> SyntaxError:", str(e)[:90])
finally:
    clause_ops.CLAUSE_OPS = ORIG

print("== 綴りの拾い方が新しい集まりの名を拾っているか")
print("  candidates ⊇ {resume,transfer,reperform,pass}:", {"resume","transfer","reperform","pass"} <= set(T._CANDIDATES))
