import sys, types
import doeff_hy, hy
from doeff_hy import handle
from doeff import run
handle._check_one_shot = lambda *a: None   # 新しい検めを外して、元の展開の実行時の動きだけを見る

HEADER = """
(require doeff-hy.macros [defk <- handle])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Bump [EffectBase] #^ int n)
(defk guarded [n] {:pre [(: n int)] :post [(: % int)]}
  (do (<- y (Bump n)) y))
"""
def ev(code, name):
    mod = types.ModuleType(name); sys.modules[name] = mod
    for f in hy.read_many(HEADER + code):
        hy.eval(f, mod.__dict__, module=mod)
    return mod

mod = ev("""
(defk subject [] {:pre [] :post [(: % int)]}
  (<- r (handle (guarded 1) (Bump [n] (<- result (resume 10)) (transfer result))))
  r)
""", "probe_bind")
try:
    print("(<- result (resume 10)) (transfer result) ->", run(mod.subject()))
except BaseException as e:
    print("raised:", type(e).__name__, str(e)[:200])
