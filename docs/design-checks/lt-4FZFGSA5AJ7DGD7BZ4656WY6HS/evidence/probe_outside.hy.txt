;; 走っているターンが無い時に届いた枠の知らせは outside に積まれ、後から始まったターンの利用枠切れの事実に使われない(持ち越さない —
;; 古い知らせの戻りの刻で口座を早く戻さない)。前のターンで届いた知らせも、次のターンには使われない。読みの thread と同じ順で absorb を呼ぶ。
(require doeff-hy.macros [val])
(import json)
(import doeff [run])
(import doeff_codex.lines [classify-line])
(import doeff_codex.handler [SessionRuntime absorb])
(import doeff_codex.values [CodexSessionSpec CodexHome])
(import doeff_codex.rpc [ApprovalPolicy SandboxMode])

(val spec (CodexSessionSpec :home (CodexHome :env {}) :cwd "/tmp" :approval-policy ApprovalPolicy.NEVER :sandbox SandboxMode.READ-ONLY))
(val runtime (SessionRuntime spec))
(val limits {"method" "account/rateLimits/updated"
             "params" {"rateLimits" {"limitId" "codex" "secondary" None
                                     "primary" {"usedPercent" 100 "windowDurationMins" 300 "resetsAt" 1000}}}})
(defn ended [turn-id status kind]
  {"method" "turn/completed"
   "params" {"threadId" "t1" "turn" {"id" turn-id "status" status
                                     "error" (if kind {"message" "You’ve hit your usage limit." "codexErrorInfo" kind} None)}}})
(val lines [limits                                                         ;; ターンの外(まだどのターンも無い)
            {"id" 3 "result" {"turn" {"id" "u1" "status" "inProgress"}}}
            limits                                                         ;; ターン u1 の中
            (ended "u1" "completed" None)
            {"id" 4 "result" {"turn" {"id" "u2" "status" "inProgress"}}}
            (ended "u2" "failed" "usageLimitExceeded")])                  ;; ターン u2 には知らせが届かない
(for [line lines]
  (with [runtime.lock]
    (run (absorb runtime (run (classify-line (json.dumps line)))))))
(print "outside:" (tuple (gfor record runtime.outside (. (type record) __name__))))
(print "u1 の終わり:" (. (get runtime.turns "u1") end status) (. (get runtime.turns "u1") end usage-limit))
(print "u2 の終わり:" (. (get runtime.turns "u2") end status) (. (get runtime.turns "u2") end usage-limit))
