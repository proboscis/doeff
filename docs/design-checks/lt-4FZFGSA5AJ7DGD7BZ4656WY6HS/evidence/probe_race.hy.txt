;; 競合の決定的な再現: 読みの thread は turn/start の答えを answers に置くだけで、current-turn を立てるのは答えを待つ handler の側。
;; その間に読んだ、ターンを名乗らない枠の知らせがどこへ積まれるかを、読みの thread と同じ順で absorb を呼んで見る。
(require doeff-hy.macros [val])
(import doeff [run])
(import doeff_codex.lines [classify-line RateLimits])
(import doeff_codex.handler [SessionRuntime absorb])
(import doeff_codex.values [CodexSessionSpec CodexHome])
(import doeff_codex.rpc [ApprovalPolicy SandboxMode])

(val spec (CodexSessionSpec :home (CodexHome :env {}) :cwd "/tmp" :approval-policy ApprovalPolicy.NEVER :sandbox SandboxMode.READ-ONLY))
(val runtime (SessionRuntime spec))
(val lines ["{\"id\":3,\"result\":{\"turn\":{\"id\":\"u1\",\"status\":\"inProgress\"}}}"
            "{\"method\":\"account/rateLimits/updated\",\"params\":{\"rateLimits\":{\"limitId\":\"codex\",\"primary\":{\"usedPercent\":100,\"windowDurationMins\":300,\"resetsAt\":1893456000},\"secondary\":null}}}"
            "{\"method\":\"turn/started\",\"params\":{\"threadId\":\"t1\",\"turn\":{\"id\":\"u1\",\"status\":\"inProgress\"}}}"])
(for [line lines]
  (with [runtime.lock]
    (run (absorb runtime (run (classify-line line))))))
(print "current-turn:" runtime.current-turn)
(print "outside:" (tuple (gfor record runtime.outside (. (type record) __name__))))
(print "turn u1 の記録:" (if (in "u1" runtime.turns)
                             (tuple (gfor event (. (get runtime.turns "u1") events) (. (type event.record) __name__)))
                             "無い"))
(print "枠の知らせがターン u1 に載った:" (and (in "u1" runtime.turns)
                                         (any (gfor event (. (get runtime.turns "u1") events) (isinstance event.record RateLimits)))))
