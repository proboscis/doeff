"""シナリオごとの失敗例(変異)を 1 つずつ入れて、狙いの検が赤になる事を確かめ、元に戻す。
使い方: uv run --script mutations.py <変異の木の根> <証拠の dir>。変異の木は着地させない実験の木。"""
import difflib, pathlib, subprocess, sys, datetime

root = pathlib.Path(sys.argv[1]); out = pathlib.Path(sys.argv[2]); out.mkdir(parents=True, exist_ok=True)
C = "packages/doeff-cluster/src/doeff_cluster/"
OWN = "packages/doeff-cluster/tests/test_group_limits_owner.hy::"
DEC = "packages/doeff-cluster/tests/test_group_limits_declared.hy::"
MUTATIONS = [
 ("M-S2-owner", "S2", C + "coordinator/core/resource_policy.hy",
  "(when (and current.system (!= limit.system current.system))", "(when False",
  "資源の口が持ち主の系を比べない(盲検 A の前の形)",
  [OWN + "test_a_group_declared_by_one_system_is_not_rewritten_by_another_system", OWN + "test_the_second_system_declaring_a_group_is_refused_on_sim"]),
 ("M-S2-form", "S2", "packages/doeff-hy/src/doeff_hy/system_form.hy",
  '(setv LIMIT-SCOPES #("worker" "cluster"))', '(setv LIMIT-SCOPES #("worker" "cluster" "node"))',
  "defsystem の形の読みの範囲の名が規則と食い違う(写しだけを直した)",
  [OWN + "test_the_defsystem_form_reads_the_same_scopes_and_keys_as_the_rule"]),
 ("M-S3-store", "S3", C + "coordinator/protocol/durable_kv.hy",
  "(dfor #(k l) (.items state.group-limits) (+ GROUP-LIMIT k) #(#(l) (partial group-limit-spec l)))", "{}",
  "耐久の鍵の保存が組の上限を綴らない(保存先の鍵の組を書き忘れた)",
  [OWN + "test_a_declared_group_and_its_system_survive_the_state_file_and_the_durable_keys"]),
 ("M-S4-tick", "S4", C + "coordinator/core/cluster_policy.hy",
  "(:= held after-hold)", "(:= held held)",
  "同じ拍で置いた分を升の数に足さない",
  [DEC + "test_the_same_tasks_sent_in_either_order_place_the_declared_number_in_one_tick", DEC + "test_two_senders_naming_one_group_are_counted_by_the_one_declared_value"]),
 ("M-S5-sim", "S5", C + "sim/local.hy",
  "(for [limit declaration.limits]", "(for [limit #()]",
  "sim の宣言の書きが組の上限を書かない(本番の declare と分かれる)",
  [DEC + "test_the_state_shows_each_group_declared_value_and_held_count_on_sim", OWN + "test_the_second_system_declaring_a_group_is_refused_on_sim"]),
]
summary = []
ONLY = set(sys.argv[3:])
for mid, scenario, rel, old, new, what, tests in [m for m in MUTATIONS if not ONLY or m[0] in ONLY]:
    path = root / rel
    original = path.read_text()
    assert original.count(old) == 1, (mid, old)
    mutated = original.replace(old, new)
    diff = "".join(difflib.unified_diff(original.splitlines(True), mutated.splitlines(True), "a/" + rel, "b/" + rel, n=1))
    path.write_text(mutated)
    try:
        run = subprocess.run(["timeout", "1500", "uv", "run", "--no-sync", "pytest", *tests, "-v", "--tb=line", "--timeout=1200"],
                             cwd=root, capture_output=True, text=True)
    finally:
        path.write_text(original)
    lines = [l for l in (run.stdout + run.stderr).splitlines()
             if any(w in l for w in ("PASSED", "FAILED", "ERROR", " passed", " failed")) or l.startswith("E ")
             or (".hy:" in l and "Error" in l)]
    failed = sum(1 for l in lines if l.rstrip().endswith("FAILED") or " FAILED" in l and "::" in l and l.startswith("packages"))
    red = run.returncode != 0 and not any(l.rstrip().endswith("PASSED") for l in lines if "::" in l)
    stamp = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    log = [f"# 変異 {mid}(シナリオ {scenario}): {what}", f"# 刻: {stamp}", f"# 狙いの検: {len(tests)} 本 — すべて赤なら、この検はこの違反を捉える",
           f"# 結果: {'すべて赤(捉えた)' if red else '緑が残った(捉えない)'} rc={run.returncode}", "", "## 入れた変異", diff, "## 検の出力",
           *[l[:400] for l in lines], ""]
    (out / f"mutation-{mid}.log").write_text("\n".join(log))
    summary.append(f"{mid} {scenario}: {'red' if red else 'NOT RED'} rc={run.returncode}")
    print(summary[-1], flush=True)
assert all(s.split()[2] == "red" for s in summary), summary
