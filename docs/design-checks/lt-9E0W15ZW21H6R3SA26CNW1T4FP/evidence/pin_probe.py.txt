"""固定の検の各行(送り手の欄の集合・数えの関数の引数)を 1 つずつ比べて出す(どの行が変更を捉えるかを見る)。"""
import inspect
from dataclasses import fields
import hy  # noqa: F401  Hy の module を import できるようにする
from doeff_cluster.shared.intent.detached_model import TaskLimit, SubmitDetached
from doeff_cluster.coordinator.intent.request_bodies import TaskBody
from doeff_cluster.coordinator.core import cluster_policy

PINNED = {
    "TaskLimit": ((f.name for f in fields(TaskLimit)), ("group", "part")),
    "SubmitDetached": ((f.name for f in fields(SubmitDetached)),
                       ("program", "key", "needs", "name", "lease_seconds", "retain_seconds", "environ", "priority", "limits")),
    "TaskBody": ((f.name for f in fields(TaskBody)),
                 ("program", "revision", "name", "needs", "requires", "env", "blob", "versions", "runtime_env", "environ",
                  "lease_seconds", "retain_seconds", "priority", "limits", "format")),
    "hold-limits": (iter(inspect.signature(cluster_policy.hold_limits).parameters), ("held", "limits", "worker", "groups")),
    "limits-misfit": (iter(inspect.signature(cluster_policy.limits_misfit).parameters), ("limits", "worker", "held", "groups")),
}
for name, (actual, pinned) in PINNED.items():
    actual = tuple(actual)
    extra = [a for a in actual if a not in pinned]
    print(f"{name}: {'一致' if actual == pinned else '不一致(赤)'} 足された欄・引数={extra}")
