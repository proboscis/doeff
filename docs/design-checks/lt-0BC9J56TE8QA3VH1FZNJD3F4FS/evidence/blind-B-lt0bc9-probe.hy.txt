;; 盲検 B(lt-0BC9J56TE8QA3VH1FZNJD3F4FS)の probe — 読みの thread と同じ順で absorb を呼び、続きの会話の各ターンの終わりが運ぶ
;; 利用枠切れの事実(UsageLimitHit)を出す。どの木の handler を読んだかも出す(元の木か、差分を当てた写しか)。
(import json)
(import doeff [run])
(import doeff_codex.handler :as handler)
(import doeff_codex.handler [SessionRuntime absorb])
(import doeff_codex.lines [classify-line])
(import doeff_codex.values [CodexSessionSpec CodexHome])
(import doeff_codex.rpc [ApprovalPolicy SandboxMode])

(setv TEXT "You’ve hit your usage limit.")
(setv T-5H-OLD 1893456000)   ; u0 で尽きた 5 時間の窓の戻りの刻(u1 の時にはもう過ぎた刻)
(setv T-7D 1893888000)       ; 7 日の窓の戻りの刻
(setv T-5H-NEW 1893474000)   ; u1 で尽きた 5 時間の窓の戻りの刻

(defn answer [rid turn] {"id" rid "result" {"turn" {"id" turn "status" "inProgress"}}})
(defn started [turn] {"method" "turn/started" "params" {"threadId" "t1" "turn" {"id" turn "status" "inProgress"}}})
(defn window [used mins at] (if (is used None) None {"usedPercent" used "windowDurationMins" mins "resetsAt" at}))
(defn limits [limit-id primary secondary]
  {"method" "account/rateLimits/updated"
   "params" {"rateLimits" {"limitId" limit-id "primary" primary "secondary" secondary}}})
(defn error [turn] {"method" "error" "params" {"error" {"message" TEXT "codexErrorInfo" "usageLimitExceeded"} "willRetry" False
                                               "threadId" "t1" "turnId" turn}})
(defn failed [turn] {"method" "turn/completed"
                     "params" {"threadId" "t1" "turn" {"id" turn "status" "failed"
                                                       "error" {"message" TEXT "codexErrorInfo" "usageLimitExceeded"}}}})

(defn play [name lines]
  (setv runtime (SessionRuntime (CodexSessionSpec :home (CodexHome :env {}) :cwd "/tmp" :approval-policy ApprovalPolicy.NEVER
                                                  :sandbox SandboxMode.READ-ONLY)))
  (for [line lines]
    (run (absorb runtime (run (classify-line (json.dumps line))))))
  (print (.format "== {}" name))
  (for [turn-id ["u0" "u1"]]
    (setv log (get runtime.turns turn-id))
    (print (.format "  {} limits={} usage-limit={}" turn-id (len log.limits) log.end.usage-limit)))
  (print (.format "  outside={}" (len runtime.outside))))

(print "handler の file:" handler.__file__)

;; 筋書き 1: u0 は 5 時間の窓で断られる。u1 は(5 時間の窓が戻った後の)利用枠切れで、u1 の中では枠の知らせが届かない。
(play "1. u1 の中に枠の知らせが無い"
      [(answer 3 "u0") (started "u0") (limits "codex" (window 100 300 T-5H-OLD) (window 37 10080 T-7D)) (error "u0") (failed "u0")
       (answer 4 "u1") (started "u1") (error "u1") (failed "u1")])

;; 筋書き 2: u0 は 7 日の窓で断られる。u1 の知らせは 5 時間の窓だけを名乗る(7 日の窓は名乗らない — null)。
(play "2. u1 の知らせが 7 日の窓を名乗らない"
      [(answer 3 "u0") (started "u0") (limits "codex" (window 40 300 T-5H-OLD) (window 100 10080 T-7D)) (error "u0") (failed "u0")
       (answer 4 "u1") (started "u1") (limits "codex" (window 100 300 T-5H-NEW) None) (error "u1") (failed "u1")])
