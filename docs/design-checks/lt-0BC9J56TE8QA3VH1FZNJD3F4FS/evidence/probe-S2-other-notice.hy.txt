;; S2(窓を運ばない、ターンを名乗らない新しい通知): codex の版上げで増えた通知(例 account/credits/updated — 今の lines では Other)は、
;; handler を変えずにターンの中なら走っているターンの出来事へ、ターンの間(前のターンの終わりの後・次の答えの前)なら outside へ積まれ、
;; 終わったターンにも次のターンにも付かない事を、absorb を読みの順に呼んで確かめる。
(require doeff-hy.macros [val])
(import json)
(import doeff [run])
(import doeff_codex.lines [classify-line])
(import doeff_codex.handler [SessionRuntime absorb])
(import doeff_codex.values [CodexSessionSpec CodexHome])
(import doeff_codex.rpc [ApprovalPolicy SandboxMode])

(val runtime (SessionRuntime (CodexSessionSpec :home (CodexHome :env {}) :cwd "/tmp" :approval-policy ApprovalPolicy.NEVER
                                               :sandbox SandboxMode.READ-ONLY)))
(val credits {"method" "account/credits/updated" "params" {"balance" 3}})
(for [line [{"id" 3 "result" {"turn" {"id" "u0" "status" "inProgress"}}}
            {"method" "turn/started" "params" {"threadId" "t1" "turn" {"id" "u0" "status" "inProgress"}}}
            credits
            {"method" "turn/completed" "params" {"threadId" "t1" "turn" {"id" "u0" "status" "completed"}}}
            credits
            {"id" 4 "result" {"turn" {"id" "u1" "status" "inProgress"}}}
            {"method" "turn/started" "params" {"threadId" "t1" "turn" {"id" "u1" "status" "inProgress"}}}
            {"method" "turn/completed" "params" {"threadId" "t1" "turn" {"id" "u1" "status" "completed"}}}]]
  (run (absorb runtime (run (classify-line (json.dumps line))))))
(print "u0 の出来事の型:" (lfor event (. (get runtime.turns "u0") events) (. (type event.record) __name__)))
(print "u1 の出来事の型:" (lfor event (. (get runtime.turns "u1") events) (. (type event.record) __name__)))
(print "outside:" runtime.outside)
