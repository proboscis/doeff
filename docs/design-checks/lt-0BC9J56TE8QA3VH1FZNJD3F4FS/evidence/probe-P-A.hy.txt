;; P-A(盲検 A の筋書きを、直した版の上で lines と limits だけを変えて受けた時): 窓を運ぶ新しい記録 AccountSnapshot(lines が名乗り
;; LimitRecord に入れた・limits の merged-limits が重ね方を受けた)が、handler を変えずに走っているターンの枠に重なり、利用枠切れの
;; 事実と adapter の AgentAccountLimit が戻りの刻を運ぶ事を、absorb を読みの順に呼んで確かめる。
(require doeff-hy.macros [val])
(import json)
(import doeff [run])
(import doeff_codex.lines [classify-line AccountSnapshot LimitRecord RateLimits RateLimitWindow])
(import doeff_codex.handler [SessionRuntime absorb])
(import doeff_codex.values [CodexSessionSpec CodexHome])
(import doeff_codex.rpc [ApprovalPolicy SandboxMode])
(import doeff_agents.handlers.headless_codex [account-limit-of])
(import doeff_codex.handler :as handler-module)

(val TEXT "You’ve hit your usage limit.")
(val AT 1893456000)
(val runtime (SessionRuntime (CodexSessionSpec :home (CodexHome :env {}) :cwd "/tmp" :approval-policy ApprovalPolicy.NEVER
                                               :sandbox SandboxMode.READ-ONLY)))
(for [line [{"id" 3 "result" {"turn" {"id" "u1" "status" "inProgress"}}}
            {"method" "turn/started" "params" {"threadId" "t1" "turn" {"id" "u1" "status" "inProgress"}}}]]
  (run (absorb runtime (run (classify-line (json.dumps line))))))
(run (absorb runtime (AccountSnapshot :limits #((RateLimits :limit-id "codex"
                                                            :primary (RateLimitWindow :used-percent 37 :minutes 10080 :resets-at (+ AT 432000))
                                                            :secondary None)
                                                (RateLimits :limit-id "codex_other"
                                                            :primary (RateLimitWindow :used-percent 100 :minutes 300 :resets-at AT)
                                                            :secondary None)))))
(run (absorb runtime (run (classify-line (json.dumps {"method" "turn/completed"
                                                      "params" {"threadId" "t1" "turn" {"id" "u1" "status" "failed"
                                                                                        "error" {"message" TEXT "codexErrorInfo" "usageLimitExceeded"}}}})))))
(val log (get runtime.turns "u1"))
(print "handler の file:" handler-module.__file__)
(print "LimitRecord:" LimitRecord)
(print "u1 の出来事の型:" (lfor event log.events (. (type event.record) __name__)))
(print "outside:" runtime.outside)
(print "u1.limits の枠:" (lfor one log.limits one.limit-id))
(print "u1.end.usage-limit:" log.end.usage-limit)
(print "AgentAccountLimit:" (run (account-limit-of log.end.usage-limit)))
