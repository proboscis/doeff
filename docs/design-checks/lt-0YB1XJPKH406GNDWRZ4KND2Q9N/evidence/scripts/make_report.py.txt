import hashlib

from pathlib import Path

root = Path(sys.argv[1])


def ref(rel: str) -> dict[str, str]:
    """記録の dir からの相対 path と、その file の sha256。"""
    return {"path": rel, "sha256": hashlib.sha256((root / rel).read_bytes()).hexdigest()}


def refs(glob: str) -> list[dict[str, str]]:
    return [ref(str(p.relative_to(root))) for p in sorted(root.glob(glob))]
