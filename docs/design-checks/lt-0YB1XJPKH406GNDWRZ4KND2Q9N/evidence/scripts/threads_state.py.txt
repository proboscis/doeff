"""S4 の正常例: 定義の macro の検体ごとに、関数の中に書いた外の関数を 8 本の thread から並べて 5 回ずつ呼び、
定義を書いた module と doeff の module の状態(検の _module_state_view)が初めの呼びの後から変わらない事を見る。
使い方(作業木を cwd に): uv run --frozen python threads_state.py <作業木の tests/test_nested_definition_scope.py>"""
import importlib.util
import sys
from concurrent.futures import ThreadPoolExecutor

spec = importlib.util.spec_from_file_location("nested_scope_tests", sys.argv[1])
tests = importlib.util.module_from_spec(spec)
spec.loader.exec_module(tests)

failed = 0
for macro, (source, defined) in tests._DEFINITION_SAMPLES.items():
    mod = tests._eval_module(f"(defn subject [] {source} None)")
    mod.subject()
    before = tests._module_state_view(mod)
    with ThreadPoolExecutor(max_workers=8) as pool:
        list(pool.map(lambda _: [mod.subject() for _ in range(5)], range(8)))
    after = tests._module_state_view(mod)
    changed = sorted(k for k in before.keys() | after.keys() if before.get(k) != after.get(k))
    failed += bool(changed)
    print(f"{macro}: 8 thread × 5 回の呼びの後に変わった (module, 名) = {changed}")
print("変わった検体の数 =", failed)
sys.exit(1 if failed else 0)
