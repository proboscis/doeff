"""盲検 A の実験: 同じ source を、実行時の展開と型検査のための展開(doeff-hy-check の project)で Python にし、
関数の中の deftest / defbench の pytest の item の記録の文と、記録の import が外の関数の局所の名を作るかを見る。
使い方(作業木を cwd に): DOEFF_BLIND_A_BENCH_GUARD=1 uv run --frozen python /tmp/lt0YB1-blind-a/probe_static_view.py
"""

import ast
import symtable
import types
from pathlib import Path

import hy
from hy.compiler import hy_compile

from doeff_hy.static_check import project

root = Path("/tmp/lt0YB1-blind-a/static")
source = root / "nested_items.hy"
text = source.read_text()


def report(label: str, rendered: str) -> None:
    tree = ast.parse(rendered)
    fn = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "make_tests")
    body_src = ast.unparse(fn)
    print(f"== {label}: make_tests の本体に record_at_import が在る: {'record_at_import' in body_src}")
    print(f"== {label}: module の直下に record_at_import が在る: "
          f"{any('record_at_import' in ast.unparse(s) for s in tree.body if not isinstance(s, ast.FunctionDef))}")
    table = symtable.symtable(rendered, str(source), "exec")
    make = next(c for c in table.get_children() if c.get_name() == "make_tests")
    local = sorted(s.get_name() for s in make.get_symbols() if s.is_local() and not s.get_name().startswith("_hy_"))
    print(f"== {label}: make_tests の局所の名(_hy_ の頭を除く): {local}")
    print(f"== {label}: doeff_hy は make_tests の局所か: {make.lookup('doeff_hy').is_local()}")
    print("---- make_tests ----")
    print(body_src)


# 実行時の展開
mod = types.ModuleType("nested_items")
runtime = hy_compile(hy.read_many(text, filename=str(source)), mod, filename=str(source), source=text)
report("実行時の展開", ast.unparse(runtime))

# 型検査のための展開(doeff-hy-check の 1 file の投影。記帳を外す後処理込み)
projection = project(root, [root], source)
report("型検査のための展開", projection.text)
