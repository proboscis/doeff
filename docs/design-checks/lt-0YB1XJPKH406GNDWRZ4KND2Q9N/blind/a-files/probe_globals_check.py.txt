"""M7 の globals の検(test_nested_definition_leaves_module_globals_unchanged)を、defbench の _CASES 相当の入力で直に呼ぶ。
S2 の強制(検体の数え)は _DEFINITION_SAMPLES の 1 行しか求めず、この検の入力 _CASES への追加は求めない事を見るため。
使い方(作業木を cwd に): uv run --frozen python /tmp/lt0YB1-blind-a/probe_globals_check.py
"""

import importlib.util
import os
import sys

spec = importlib.util.spec_from_file_location("t_nested", os.path.join(os.getcwd(), "tests/test_nested_definition_scope.py"))
t = importlib.util.module_from_spec(spec)
sys.modules["t_nested"] = t
spec.loader.exec_module(t)

print("DOEFF_BLIND_A_BENCH_GUARD =", os.environ.get("DOEFF_BLIND_A_BENCH_GUARD"))
print("_DEFINITION_SAMPLES に defbench:", "defbench" in t._DEFINITION_SAMPLES)
print("_CASES に defbench を使う入力:", any("defbench" in src for src, _, _ in t._CASES.values()))
source = """
(defn subject [n]
  (defbench test-bench-inner (sum (range n)))
  n)
"""
try:
    t.test_nested_definition_leaves_module_globals_unchanged(source, lambda mod: mod.subject(3), 3)
    print("globals の検: 緑")
except AssertionError as exc:
    print("globals の検: 赤(AssertionError — module の globals が変わった)")
