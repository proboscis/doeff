"""補足の実験: class の本体の deftest は _in-function-body? が偽なので、module の段の関数として記録される。
使い方(作業木を cwd に): uv run --frozen python /tmp/lt0YB1-blind-a/probe_class_body.py
"""
import sys
import types

import doeff_hy.macros  # noqa: F401
import hy
from doeff_adr.lazy_collection import verify_records

mod = types.ModuleType("blind_a_class_body")
mod.__file__ = "/tmp/lt0YB1-blind-a/class_body.hy"
sys.modules[mod.__name__] = mod
for form in hy.read_many("(require doeff-hy.macros [deftest])\n(defclass Group [] (deftest test-in-class (assert True)))"):
    hy.eval(form, mod.__dict__, module=mod)
print("記録:", vars(mod).get("__doeff_pytest_items__"))
print("module に test_in_class が在る:", hasattr(mod, "test_in_class"))
print("verify_records の problems:", verify_records(mod, lambda a: a.startswith("test")).problems)
