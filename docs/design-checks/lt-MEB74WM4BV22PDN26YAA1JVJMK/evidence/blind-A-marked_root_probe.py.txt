"""S1 の後の 2 つの venv の置き場から、入口の検めの root の辿り(runtime_facts.marked-root)が何を返すかを見る。"""
import os
import sys

import hy  # noqa: F401
from doeff import run, with_handlers
from doeff_core_effects.os_file import os_file_handler
from doeff_cluster.shared.protocol.runtime_facts import marked_root

for start in sys.argv[1:]:
    real = os.path.realpath(start)  # ReadInterpreter は realpath(sys.prefix) を答える
    print(f"prefix={real} -> marked-root={run(with_handlers([os_file_handler], marked_root(real)))}")
