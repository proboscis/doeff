;;   (1) 実行環境の service(once でない job)にも job-launch が entry-env(root の venv の 2 つ)を作るか — env の値の面の主張。
;;   (2) 今の道の選び forks-from-warm-child が service を分けない事。
;;   (3) 冷えた道の shim が足す退きの知らせの fd の名(宿の契約 HOST-CONTRACT の notice-env)— 分かれる道には無い(盲検 A の指摘)。
;; 走らせ方: doeff の根で `uv run --no-sync hy <この file>`(packages/doeff-cluster を探し道の頭に足す)。
(import sys)
(.insert sys.path 0 "packages/doeff-cluster")
(import json)
(import doeff [run])
(import doeff_cluster.foundation.host_contract [HOST-CONTRACT])
(import doeff_cluster.shared.intent.job_model [JobSpec])
(import doeff_cluster.shared.core.runtime_env_rules [runtime-env->json])
(import doeff_cluster.worker.intent.worker_model [CodeLayout WorkerPolicy])
(import doeff_cluster.worker.core.launch [job-launch])
(import doeff_cluster.worker.core.warm_rules [forks-from-warm-child])
(import doeff_cluster.worker.core.shim_timing [shim-spans])
(import tests.env_fixtures [LOCK env-of])

(setv env (run (env-of "app" "lib" LOCK)))
(setv text (json.dumps (run (runtime-env->json env)) :sort-keys True))
(setv service (JobSpec "svc" "app.main" #("service") "rev1" :runtime-env text :env-key "k1"))
(setv task (JobSpec "task/t1" "app.main" #("task") "rev1" :once True :runtime-env text :env-key "k1"))
(setv shim (run (shim-spans (WorkerPolicy))))
(for [spec #(service task)]
  (setv plan (run (job-launch spec "/roots/env-k1" "1-x" 1 :python "/py" :hy-command "/bin/hy" :uv "uv" :extra-env {}
                              :layout (CodeLayout) :allowed-env {"PATH" "/boot/.venv/bin:/usr/bin" "HOME" "/home/w"} :worker-pid 1
                              :program-path None :program-env HOST-CONTRACT.program-env :work-dir "/jobs/x"
                              :shim-grace-ms shim.shim-grace-ms :notice-env HOST-CONTRACT.notice-env)))
  (setv entry (dfor e plan.entry-env e.name e.value))
  (print (.format "{}: once={} forks-from-warm-child={} entry-env PATH={} VIRTUAL_ENV={} env に VIRTUAL_ENV={}"
                  spec.name spec.once (forks-from-warm-child spec) (get entry "PATH") (.get entry "VIRTUAL_ENV")
                  (in "VIRTUAL_ENV" (sfor e plan.env e.name)))))
(print (.format "冷えた道の shim が job へ渡す退きの知らせの fd の名(HOST-CONTRACT.notice-env)= {}" HOST-CONTRACT.notice-env))
