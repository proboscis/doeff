"""設計検証 lt-MEB74WM4BV22PDN26YAA1JVJMK: 2 つの道で入口が見た環境の名の違いを 1 度測るため(検 test_task_venv_on_both_paths.hy の部品をそのまま使う)。

走らせ方: doeff の根で `PYTHONDONTWRITEBYTECODE=1 uv run --no-sync python <この file>`(python で起こす — 検は sys.executable で shim と boot の
venv の代わりを起こすので、hy の起動 script を sys.executable にしない)。
"""

import os
import shutil
import sys
import tempfile

sys.path.insert(0, "packages/doeff-cluster")
import hy  # noqa: E402,F401 — Hy の検の module を読む importer を据える

from tests.test_task_venv_on_both_paths import REPO, both_paths_seen, on_real_processes  # noqa: E402

place = tempfile.mkdtemp(prefix="venv-", dir="/tmp")
os.makedirs(place + "/root")
os.symlink(str(REPO), place + "/root/app")
seen = on_real_processes(place, place + "/warm", both_paths_seen(place, place + "/warm"))
shutil.rmtree(place, ignore_errors=True)
cold = seen[0]["env"]
warm = seen[1]["env"]
print("冷えた道にだけ在る名:", sorted(set(cold) - set(warm)))
print("分かれる道にだけ在る名:", sorted(set(warm) - set(cold)))
print("同じ名で値が違う名:", sorted(k for k in set(cold) & set(warm) if cold[k] != warm[k]))
