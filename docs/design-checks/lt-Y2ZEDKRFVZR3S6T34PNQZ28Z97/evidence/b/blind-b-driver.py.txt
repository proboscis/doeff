"""盲検 B の最小の実験: 記録の印の違う 2 つの世代の作業木(import の口)が、1 つの保存先で新しい木を交互に読む。

A = 27f354830 の loader_hooks(変更なし)・B = 提案の差分(_drop_older_generations)を入れた loader_hooks。
6 = records.RECORD_TAG が .../6(今の版)・7 = .../7 に上げた写し(S3 の印の上げだけ — 欄は足さない)。
macro の中身は両方の世代で同じ(違いは記録の印だけ)。展開の回数は macro が木の expansions.log に書く行の数。

使い方: uv run --no-project python driver.py <実験の dir> <scenario>
  scenario = s3   : 世代 6 と 7 を交互に 3 巡(A と B)
  scenario = same : 同じ世代 6 の 2 つの書き手(macro の中身 1 と 5 — 既存の検 test_two_macro_versions の筋書き)を B で 3 巡
"""

import os
import shutil
import subprocess
import sys
from pathlib import Path

HY = "1.3.1"  # uv.lock の hy の版
PYTHON = "3.14"

LOADER = """\
import sys
sys.path.insert(0, {root!r})
sys.dont_write_bytecode = True
sys.path.insert(0, {guard_root!r})
import doeff_hy_bytecode_guard
import doeff_hy_bytecode_guard.records
import doeff_hy_bytecode_guard.macro_use
doeff_hy_bytecode_guard.install()
sys.dont_write_bytecode = False
import hy
import pkg.user
print(hy.__version__, pkg.user.value)
"""

NOT_PASSED = ("VIRTUAL_ENV", "CONDA_PREFIX", "PYTHONDONTWRITEBYTECODE", "PYTHONPYCACHEPREFIX", "PYTHONPATH")


def write_package(root: Path, add: int) -> None:
    package = root / "pkg"
    package.mkdir(parents=True)
    (package / "__init__.py").write_text("")
    (package / "helpers.hy").write_text("(defn base [] 10)\n")
    (package / "macros.hy").write_text(
        "(import pkg.helpers [base])\n"
        '(defmacro answer [] (with [f (open "expansions.log" "a")] (.write f "x\\n"))'
        f" (+ (base) {add}))\n"
    )
    (package / "user.hy").write_text("(require pkg.macros [answer])\n(setv value (answer))\n")


def run(root: Path, guard_root: Path, store: Path) -> tuple[int, int, list[str]]:
    script = root / "load.py"
    script.write_text(LOADER.format(root=str(root), guard_root=str(guard_root)))
    env = {k: v for k, v in os.environ.items() if k not in NOT_PASSED}
    env["DOEFF_HY_CODE_STORE"] = str(store)
    done = subprocess.run(
        ["uv", "run", "--no-project", "--isolated", "--python", PYTHON, "--with", f"hy=={HY}", "python", str(script)],
        capture_output=True, text=True, cwd=root, env=env, timeout=300, check=False,
    )
    if done.returncode != 0:
        raise SystemExit(f"child failed: {done.stderr[-3000:]}")
    value = int(done.stdout.split()[-1])
    log = root / "expansions.log"
    expansions = len(log.read_text().splitlines()) if log.exists() else 0
    named = [line for line in done.stderr.splitlines() if "doeff-hy:" in line]
    return value, expansions, named


def entry_count(store: Path) -> int:
    # 鍵の印 3 の並び <store>/<2 字>/<鍵の残り>/<指紋>.code の数(実験の観測 — 判断ではない)
    return len(list(store.glob("*/*/*.code")))


def main() -> None:
    base, scenario = Path(sys.argv[1]), sys.argv[2]
    if scenario == "s3":
        plans = {variant: ((f"{variant}6", 1), (f"{variant}7", 1)) for variant in ("A", "B")}
    else:
        plans = {"B-same": (("B6", 1), ("B6", 5))}
    for name, writers in plans.items():
        work = base / f"run-{scenario}-{name}"
        if work.exists():
            shutil.rmtree(work)
        store = work / "store"
        for turn in (1, 2, 3):
            for variant, add in writers:
                root = work / f"{variant}-add{add}-turn{turn}"
                write_package(root, add)
                value, expansions, named = run(root, base / variant / "src", store)
                print(f"{name} turn={turn} writer={variant}(add {add}) value={value} expansions={expansions} "
                      f"code_entries_after={entry_count(store)} named_lines={len(named)}", flush=True)
                for line in named:
                    print("    " + line[:200], flush=True)


if __name__ == "__main__":
    main()
