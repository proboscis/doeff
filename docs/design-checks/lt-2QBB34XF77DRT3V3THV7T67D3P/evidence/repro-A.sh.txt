#!/bin/sh
# 盲検 A の再現(設計検証 lt-2QBB34XF77DRT3V3THV7T67D3P): 門を持つ doeff-linter(86ecd134 の断面)を、枠 K = 1・待ちの上限 2 秒の
# 一時の HOME で、枠が埋まっている間に日次の段 doeff-linter-agentcli の judge から撃つ。段の版 2 つ(直す前 f333bd2f・直した後
# f8a49209)の答えと段の終了コードを出す。共有の file には書かない(一時の dir は mktemp -d)。
set -u
LINTER=/home/ubuntu/.cache/doeff-linter-snapshots/86ecd134449adb6d33e3a2ce348c9bc01d96ba02/doeff-linter
DOT=/home/ubuntu/dotfiles
T=$(mktemp -d -p /home/ubuntu/ki-ce35-IcfP repro-A-XXXX)
mkdir -p "$T/home/dotfiles/agent/tests" "$T/repo/agentcli/src"
git -C "$DOT" show f8a49209:agent/tests/machine_check_slots.py > "$T/home/dotfiles/agent/tests/machine_check_slots.py"
printf '{"slots": 1, "dir": "%s/slots", "waitLimitSeconds": 2}\n' "$T" > "$T/home/dotfiles/agent/tests/machine_check_slots.json"
printf '[tool.doeff-linter]\nenable = ["DOEFF016"]\n' > "$T/repo/agentcli/pyproject.toml"
printf 'x = 1\n' > "$T/repo/agentcli/src/x.py"
echo "linter: $("$LINTER" --version)"
HOME="$T/home" uv run --no-project python "$T/home/dotfiles/agent/tests/machine_check_slots.py" run --label long-check -- sleep 30 &
holder=$!
i=0
while ! grep -q "^pid " "$T/slots/slot-0.lock" 2>/dev/null && [ "$i" -lt 100 ]; do i=$((i + 1)); sleep 0.1; done
echo "保持者: $(cat "$T/slots/slot-0.lock")"
for ver in f333bd2f f8a49209; do
  mkdir -p "$T/stage-$ver/agent/tests"
  git -C "$DOT" show "$ver:agent/tests/doeff_linter_agentcli.py" > "$T/stage-$ver/agent/tests/doeff_linter_agentcli.py"
  git -C "$DOT" show "$ver:agent/tests/machine_check_slots.py" > "$T/stage-$ver/agent/tests/machine_check_slots.py"
  echo "== 段の版 $ver"
  HOME="$T/home" sh "$DOT/agent/venv-run.sh" "$DOT/agentcli" python -c "
import sys
from pathlib import Path
from agentcli.ai.core.module_from_path import load_module_from_path
stage = load_module_from_path(Path('$T/stage-$ver/agent/tests/doeff_linter_agentcli.py'), name='stage_$ver')
answer = stage.judge(stage.Linter(path=Path('$LINTER'), pin='0' * 40), Path('$T/repo'))
print('答え:', answer)
print('報告:', stage.report(answer))
print('段の終了コード:', stage.exit_code(answer))
" 2>&1 | grep -v "^  $"
done
kill "$holder" 2>/dev/null
wait "$holder" 2>/dev/null
echo "一時の dir: $T"
