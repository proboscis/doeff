#!/bin/sh
# 設計検証 lt-2QBB34XF77DRT3V3THV7T67D3P の変異の実験(doeff の check_slot.rs に変異を当て、check_slot の検を走らせて log を残し、元へ戻す)と、
# 正本の版を替えた HOME での正常例(linter は変えずに、取り方の違う正本の版 2 つで同じ検が緑)。作業木 = doeff-wt-ki-ce35-mut(origin/main a3c8aa86)。
set -u
WT=/home/ubuntu/.worktrees/doeff-wt-ki-ce35-mut
SRC=$WT/packages/doeff-linter/src/check_slot.rs
OUT=/home/ubuntu/ki-ce35-IcfP/mut-logs
mkdir -p "$OUT"
cp "$SRC" "$OUT/check_slot.rs.orig"
export CARGO_TARGET_DIR=/home/ubuntu/ki-ce35-IcfP/target

run_tests() {  # $1 = log の名 — 結合の検と単体の検を走らせる
  ( cd "$WT/packages/doeff-linter" && cargo test --locked -j 8 --test check_slot && cargo test --locked -j 8 --lib check_slot:: ) > "$OUT/$1.log" 2>&1
  echo "== $1: 終了 $?"; grep -E "^test |test result" "$OUT/$1.log"
}

mutate() {  # $1 = 名, $2 = python の置換(引数 p = check_slot.rs の文字列)
  uv run --no-project python -c "import pathlib,sys; p=pathlib.Path('$SRC'); t=p.read_text(encoding='utf-8'); u=$2; assert u!=t, '変異が当たらない'; p.write_text(u, encoding='utf-8')"
  diff "$OUT/check_slot.rs.orig" "$SRC" > "$OUT/$1.diff.txt"
  run_tests "$1"
  cp "$OUT/check_slot.rs.orig" "$SRC"
}

# 正常例: 変えていない版
run_tests base

# 失敗例 S1・S2: 置き場の値を linter に写す
mutate mutant-S1S2-copied-dir "t.replace('pub const CANON_UNDER_HOME', 'const SLOTS_DIR: &str = \"/tmp/agora-check-slots\";\n\npub const CANON_UNDER_HOME', 1)"
# 失敗例 S3: 取り方(枠の file の名の組み方)を linter に写す
mutate mutant-S3-copied-acquisition "t.replace('pub const CANON_UNDER_HOME', 'fn slot_file(dir: &str, index: usize) -> String {\n    format!(\"{dir}/slot-{index}.lock\")\n}\n\npub const CANON_UNDER_HOME', 1)"
# 失敗例 S4: 新しい mode の足し方を誤って、repo 全体の editor-json を軽い側へ倒す
mutate mutant-S4-heavy-mode-as-light "t.replace('mode.affected_tests || mode.stdin {', 'mode.affected_tests || mode.stdin || mode.editor_json {', 1)"

# 正常例 S3: linter は変えずに、取り方の違う正本の版(c97aefc38 = 親の最初の版・f8a49209 = 今の master)を HOME に置いて同じ検
for ver in c97aefc38 f8a49209; do
  H=$(mktemp -d -p /home/ubuntu/ki-ce35-IcfP canon-home-XXXX)
  mkdir -p "$H/dotfiles/agent/tests"
  git -C /home/ubuntu/dotfiles show "$ver:agent/tests/machine_check_slots.py" > "$H/dotfiles/agent/tests/machine_check_slots.py"
  git -C /home/ubuntu/dotfiles show "$ver:agent/tests/machine_check_slots.json" > "$H/dotfiles/agent/tests/machine_check_slots.json"
  ( cd "$WT/packages/doeff-linter" && HOME="$H" cargo test --locked -j 8 --test check_slot ) > "$OUT/canon-$ver.log" 2>&1
  echo "== canon-$ver: 終了 $?"; grep -E "^test |test result" "$OUT/canon-$ver.log"
done
git -C "$WT" status -s
