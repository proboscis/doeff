"""defmcp-tool の展開を 3 つの場所 × 2 つの展開(実行時・型検査)で出す。"""
import ast

import doeff_hy  # noqa: F401
import hy
from doeff_hy.static_view import static_view

TOOL = """(defmcp-tool echo-tool
  "Echo"
  [{:name "text" :type "string" :description "text"}]
  (<- r (Pure text))
  r)"""
PLACES = {
    "module": TOOL,
    "class": f"(defclass Box []\n{TOOL})",
    "function": f"(defn outer []\n{TOOL}\n echo-tool)",
}


def expand(body: str) -> str:
    source = "(require doeff-hy.macros [defmcp-tool <-])\n" + body
    return ast.unparse(hy.compiler.hy_compile(hy.read_many(source), "__main__"))


for place, body in PLACES.items():
    print(f"===== {place} / runtime")
    print(expand(body))
    with static_view():
        print(f"===== {place} / static")
        print(expand(body))
