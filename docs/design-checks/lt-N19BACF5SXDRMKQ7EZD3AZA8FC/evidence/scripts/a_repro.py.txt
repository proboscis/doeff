"""盲検 A の再現: defmcp-tool を N 個(module の直下か class の本体)並べ、最後の tool の `.name`(str)を int と注記した関数から
返す検体を doeff-hy-check(既定の設定)にかける。McpToolDef の型が見えていれば reportReturnType が 1 つ出る。

使い方: uv run python -I a_repro.py <置き場の dir> <N> [<N> ...]
"""
import contextlib
import io
import json
import re
import sys
from pathlib import Path

from doeff_hy.static_check import Projection, main, project

TOOL = (
    "{indent}(defmcp-tool tool-{index}\n"
    '{indent}  "tool {index}"\n'
    '{indent}  [{{:name "text" :type "string" :description "text"}}]\n'
    "{indent}  (<- r (Pure text))\n"
    "{indent}  r)"
)


def source(count: int, placement: str) -> str:
    head = "(require doeff-hy.macros [defmcp-tool <-])\n(import doeff [Pure])\n\n"
    if placement == "module":
        tools = "\n".join(TOOL.format(indent="", index=i) for i in range(count))
        last = f"tool-{count - 1}"
    else:
        tools = "(defclass Box []\n" + "\n".join(TOOL.format(indent="  ", index=i) for i in range(count)) + ")"
        last = f"Box.tool-{count - 1}"
    return head + tools + f"\n\n(defn #^ int last-tool-name [] (. {last} name))\n"


base = Path(sys.argv[1])
for count in map(int, sys.argv[2:]):
    for placement in ("module", "class"):
        root = base / f"{placement}{count}"
        root.mkdir(parents=True, exist_ok=True)
        path = root / "tools.hy"
        path.write_text(source(count, placement), encoding="utf-8")
        projection = project(root, [root], path)
        assert isinstance(projection, Projection), projection
        mcp_imports = len(re.findall(r"^\s*from doeff\.mcp import .*$", projection.text, re.MULTILINE))
        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            main(["--root", str(root), "--json", str(path)])
        found = [d["rule"] for d in json.loads(out.getvalue()) if d["severity"] == "error"]
        print(f"{placement} N={count}: 投影の 'from doeff.mcp import' の文 = {mcp_imports} / 赤 = {found}")
