def ref(path: str) -> dict[str, str]:
    data = (ROOT / path).read_bytes()
    return {"path": path, "sha256": hashlib.sha256(data).hexdigest()}


def refs(*paths: str) -> list[dict[str, str]]:
    return [ref(p) for p in paths]
