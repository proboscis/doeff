"""S1 を当てる: defmcp-tool の handler の本体の文に guard を巻き、handler に `_install-guard-globals` を呼ぶ(M3 の本体の組み立てだけ)。
盲検 B の apply_s1.py と同じ変更。直した版は McpToolDef の綴りが `(~tool-def` なので、両方の綴りを受ける。
使い方: uv run --no-project python -I apply_s1_fix.py <macros.hy>"""
import sys
from pathlib import Path

HANDLER = """     `(defn [_doeff_do] ~handler-name ~param-syms
       ~@expanded-forms)
     `(setv ~name
"""
GUARDED = """     `(defn [_doeff_do] ~handler-name ~param-syms
       ~@(lfor f (cut expanded-forms 0 -1) (_wrap-statement-guard f (str name)))
       ~@(cut expanded-forms -1 None))
     `(_install-guard-globals ~handler-name)
     `(setv ~name
"""
path = Path(sys.argv[1])
text = path.read_text(encoding="utf-8")
assert text.count(HANDLER) == 1, "defmcp-tool の handler の組み立てが 1 か所でない"
path.write_text(text.replace(HANDLER, GUARDED), encoding="utf-8")
