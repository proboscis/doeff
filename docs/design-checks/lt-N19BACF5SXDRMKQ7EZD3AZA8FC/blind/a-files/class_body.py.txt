"""補い: tool を class の本体に 65 個書いた検体で、module の直下だけを畳む直し(patch_fold.py の (b))が効くかを見る。

使い方: uv run python -I class_body.py <置き場の dir> [<N>(既定 65)]
"""
import ast
import json
import shutil
import subprocess
import sys
from pathlib import Path

import doeff_hy.static_check as static_check

base = Path(sys.argv[1])
count = int(sys.argv[2]) if len(sys.argv) > 2 else 65
lines = ["(require doeff-hy.macros [defmcp-tool <-])", "(import doeff [Pure])", "", "(defclass Box []"]
for index in range(count):
    lines += [
        f"  (defmcp-tool tool-{index}",
        f'    "tool {index}"',
        '    [{:name "text" :type "string" :description "text"}]',
        "    (<- r (Pure text))",
        "    r)",
    ]
lines[-1] += ")"
lines += ["", f"(defn #^ int last-tool-name [] (. Box tool-{count - 1} name))", ""]
src_root = base / f"class{count}"
src_root.mkdir(parents=True, exist_ok=True)
source = src_root / "tools.hy"
source.write_text("\n".join(lines), encoding="utf-8")

original = static_check._synthesized_import


def with_mcp(statement: ast.stmt) -> bool:
    match statement:
        case ast.ImportFrom(module="doeff.mcp", level=0):
            return True
        case _:
            return original(statement)


static_check._synthesized_import = with_mcp
try:
    projection = static_check.project(src_root, [src_root], source)
finally:
    static_check._synthesized_import = original
assert isinstance(projection, static_check.Projection), projection
out = base / f"class{count}-patched"
out.mkdir(parents=True, exist_ok=True)
(out / "tools.py").write_text(projection.text, encoding="utf-8")
(out / "pyrightconfig.json").write_text('{"typeCheckingMode": "basic"}\n', encoding="utf-8")
print("class の本体の 'from doeff.mcp import' =", projection.text.count("from doeff.mcp import"), "文(畳む直しの後)")
run = subprocess.run(
    [shutil.which("pyright"), "--outputjson", "--pythonpath", sys.executable, "tools.py"],
    cwd=out, capture_output=True, text=True, check=False,
)
found = [
    (d["range"]["start"]["line"] + 1, d.get("rule"), d["message"].splitlines()[0])
    for d in json.loads(run.stdout)["generalDiagnostics"]
    if d["severity"] == "error"
]
print("pyright の赤 =", found)
