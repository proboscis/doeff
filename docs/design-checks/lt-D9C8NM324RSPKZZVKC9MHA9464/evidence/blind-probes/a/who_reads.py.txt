"""env_places._read の閉包の歩みで、環境変数の読みの印を立てた所(歩みの中の関数と、印を立てた呼び手)を出す。"""
import json, sys
import hy  # noqa
import doeff_hy  # noqa
from doeff_hy_bytecode_guard import loader_hooks, macro_use, records

stack = []
hits = []
original_function = macro_use._Walker.function
original_bound = macro_use._Walker._bound


def function(self, fn):
    stack.append(f"fn {getattr(fn, '__module__', '')}.{getattr(fn, '__qualname__', '')}")
    try:
        return original_function(self, fn)
    finally:
        stack.pop()


def bound(self, b):
    stack.append(f"bound {b.module.__name__}.{b.name}")
    try:
        return original_bound(self, b)
    finally:
        stack.pop()


def note(self, reads):
    if reads:
        hits.append({"caller": sys._getframe(1).f_code.co_name, "at": stack[-1] if stack else "", "path": stack[-4:]})
    if reads:
        self._mut_environment = True


macro_use._Walker.function = function
macro_use._Walker._bound = bound
macro_use._Walker._note_environment = note
name = sys.argv[1] if len(sys.argv) > 1 else "_read"
macro_use._REMEMBERED.clear()
closure = macro_use.value_digest(records.ValueReference("doeff_hy.env_places", name), loader_hooks.file_sha256)
print(json.dumps({"digest": closure.digest, "hits": hits}, ensure_ascii=False, indent=1))
