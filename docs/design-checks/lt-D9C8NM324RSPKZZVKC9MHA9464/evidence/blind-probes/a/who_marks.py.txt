"""表の形ごとに、環境変数の読みの印を立てた歩みの所(呼び手の method と、その時に歩いていた物)を出す。"""
import importlib, json, sys
sys.path.insert(0, sys.argv[1])
import hy
import doeff_hy  # noqa
from doeff_hy_bytecode_guard import loader_hooks, macro_use, records

stack, hits = [], []
of, ob = macro_use._Walker.function, macro_use._Walker._bound


def function(self, fn):
    stack.append(f"fn {fn.__qualname__}")
    try:
        return of(self, fn)
    finally:
        stack.pop()


def bound(self, b):
    stack.append(f"bound {b.name}")
    try:
        return ob(self, b)
    finally:
        stack.pop()


def note(self, reads):
    if reads:
        hits.append(f"{sys._getframe(1).f_code.co_name} @ {' > '.join(stack)}")
        self._mut_environment = True


macro_use._Walker.function, macro_use._Walker._bound, macro_use._Walker._note_environment = function, bound, note
module = importlib.import_module("probe_s2.macros")
table = vars(module)[records.MACRO_TABLE]
out = {}
for name in ("table", "aliased", "got", "built", "helper", "both"):
    hits.clear()
    macro_use._REMEMBERED.clear()
    key = macro_use.UsedMacroKey(records.MACRO_TABLE, "probe_s2.macros", name)
    closure = macro_use.macro_digest(key, dict.__getitem__(table, name), loader_hooks.file_sha256)
    out[name] = {"digest": closure.digest, "marked_at": list(hits)}
print(json.dumps(out, ensure_ascii=False, indent=1))
