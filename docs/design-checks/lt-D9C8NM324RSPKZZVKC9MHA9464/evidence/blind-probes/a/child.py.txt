"""子の process: 根を sys.path に置き、(必要なら S1 を W の定数だけで模して)名指した module を順に import し、
compile した Hy の file と module の value を JSON で出す。"""
import importlib, importlib.machinery, json, os, sys
from pathlib import Path

root, package, emulate, *steps = sys.argv[1], sys.argv[2], sys.argv[3], *sys.argv[4:]
sys.path.insert(0, root)
import hy

compiled = []
previous = importlib.machinery.SourceFileLoader.source_to_code


def counting(self, data, path, *args, **kwargs):
    code = previous(self, data, path, *args, **kwargs)
    if isinstance(path, str) and path.endswith(".hy"):
        compiled.append(Path(path).name)
    return code


importlib.machinery.SourceFileLoader.source_to_code = counting
import doeff_hy  # noqa: F401
from doeff_hy_bytecode_guard import macro_use

if emulate == "s1":
    # S1 を主張どおり「W の検出の関数と定数だけ」で入れた形を模す: os.getcwd を照らせない読みの名に足す。
    macro_use._ENVIRONMENT_NAMES = macro_use._ENVIRONMENT_NAMES | {"getcwd"}

values = {}
for step in steps:
    module = importlib.import_module(f"{package}.{step}")
    values[step] = getattr(module, "value", None)
print(json.dumps({"cwd": os.getcwd(), "compiled": compiled, "values": values}))
