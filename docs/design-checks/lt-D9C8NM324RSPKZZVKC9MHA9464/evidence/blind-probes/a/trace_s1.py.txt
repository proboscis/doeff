"""歩みの中の file 単位の覆い(_fold)を、呼び手の関数と束縛の読みの結果つきで記録する。
引数: 根・mode(probe = 検の macro・real = 実物の doeff_hy の macro)。"""
import importlib, json, sys
from collections import Counter

root, mode = sys.argv[1], sys.argv[2]
sys.path.insert(0, root)
import hy  # noqa: F401
import doeff_hy  # noqa: F401
from doeff_hy_bytecode_guard import bound_source, loader_hooks, macro_use, records

events = []
fold = macro_use._Walker._fold


def traced_fold(self, module):
    caller = sys._getframe(1).f_code.co_name
    events.append((caller, None if module is None else module.__name__))
    return fold(self, module)


macro_use._Walker._fold = traced_fold
bindings = []
binding_of = bound_source.binding_of


def traced_binding_of(path, name):
    found = binding_of(path, name)
    bindings.append((path.rsplit("/", 2)[-2] + "/" + path.rsplit("/", 1)[-1], name,
                     None if found is None else {"unresolved": found.unresolved, "bound": found.bound,
                                                 "imports": len(found.imports)}))
    return found


bound_source.binding_of = traced_binding_of
macro_use.bound_source.binding_of = traced_binding_of


def digest_of(module_name, macro_name):
    module = importlib.import_module(module_name)
    table = vars(module)[records.MACRO_TABLE]
    key = macro_use.UsedMacroKey(records.MACRO_TABLE, module_name, hy.mangle(macro_name))
    macro_use._REMEMBERED.clear()
    return macro_use.macro_digest(key, dict.__getitem__(table, hy.mangle(macro_name)), loader_hooks.file_sha256)


if mode == "probe":
    targets = [("probe_s1.m_name", "cwd-name"), ("probe_s1.m_module", "cwd-mod"), ("probe_s1.m_bridge", "cwd-bridge")]
    macro_use._ENVIRONMENT_NAMES = macro_use._ENVIRONMENT_NAMES | {"getcwd"}
else:
    targets = [("doeff_hy.macros", "defk"), ("doeff_hy.macros", "<-"), ("doeff_hy.macros", "val"),
               ("doeff_hy.record", "defrecord"), ("doeff_hy.handle", "defhandler"), ("doeff_hy.macros", "deff")]
out = {}
for module_name, macro_name in targets:
    events.clear(); bindings.clear()
    closure = digest_of(module_name, macro_name)
    out[f"{module_name}:{macro_name}"] = {
        "digest": closure.digest if closure.digest in (None, records.ENVIRONMENT_READ) else "sha256",
        "folds": Counter(f"{c}->{m}" for c, m in events).most_common(),
        "unresolved_bindings": sorted({f"{p}:{n}" for p, n, b in bindings if b is not None and b["unresolved"]}),
    }
print(json.dumps(out, ensure_ascii=False, indent=1))
