"""env_places(DOEFF004 に従い環境変数を ReadEnvironment の効果で読む module)の関数を、macro の閉包の歩みが
環境変数の読みと数えるかを確かめる(source は読まず、歩みの結果と、歩みが辿った module だけを出す)。"""
import importlib, json, sys
import hy  # noqa
import doeff_hy  # noqa
from doeff_hy_bytecode_guard import loader_hooks, macro_use, records

places = importlib.import_module("doeff_hy.env_places")
out = {"EXPANDERS_ENV": places.EXPANDERS_ENV}
walked = []
original = macro_use._Walker.function


def logging_function(self, function):
    walked.append(f"{getattr(function, '__module__', '')}.{getattr(function, '__qualname__', '')}")
    return original(self, function)


macro_use._Walker.function = logging_function
for name in ("_read", "expanders_setting", "check_cache_setting", "cache_home"):
    walked.clear()
    macro_use._REMEMBERED.clear()
    closure = macro_use.value_digest(records.ValueReference("doeff_hy.env_places", name), loader_hooks.file_sha256)
    out[name] = {
        "digest": closure.digest if closure.digest in (None, records.ENVIRONMENT_READ) else "sha256",
        "modules": list(closure.modules),
        "walked_functions": sorted(set(walked))[:40],
    }
print(json.dumps(out, ensure_ascii=False, indent=1))
