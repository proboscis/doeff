"""S3(storage): 前の版(基準版 4ca9fcd50)の照らしの行の判じ `_row_check` が、次の版の記録の印 "environment-read" をどう判じるかを、前の版の
関数の本文そのもの(git show で取り出す)を走らせて見る。引数: doeff の木・前の版の sha。前の版の RowCheck と ClosureDigest も同じ file から
取り出す(同じ版の定義で走らせる)。"""
import ast
import enum
import subprocess
import sys
from dataclasses import dataclass

tree, revision = sys.argv[1], sys.argv[2]
source = subprocess.run(
    ["git", "-C", tree, "show", f"{revision}:packages/doeff-hy/src/doeff_hy_bytecode_guard/macro_use.py"],
    capture_output=True, text=True, check=True,
).stdout
module = ast.parse(source)
wanted = {"RowCheck", "ClosureDigest", "_row_check"}
picked = [node for node in module.body if getattr(node, "name", None) in wanted]
assert {node.name for node in picked} == wanted, [node.name for node in picked]
namespace: dict[str, object] = {"enum": enum, "dataclass": dataclass}
exec(compile(ast.Module(body=picked, type_ignores=[]), f"{revision}:macro_use.py", "exec"), namespace)
row_check, closure = namespace["_row_check"], namespace["ClosureDigest"]
mark = "environment-read"
for label, recorded, now in (
    ("記録が印・今の歩みが完了(16 進)", mark, "0" * 64),
    ("記録が印・今の歩みが import の途中(None)", mark, None),
    ("記録が 16 進・今の歩みが完了で同じ 16 進", "0" * 64, "0" * 64),
):
    print(f"{label}: {row_check(recorded, closure(now, ()))}")
