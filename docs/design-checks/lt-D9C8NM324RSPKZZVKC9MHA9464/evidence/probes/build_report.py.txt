def sha(path: Path) -> str:
    """file の中身の sha256(16 進)。"""
    return hashlib.sha256(path.read_bytes()).hexdigest()

def record(check: Check) -> str:
    """確かめ 1 つの記録の本文(参照する file と sha256・期待・観測)。"""
    rows = "\n".join(f"| {name} | {sha(root / name)} |" for name in check.files)
    return (f"# 確かめ {check.id}\n\n- シナリオ: {', '.join(check.scenarios)}\n- 種類: {check.control}\n- 期待: {check.expectation}\n"
            f"- 観測: {check.observation}\n\n| 証拠の file(記録の dir からの path) | sha256 |\n| --- | --- |\n{rows}\n\n"
            "各 file の版と命令は design.md の「証拠の索引」。\n")

for check in CHECKS:

    path.write_text(record(check), encoding="utf-8")
    refs[check.id] = {"path": f"evidence/checks/{check.id}.md", "sha256": sha(path)}
