"""直した版で、本物の macro の閉包の歩みが「照らせない」印(records.ENVIRONMENT_READ)になる数を数える。引数: module の名の一覧の file・
始め・終わり(一覧の切り出し)。PYTHONPYCACHEPREFIX を空の dir に向け、DOEFF_HY_CODE_STORE=off で走らせる(全部の Hy の module を
compile し直し、記録を作る歩みを通す)。答え = 歩みの数・印になった歩みの在処・読み込めなかった module の数。"""
import importlib
import json
import sys
from collections import Counter

import doeff_hy  # noqa: F401 — macro の module を先に読む(子の検と同じ順)
import hy  # noqa: F401
from doeff_hy_bytecode_guard import macro_use, records

walks = Counter()
marked = Counter()
original = macro_use._remembered


def remembered(key, subject, walk, sha256_of):
    """歩み 1 つの結果を数える(印になった歩みは在処を残す)。"""
    closure = original(key, subject, walk, sha256_of)
    walks["all"] += 1
    if closure.digest == records.ENVIRONMENT_READ:
        marked[str(key)] += 1
    return closure


macro_use._remembered = remembered
names = [line.strip() for line in open(sys.argv[1]) if line.strip()][int(sys.argv[2]) : int(sys.argv[3])]
failed = Counter()
for name in names:
    try:
        importlib.import_module(name)
    except BaseException as error:  # noqa: BLE001 — 読めない module は数えるだけ
        failed[type(error).__name__] += 1
print(json.dumps({"modules": len(names), "failed": dict(failed), "walks": walks["all"], "marked": dict(marked)}, ensure_ascii=False))
