"""直し(5c16f9fc3)の 2 か所を外す変異と、盲検 B の差分(読み込みの口の変更)を 1 つずつ当て、環境変数の検が狙った形で赤になるかを見る。
引数: 当てる作業木・出力の dir・盲検 B の差分の file。変異ごとに対象の file を書き換え(B は patch で当て)、
tests/test_macro_use_record.py -k environment を走らせ、落ちた検の名と終了の code を file に残し、git checkout で対象の file だけを戻す。"""
import subprocess
import sys
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class Mutation:
    """変異 1 つ — 名・対象の file(木からの path)・外す所の文と替える文(空なら patch を当てる)・赤になるはずの検。"""

    name: str
    target: str
    old: str
    new: str
    expected: tuple[str, ...]


GUARD = "packages/doeff-hy/src/doeff_hy_bytecode_guard"
MUTATIONS = (
    Mutation(
        "blind-b-loader-hooks-diff",
        f"{GUARD}/loader_hooks.py",
        "",
        "",
        (
            "test_an_environment_reading_user_is_compiled_again_after_its_macro_changed_in_another_tree",
            "test_the_environment_mark_is_read_only_by_the_walk_and_the_check",
        ),
    ),
    Mutation(
        "hy-import-left-unresolved",
        f"{GUARD}/bound_source.py",
        "unresolved=any(clauses is None for clauses in imported),",
        "unresolved=bool(imported),",
        ("[hy-import]",),
    ),
    Mutation(
        "opaque-fold-without-mark",
        f"{GUARD}/macro_use.py",
        "self._note_environment(names is not None and _source_reads_environment(names))",
        "self._note_environment(False)",
        ("[default]", "[cell]"),
    ),
)


def run(tree: Path, out: Path, diff: Path, mutation: Mutation) -> str:
    """変異 1 つを当てて検を走らせ、結果の 1 行を返す(出力の全文は out の下の file)。"""
    target = tree / mutation.target
    if mutation.old:
        text = target.read_text(encoding="utf-8")
        assert text.count(mutation.old) == 1, f"{mutation.name}: 外す所が 1 つでない"
        target.write_text(text.replace(mutation.old, mutation.new), encoding="utf-8")
    else:
        subprocess.run(["patch", str(target), str(diff)], check=True, capture_output=True)
    try:
        done = subprocess.run(
            ["uv", "run", "--frozen", "pytest", "-q", "-p", "no:cacheprovider", "-k", "environment",
             "tests/test_macro_use_record.py"],
            cwd=tree / "packages/doeff-hy", capture_output=True, text=True, timeout=500, check=False,
        )
        applied = subprocess.run(["git", "-C", str(tree), "diff", "--", mutation.target], capture_output=True, text=True, check=False).stdout
    finally:
        subprocess.run(["git", "-C", str(tree), "checkout", "--", mutation.target], check=True)
    failed = tuple(line.split("::", 1)[1] for line in done.stdout.splitlines() if line.startswith("FAILED "))
    hit = all(any(want in name for name in failed) for want in mutation.expected)
    summary = next((line for line in reversed(done.stdout.splitlines()) if " in " in line and ("passed" in line or "failed" in line)), "")
    head = subprocess.run(["git", "-C", str(tree), "rev-parse", "HEAD"], capture_output=True, text=True, check=False).stdout.strip()
    (out / f"mut-{mutation.name}.txt").write_text(
        f"変異: {mutation.name}\n対象の版: {head}\n狙った検: {mutation.expected}\n終了の code: {done.returncode}\n要約: {summary}\n"
        f"落ちた検: {failed}\n---- 当てた差分\n{applied}\n---- pytest の出力\n{done.stdout}\n{done.stderr[-3000:]}",
        encoding="utf-8",
    )
    return f"{mutation.name}: code={done.returncode} {summary} 狙いが赤={'はい' if hit else 'いいえ'} 落ちた={len(failed)} {failed}"


def main() -> None:
    """変異を順に当てて結果を出す。"""
    tree, out, diff = Path(sys.argv[1]), Path(sys.argv[2]), Path(sys.argv[3])
    out.mkdir(parents=True, exist_ok=True)
    for line in map(lambda mutation: run(tree, out, diff, mutation), MUTATIONS):
        print(line, flush=True)


main()
