"""検の package を置き、環境変数か cwd だけを替えた 2 つの process で使い手を読む。"""
import json, os, shutil, subprocess, sys
from pathlib import Path

base = Path("/work/agent-workspaces/a-P3XK61KTD6P4A1FJZ3Y6RKN9QY/dc/s1probe")
PKG = "probe_s1"
FILES = {
    "__init__.py": "",
    # 表を組む Python の module(環境変数と cwd から)。
    "tables.py": 'import os\nROWS = {"a": os.environ.get("PROBE_MODE", "plain")}\nROOTS = {"root": os.getcwd()}\n',
    # Python の module が名で import して持ち直す(Python の import は B が解く)。
    "bridge.py": f"from {PKG}.tables import ROWS, ROOTS\n",
    # macro の module 3 つ: module の属性で読む・Hy の import で名を束縛して読む・Python の橋を通して読む。
    "m_module.hy": f'(import {PKG}.tables :as tables)\n(defmacro env-mod [] (get tables.ROWS "a"))\n(defmacro cwd-mod [] (get tables.ROOTS "root"))\n',
    "m_name.hy": f'(import {PKG}.tables [ROWS ROOTS])\n(defmacro env-name [] (get ROWS "a"))\n(defmacro cwd-name [] (get ROOTS "root"))\n',
    "m_bridge.hy": f'(import {PKG}.bridge :as bridge)\n(defmacro env-bridge [] (get bridge.ROWS "a"))\n(defmacro cwd-bridge [] (get bridge.ROOTS "root"))\n',
}
USERS = {
    "u_env_mod": ("m_module", "env-mod"),
    "u_env_name": ("m_name", "env-name"),
    "u_env_bridge": ("m_bridge", "env-bridge"),
    "u_cwd_mod": ("m_module", "cwd-mod"),
    "u_cwd_name": ("m_name", "cwd-name"),
    "u_cwd_bridge": ("m_bridge", "cwd-bridge"),
}
for user, (mod, mac) in USERS.items():
    FILES[f"{user}.hy"] = f"(require {PKG}.{mod} [{mac}])\n(setv value ({mac}))\n"


def setup(tag):
    work = base / tag
    if work.exists():
        shutil.rmtree(work)
    (work / "tree" / PKG).mkdir(parents=True)
    for name, text in FILES.items():
        (work / "tree" / PKG / name).write_text(text, encoding="utf-8")
    (work / "store").mkdir()
    (work / "cwd1").mkdir()
    (work / "cwd2").mkdir()
    return work


def run(work, emulate, cwd, mode, users):
    env = dict(os.environ)
    env["DOEFF_HY_CODE_STORE"] = str(work / "store")
    env["PYTHONPYCACHEPREFIX"] = str(work / "pycache")
    env.pop("PYTHONDONTWRITEBYTECODE", None)
    env.pop("PROBE_MODE", None)
    if mode is not None:
        env["PROBE_MODE"] = mode
    done = subprocess.run(
        [sys.executable, str(base / "child.py"), str(work / "tree"), PKG, emulate, *users],
        capture_output=True, text=True, timeout=120, cwd=str(cwd), env=env,
    )
    if done.returncode != 0:
        print(done.stderr)
        raise SystemExit(1)
    return json.loads(done.stdout.strip().splitlines()[-1])


for emulate in ("base", "s1"):
    work = setup(emulate)
    users = list(USERS)
    first = run(work, emulate, work / "cwd1", None, users)
    same = run(work, emulate, work / "cwd1", None, users)
    changed = run(work, emulate, work / "cwd2", "fancy", users)
    print(f"== {emulate}(1 つ目: cwd1・PROBE_MODE 無し → 同じ環境の 2 つ目 → cwd2・PROBE_MODE=fancy の 3 つ目)")
    print("1st compiled:", sorted(first["compiled"]))
    print("2nd(同じ環境)compiled:", sorted(same["compiled"]))
    print("3rd(環境を替えた)compiled:", sorted(changed["compiled"]))
    for user in users:
        print(f"  {user:14s} 1st={first['values'][user]!r:40s} 3rd={changed['values'][user]!r}")
