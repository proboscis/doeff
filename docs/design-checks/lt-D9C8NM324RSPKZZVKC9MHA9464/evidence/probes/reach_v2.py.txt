"""直した版(5c16f9fc3)で、本物の macro の閉包の歩みが印(records.ENVIRONMENT_READ)になる数と、辿れない値の逃げ道(_Walker._fold_opaque)と
file で覆う所(_Walker._fold)に落ちる module・呼び手を数える。引数: module の名の一覧の file・始め・終わり。PYTHONPYCACHEPREFIX を空の dir
に向け、DOEFF_HY_CODE_STORE=off で走らせる。"""
import importlib
import json
import sys
from collections import Counter

import doeff_hy  # noqa: F401
import hy  # noqa: F401
from doeff_hy_bytecode_guard import macro_use, records

walks = Counter()
marked = Counter()
folds = Counter()
remembered_original = macro_use._remembered
fold_original = macro_use._Walker._fold
opaque_original = macro_use._Walker._fold_opaque


def remembered(key, subject, walk, sha256_of):
    """歩み 1 つの結果を数える(印になった歩みは在処を残す)。"""
    closure = remembered_original(key, subject, walk, sha256_of)
    walks["all"] += 1
    if closure.digest == records.ENVIRONMENT_READ:
        marked[str(key)] += 1
    return closure


def fold(self, module):
    """file で覆う 1 回を、覆う module と呼び手の関数の名で数える。"""
    folds[("_fold", getattr(module, "__name__", "none"), sys._getframe(1).f_code.co_name)] += 1
    return fold_original(self, module)


def opaque(self, module):
    """辿れない値の逃げ道 1 回を、覆う module と呼び手の関数の名で数える。"""
    folds[("_fold_opaque", getattr(module, "__name__", "none"), sys._getframe(1).f_code.co_name)] += 1
    return opaque_original(self, module)


macro_use._remembered = remembered
macro_use._Walker._fold = fold
macro_use._Walker._fold_opaque = opaque
names = [line.strip() for line in open(sys.argv[1]) if line.strip()][int(sys.argv[2]) : int(sys.argv[3])]
failed = Counter()
for name in names:
    try:
        importlib.import_module(name)
    except BaseException as error:  # noqa: BLE001 — 読めない module は数えるだけ
        failed[type(error).__name__] += 1
print(json.dumps({"modules": len(names), "failed": dict(failed), "walks": walks["all"], "marked": dict(marked),
                  "folds": sorted(([kind, module, caller, count] for (kind, module, caller), count in folds.items()), key=lambda row: -row[3])},
                 ensure_ascii=False))
