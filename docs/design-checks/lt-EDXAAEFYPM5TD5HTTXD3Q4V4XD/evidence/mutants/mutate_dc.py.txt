"""card ki-f8a58fb49dd9 の変異の確かめ: 変異ごとに新しい検と旧い検(分かれ目の版)を走らせ、赤になった検を書く。

使い方: uv run python -I /tmp/ki-f8a5-mut/mutate.py <作業木> <変異の名 …>
"""

import re
import subprocess
import sys
from pathlib import Path

WT = Path(sys.argv[1])
PKG = WT / "packages/doeff-traverse"
EX = PKG / "examples"
EXP = EX / "expected"
NEW = PKG / "tests/test_examples_run.py"
OLD = PKG / "tests/test_examples_run_before.py"
RUNNER = PKG / "tests/run_example_failing_a_case.py"


def hy_output(source: str) -> str:
    probe = EX / "_probe_ki_f8a5.hy"
    probe.write_text(source, encoding="utf-8")
    try:
        return subprocess.run(
            [sys.executable, "-m", "hy", str(probe)], capture_output=True, text=True, check=True
        ).stdout
    finally:
        probe.unlink()


def replaced(path: Path, old: str, new: str) -> str:
    text = path.read_text(encoding="utf-8")
    assert text.count(old) >= 1, (path, old)
    return text.replace(old, new, 1)


SIMPLE = (EX / "ex04_simple.hy").read_text(encoding="utf-8").split('(print "Result:"')[0]


def mutations(name: str) -> dict[Path, str]:
    if name == "M1-frame-mark-swallows-case-lines":
        swallowed = (
            "      SymbolNotFound: symbol not found: BADCO\n"
            "  [2] OK: ({'symbol': 'GOOGL', 'time': '2024-01-01'}, 2800.0)\n"
            "      index: ok\n      signal: ok\n      price: ok\n"
        )
        return {EXP / "ex07_proboscis_ema_rewrite.txt": replaced(EXP / "ex07_proboscis_ema_rewrite.txt", swallowed, "")}
    if name == "M2-inline-mark-hides-a-sum":
        return {EXP / "ex05_parallel_timing.txt": replaced(EXP / "ex05_parallel_timing.txt", "async + sequential:   550 in", "async + sequential:   … in")}
    if name == "M3-inline-mark-hides-a-count":
        return {EXP / "ex07_proboscis_ema_rewrite.txt": replaced(EXP / "ex07_proboscis_ema_rewrite.txt", "Success: 3\n", "Success: …\n")}
    if name == "M4-example-without-cases":
        source = ";;; debug: traverse の macro の形を印字する(件を走らせない)\n(print (hy.repr '(traverse (<- x (Iterate items)) x)))\n"
        return {EX / "ex10_debug.hy": source, EXP / "ex10_debug.txt": hy_output(source)}
    if name == "M5-example-prints-no-case-result":
        source = SIMPLE + '(print "done:" (len (. result all_items)))\n'
        return {EX / "ex10_silent.hy": source, EXP / "ex10_silent.txt": hy_output(source)}
    if name == "M6-no-cases-row-removed":
        text = NEW.read_text(encoding="utf-8")
        return {NEW: re.sub(r'\n    "ex01_basic_fail\.hy": [^\n]*\n', "\n", text, count=1)}
    if name == "M7-machine-path-and-frame-line-pinned":
        source = (
            SIMPLE
            + '(print "Valid:" (. result valid_values))\n(import sys)\n'
            + '(print (+ sys.base_prefix "/lib/python3.14/runpy.py"))\n'
            + '(print "  File \\"<frozen runpy>\\", line 88, in _run_code")\n'
        )
        return {EX / "ex10_machine.hy": source, EXP / "ex10_machine.txt": hy_output(source)}
    if name == "M8-example-rewritten-without-output-change":
        return {EX / "ex05_parallel_timing.hy": replaced(EX / "ex05_parallel_timing.hy", "(resume (* x 10))", "(resume (* 10 x))")}
    if name == "M9-runner-patch-dropped":
        return {RUNNER: replaced(RUNNER, "    Traverse.__init__ = _init\n", "    pass\n")}
    if name == "M10-run-only-in-a-never-called-function":
        source = (
            ";;; debug: run は呼ばない関数の中にだけ在る\n(import doeff [run])\n"
            "(defn never-called [] (run None))\n(print \"macro の展開の形だけを印字する\")\n"
        )
        return {EX / "ex10_unrun.hy": source, EXP / "ex10_unrun.txt": hy_output(source)}
    if name == "M11-runner-stops-counting-runs":
        return {RUNNER: replaced(RUNNER, "    _count_program_runs()\n", "    pass\n")}
    if name == "M12-child-time-zone-pin-dropped":
        return {NEW: replaced(NEW, 'return {**os.environ, "TZ": CHILD_TIME_ZONE}', "return dict(os.environ)")}
    raise SystemExit(f"unknown mutation {name}")


def run_pytest() -> tuple[str, list[str]]:
    done = subprocess.run(
        [sys.executable, "-m", "pytest", str(NEW), *([str(OLD)] if OLD.exists() else []), "-q", "-rf"],
        capture_output=True, text=True, cwd=WT,
    )
    failed = [line.split(" - ")[0].removeprefix("FAILED ") for line in done.stdout.splitlines() if line.startswith("FAILED ")]
    summary = [line for line in done.stdout.splitlines() if re.search(r"\d+ (passed|failed)", line)]
    return (summary[-1] if summary else done.stdout[-2000:]), failed, done.stdout


for name in sys.argv[2:]:
    changes = mutations(name)
    originals = {path: path.read_text(encoding="utf-8") if path.exists() else None for path in changes}
    try:
        for path, text in changes.items():
            path.write_text(text, encoding="utf-8")
        summary, failed, out = run_pytest()
    finally:
        for path, text in originals.items():
            if text is None:
                path.unlink()
            else:
                path.write_text(text, encoding="utf-8")
    head = subprocess.run(["git", "-C", str(WT), "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()
    record = (f"# 変異: {name}(/tmp/ki-f8a5-mut/mutate_dc.py の同じ名の変異を当て、検を走らせ、元へ戻した)\n"
              f"# 版: {head} + 作業木の直し(test_examples_run.py・run_example_failing_a_case.py)\n"
              f"# 当てた file: {', '.join(str(p.relative_to(WT)) for p in changes)}\n"
              f"# 命令: uv run python -m pytest packages/doeff-traverse/tests/test_examples_run.py -q -rf\n"
              f"## 要約: {summary}\n## 赤: {failed}\n## 出力\n{out}")
    (WT / "docs/design-checks/lt-EDXAAEFYPM5TD5HTTXD3Q4V4XD/evidence/mutants" / f"{name}.txt").write_text(record, encoding="utf-8")
    print(f"== {name}: {summary}")
    for test in failed:
        print(f"   red: {test.replace(str(PKG) + '/', '')}")
