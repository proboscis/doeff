"""実験: traverse-lib の件の失敗の方針を「Fail を通った失敗だけを件の失敗にし、それ以外の例外は上げる」に変える。
引数: doeff-traverse の package の dir。"""
import sys
from pathlib import Path

root = Path(sys.argv[1])
h = root / "doeff_traverse" / "handlers.py"
src = h.read_text()

# 印: fail_handler を通った例外(Fail の cause)にだけ付ける
src = src.replace(
    '''# Sentinel for Skip — traverse handler checks identity
_SKIPPED = object()
''',
    '''# Sentinel for Skip — traverse handler checks identity
_SKIPPED = object()

# 件の失敗の方針(実験): Fail を通って上がった例外だけが件の失敗。f の bug・展開の不具合などの
# ほかの例外は件の失敗として隠さず、Traverse を越えて上げる。
_ITEM_FAILURE_ATTR = "__doeff_traverse_item_failure__"


def _is_item_failure(error: BaseException) -> bool:
    return bool(getattr(error, _ITEM_FAILURE_ATTR, False))
''', 1)

# sequential
old_seq = '''                    except ItemOutcomeNotPassedError:
                        raise
                    except Exception as error:
                        result, outcome = ItemResult('''
new_seq = '''                    except ItemOutcomeNotPassedError:
                        raise
                    except Exception as error:
                        if not _is_item_failure(error):
                            raise
                        result, outcome = ItemResult('''
assert src.count(old_seq) == 1
src = src.replace(old_seq, new_seq)

# parallel
old_par = '''                        raise result.error
                    if not isinstance(result, Ok):
                        done, outcome = ItemResult('''
new_par = '''                        raise result.error
                    if not isinstance(result, Ok) and not _is_item_failure(result.error):
                        raise result.error
                    if not isinstance(result, Ok):
                        done, outcome = ItemResult('''
assert src.count(old_par) == 1
src = src.replace(old_par, new_par)

# fail_handler が印を付ける
old_fail = '''        exc = effect.cause if isinstance(effect.cause, BaseException) else RuntimeError(str(effect.cause))
        return (yield ResumeThrow(k, exc))'''
new_fail = '''        exc = effect.cause if isinstance(effect.cause, BaseException) else RuntimeError(str(effect.cause))
        setattr(exc, _ITEM_FAILURE_ATTR, True)
        return (yield ResumeThrow(k, exc))'''
assert src.count(old_fail) == 1
src = src.replace(old_fail, new_fail)
h.write_text(src)

# 利用者(例)の追従: ex08 の backend は件の失敗を raise ではなく Fail で知らせる(出力は変えない)
e = root / "examples" / "ex08_tqdm_progress.hy"
ex = e.read_text()
ex = ex.replace("(import doeff_traverse.effects [Reduce])", "(import doeff_traverse.effects [Reduce Fail])", 1)
old_ex = '(raise (ConnectionError "backend down for 7")))'
assert ex.count(old_ex) == 1
ex = ex.replace(old_ex, '(<- (Fail (ConnectionError "backend down for 7"))))')
e.write_text(ex)
print("applied")
