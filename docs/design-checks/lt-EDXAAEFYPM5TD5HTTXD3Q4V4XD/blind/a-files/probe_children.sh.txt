#!/bin/bash
# 件の中で失敗させる子を各例で走らせ、終了コードと実行の印を並べる(木の path を引数にとる)
TREE="$1"; shift
cd "$TREE" || exit 2
for ex in packages/doeff-traverse/examples/*.hy; do
  name=$(basename "$ex")
  out=$(uv run python packages/doeff-traverse/tests/run_example_failing_a_case.py "$ex" 2>/tmp/ki-f8a5-blind-a/stderr-$name.txt >/tmp/ki-f8a5-blind-a/stdout-$name.txt; echo $?)
  marks=$(grep -E '^case-calls:' /tmp/ki-f8a5-blind-a/stderr-$name.txt)
  last=$(grep -vE '^case-calls:' /tmp/ki-f8a5-blind-a/stderr-$name.txt | grep -E 'Error|Exception' | tail -1)
  echo "$name exit=$out $marks ${last:+| $last}"
done
