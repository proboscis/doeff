#!/bin/bash
# 再現: traverse-lib の件の失敗の方針を変えると、件の結果を印字しなくなった例を検が黙って通す。
# 使い方: repro.sh <86ecd1344 の実験の木>(木は変更の無い状態で渡す。終わると元に戻す)
set -u -o pipefail
TREE="$1"; D=/tmp/ki-f8a5-blind-a
cd "$TREE" || exit 2
t() { uv run pytest packages/doeff-traverse/tests/test_examples_run.py -q -p no:cacheprovider -k "$1" 2>&1 | grep -E "^FAILED|passed|failed" ; }
echo "## 1. 今の方針 + 変異 M(ex04_simple が件の結果を印字しない)"; git apply "$D/mutation-M.diff"; t ex04_simple
echo "## 2. 新しい方針(Fail を通った失敗だけ件の失敗)+ 変異 M"; git apply "$D/policy.diff"; t ex04_simple
echo "## 3. 2 に failing-child の追従 (b')(注入を Fail にする)"; git apply "$D/child-via-fail-b2.diff"; t ex04_simple
git apply -R "$D/child-via-fail-b2.diff" "$D/policy.diff" "$D/mutation-M.diff" 2>/dev/null || { git apply -R "$D/child-via-fail-b2.diff"; git apply -R "$D/policy.diff"; git apply -R "$D/mutation-M.diff"; }
echo "## 後: git status(空なら元どおり)"; git status --short
