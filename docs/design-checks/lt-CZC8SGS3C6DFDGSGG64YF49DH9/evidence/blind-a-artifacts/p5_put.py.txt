"""session var を持つ class の本体の defhandler が、書き手の KV の effect `Put` を受ける(副次の確かめ)。"""
import sys; sys.path.insert(0, "/tmp/blind-a-ki-edab")
from harness import load, go
H = """
(require doeff-hy.handle [defhandler])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Put [EffectBase] #^ str key #^ int value)
"""
src = """
(defclass FakeStore []
  (defhandler store (session var writes 0)
    (Put [key value] (:= writes (+ writes 1)) (resume writes))))
"""
try:
    m = load(src, name="kv_store", header=H)
    print("module の Put:", m.Put.__module__)
    print("結果:", repr(go(m.FakeStore.store(m.Put("a", 1)))))
except Exception as e:
    print("例外:", type(e).__name__, str(e).splitlines()[0][:200])
