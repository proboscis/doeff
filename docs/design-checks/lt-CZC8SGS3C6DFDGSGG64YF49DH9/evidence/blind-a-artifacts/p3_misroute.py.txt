"""外に「本番の振込」の handler、内に class の本体の差し替え(FakeBank)を置く — 内の節が受けるはずの Transfer が外へ抜けるか。"""
import sys; sys.path.insert(0, "/tmp/blind-a-ki-edab")
from harness import load, go
from doeff.do import do
from doeff import Transfer as VMTransfer, Pass
from doeff_vm import WithHandler
from p2_transfer import BANK  # noqa: E402  (p2 の CASES も走るが無害)

def production_bank(effect_type):
    """Python で書いた外の handler(書き手の Transfer を受け、本番の振込が走った印を答える)。"""
    @do
    def h(effect, k):
        if isinstance(effect, effect_type):
            return (yield VMTransfer(k, "本番の振込が走った"))
        return (yield Pass(effect, k))
    return lambda prog: WithHandler(h, prog)

m = load("(defclass FakeBank [] (defhandler bank (Transfer [amount] (resume (+ amount 100)))))",
         name="bank_misroute", header=BANK)
print("期待 105(内の FakeBank が答える)/ 実際:", repr(go(production_bank(m.Transfer)(m.FakeBank.bank(m.Transfer(5))))))
