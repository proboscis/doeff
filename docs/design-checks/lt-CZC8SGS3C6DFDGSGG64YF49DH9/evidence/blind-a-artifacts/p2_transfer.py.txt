"""振込の effect `Transfer` を書き手の module の段で定義し、class の本体の handle・defhandler の節で受ける。"""
import sys; sys.path.insert(0, "/tmp/blind-a-ki-edab")
import traceback
from harness import load, go

BANK = """
(require doeff-hy.handle [handle defhandler])
(import dataclasses [dataclass])
(import doeff [EffectBase])
;; 書き手の業務の effect: 振込(module の段 — S1 の前提「effect の型は module の段に在る」を守る)
(defclass [(dataclass :frozen True)] Transfer [EffectBase] #^ int amount)
(defclass [(dataclass :frozen True)] Bump [EffectBase] #^ int n)
"""

CASES = {
    "class-handle-Transfer": ("""
(defclass FakeBank []
  (setv program (handle (Transfer 5) (Transfer [amount] (resume (+ amount 100))))))
""", lambda m: m.FakeBank.program),
    "class-defhandler-Transfer": ("""
(defclass FakeBank [] (defhandler bank (Transfer [amount] (resume (+ amount 100)))))
""", lambda m: m.FakeBank.bank(m.Transfer(5))),
    "class-defhandler-params-Transfer": ("""
(defclass FakeBank [] (defhandler bank [fee] (Transfer [amount] (resume (+ amount fee)))))
""", lambda m: m.FakeBank.bank(100)(m.Transfer(5))),
    # 対照: 同じ形で effect の名だけ Bump
    "class-defhandler-Bump(control)": ("""
(defclass FakeBank [] (defhandler bank (Bump [n] (resume (+ n 100)))))
""", lambda m: m.FakeBank.bank(m.Bump(5))),
}

for name, (src, prog) in CASES.items():
    try:
        m = load(src, name="bank_" + name.replace("-", "_").replace("(", "").replace(")", ""), header=BANK)
        print(f"{name}: module の Transfer = {m.Transfer!r}")
        print(f"  結果 = {go(prog(m))!r}")
    except Exception as e:
        print(f"  例外 = {type(e).__name__}: {str(e)[:300]}")
