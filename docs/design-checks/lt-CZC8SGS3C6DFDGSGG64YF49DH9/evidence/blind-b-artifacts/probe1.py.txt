import sys, types
import doeff_hy  # noqa
import hy
HEADER = """
(require doeff-hy.handle [handle defhandler])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Bump [EffectBase] #^ int n)
"""
src = """
(defclass C []
  (setv base 100)
  (setv program (handle (Bump base) (Bump [n] (resume (+ n 1)))))
  (defhandler h (Bump [n] (resume (+ n 100))))
  (defhandler hp [b] {:effects [Bump]} (Bump [n] (resume (+ n b))))
  (defhandler hs (session var calls 99) (Bump [n] (:= calls (+ calls 1)) (resume (+ n calls)))))
"""
mod = types.ModuleType("probe1"); sys.modules["probe1"] = mod
for f in hy.read_many(HEADER + src):
    hy.eval(f, mod.__dict__, module=mod)
print(sorted(k for k in vars(mod.C) if not k.startswith("__")))
print(hy.repr(hy.macroexpand(hy.read("(handle (Bump base) (Bump [n] (resume (+ n 1))))"), module=mod))[:300])
