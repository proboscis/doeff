+  "defhandler の節が `<-` で外へ出す effect を、その handler の :effects 宣言(__doeff_effects__)と照らすため — 宣言の外の
+   effect を外へ出したら TypeError。宣言が無い(None)handler は照らさない。宣言の要素が effect を作る関数なら、答えの注釈の
+   型で照らす(effect-refusal と同じ受け方)。照らした effect をそのまま返す。"
+  (import typing)
+  (setv declared (getattr handler "__doeff_effects__" None))
+  (when (is-not declared None)
+    (setv types (tuple (gfor item declared

+      (raise (TypeError (.format "defhandler {}: 節が :effects に宣言していない effect {} を <- で外へ出した(宣言: {})"
+                                 (getattr handler "__doeff_name__" "?") (. (type effect) __name__)
+                                 (.join " " (gfor t types (. t __name__))))))))
+  effect)
