# 設計検証 lt-0ZP7AYKXMDX83R8DKGFBA73DBJ の確かめ c0〜c11 を、直した版を切った作業木で走らせ、出力を <evidence>/checks/ へ書く。
# 使い方: bash measure.sh.txt <作業木> <この記録の evidence の dir> <確かめの名 …>(例: c0 c1 c2)
# 変異は作業木の file を退避から書き戻して戻す(git の checkout・reset は使わない)。終わりに作業木の git status を印字する。
set -u
W=$1
OUT=$2
shift 2
EX=$W/packages/doeff-traverse/examples
TEST=packages/doeff-traverse/tests/test_examples_run.py
MACROS=$W/packages/doeff-hy/src/doeff_hy/macros.hy
STUB=$W/packages/doeff-hy/src/doeff_hy/static_stub.py
SAVE=$(mktemp -d)
cp "$W/$TEST" "$SAVE/test.py"
cp "$MACROS" "$SAVE/macros.hy"
cp "$STUB" "$SAVE/static_stub.py"

restore() {
  cp "$SAVE/test.py" "$W/$TEST"
  cp "$SAVE/macros.hy" "$MACROS"
  cp "$SAVE/static_stub.py" "$STUB"
  rm -f "$EX"/ex04_zz_*.hy "$EX"/expected/ex04_zz_*.txt "$EX/ex10_val_inputs.hy" "$EX/expected/ex10_val_inputs.txt"
}

pytest_run() {
  echo "\$ uv run pytest $TEST -q --tb=line -rf $*"
  (cd "$W" && uv run pytest "$TEST" -q --tb=line -rf "$@" 2>&1 \
    | grep -vE '^check-slot|PytestWarning|warnings.warn|^ +- docs/adr|Add their directories|^deadline scale|^test-admission|^$' \
    | cut -c1-600
   echo "exit=${PIPESTATUS[0]}")
}

head_lines() {
  echo "# 確かめ $1 — $2"
  echo "# 版: $(git -C "$W" rev-parse HEAD)・作業木 ${W}・$(date -u +%FT%TZ)"
}

# ex04_simple.hy の写しに 1 行足した例と、ex04_simple の期待の出力に 1 行足した期待の出力を置く。
copy_example() {
  cp "$EX/ex04_simple.hy" "$EX/$1.hy"
  printf '%s\n' "$2" >> "$EX/$1.hy"
  cp "$EX/expected/ex04_simple.txt" "$EX/expected/$1.txt"
  printf '%s\n' "$3" >> "$EX/expected/$1.txt"
  echo "# 足した例 $1.hy の末尾の行: $2"
}

# module の直下で val を使う例(ex04_simple.hy の入力を (val inputs [1 2 3]) にした写し・出力は ex04_simple と同じ)。
val_example() {
  sed -e 's/(require doeff-hy.macros \[defk <- traverse\])/(require doeff-hy.macros [defk <- traverse val])/' \
      -e 's/^(setv program (pipeline \[1 2 3\]))/(val inputs [1 2 3])\n(setv program (pipeline inputs))/' \
      "$EX/ex04_simple.hy" > "$EX/ex10_val_inputs.hy"
  cp "$EX/expected/ex04_simple.txt" "$EX/expected/ex10_val_inputs.txt"
  diff "$EX/ex04_simple.hy" "$EX/ex10_val_inputs.hy"
}

val_row() {
  sed -i 's/^UNMARKED_EXPANSIONS: dict\[str, str\] = {}$/UNMARKED_EXPANSIONS: dict[str, str] = {"val": "1 段の展開は利用者の書く setv と同じ形"}/' "$W/$TEST"
  grep -n '^UNMARKED_EXPANSIONS' "$W/$TEST"
}

unplaceable_in() {
  (cd "$W" && uv run python -c 'import sys; sys.path.insert(0, "packages/doeff-traverse/tests"); import test_examples_run as t; print(t._unplaceable_in(sys.argv[1]))' "$1" 2>&1 | tail -1)
}

for check in "$@"; do
  case $check in
    c0)
      { head_lines c0 "直した版で例の検が全部緑(R3・S2 の正常例)"; pytest_run; } > "$OUT/checks/c0-all-green.txt" 2>&1 ;;
    c1)
      { head_lines c1 "今の traverse の展開を文字列で写して印字する例を足すと、期待の出力の検が file と「展開の形の印」を名指して赤、例の他の検は緑(R1・S2(b) の失敗例)"
        copy_example ex04_zz_current_copy \
          '(print "traverse の展開: (_doeff_traverse_Traverse (_doeff_do (fn [x] (do (setv y (yield (open_bind (double x)))) y))) items)")' \
          'traverse の展開: (_doeff_traverse_Traverse (_doeff_do (fn [x] (do (setv y (yield (open_bind (double x)))) y))) items)'
        pytest_run -k "zz or unplaceable or printed_expansion"; } > "$OUT/checks/c1-current-copy-named.txt" 2>&1
      restore ;;
    c2)
      { head_lines c2 "古い展開の形(外した ex04_debug2 が「macro が作る形」として写した形)を印字する例も名指す(S2(b) の失敗例)"
        copy_example ex04_zz_old_copy \
          '(print "macro が作る形: (fn [x] (_doeff_do (fn [] (yield (double x)))))")' \
          'macro が作る形: (fn [x] (_doeff_do (fn [] (yield (double x)))))'
        pytest_run -k "zz or unplaceable"; } > "$OUT/checks/c2-old-copy-named.txt" 2>&1
      restore ;;
    c3)
      { head_lines c3 "__import__ を印から外すと、展開に印の無い使い所(fnk)を C2 の検が <行>: <macro> で名指す(S1(c) の失敗例)"
        sed -i 's/^_EXPANSION_IMPORT = "__import__"$/_EXPANSION_IMPORT = "__not_a_mark__"/' "$W/$TEST"
        grep -n '^_EXPANSION_IMPORT' "$W/$TEST"
        pytest_run -k "expansions_show or unmarked"; } > "$OUT/checks/c3-unmarked-expansion-named.txt" 2>&1
      restore ;;
    c4)
      { head_lines c4 "盲検 A の付けたり: doeff-hy が文脈の外の (<- …) を展開の時に断る変更(変異 M-A2)の下でも、例と例の検は全部緑(S1 の正常例・直しの確かめ)"
        (cd "$W" && git apply "$OUT/m-a2-module-level-bind-refused.patch.txt" && git diff --stat)
        pytest_run; } > "$OUT/checks/c4-context-bound-macro-green.txt" 2>&1
      restore ;;
    c5)
      { head_lines c5 "本当に隠れている名(defk の本体の束ねが引く HELPERS-NAME = _doeff_outcomes)を改名しても、例と例の検は変わらず緑(S1(a) の正常例)"
        sed -i 's/^(setv HELPERS-NAME "_doeff_outcomes")$/(setv HELPERS-NAME "_doeff_oc")/' "$MACROS"
        (cd "$W" && git diff --stat && git diff | grep '^[-+](setv HELPERS')
        pytest_run; } > "$OUT/checks/c5-hidden-helper-rename-green.txt" 2>&1
      restore ;;
    c6)
      { head_lines c6 "盲検 A の失敗例: 呼び手が import する別名 _doeff_traverse_Traverse を macros.hy の中だけで改名すると、例 7 本が赤(事前の主張 S1(a) の誤り — traverse・for/do が呼び手に別名の import を求める、前からの漏れ)"
        sed -i 's/_doeff_traverse_Traverse/_doeff_tv_Traverse/g' "$MACROS"
        (cd "$W" && git diff --stat)
        pytest_run --tb=no; } > "$OUT/checks/c6-caller-imported-alias-rename-red.txt" 2>&1
      restore ;;
    c7)
      { head_lines c7 "盲検 B の付けたり: module の直下で val を使う例は、UNMARKED_EXPANSIONS に理由つきの行を置けば緑(S2(a) の正常例・直しの確かめ)"
        val_example
        val_row
        pytest_run -k "ex10_val or unmarked or expansion"; } > "$OUT/checks/c7-val-example-with-row-green.txt" 2>&1
      restore ;;
    c8)
      { head_lines c8 "val を使う例を、UNMARKED_EXPANSIONS の行なしに足すと、C2 の検が <行>: val を名指して赤(S2(a) の失敗例)"
        val_example
        pytest_run -k "ex10_val"; } > "$OUT/checks/c8-val-example-without-row-red.txt" 2>&1
      restore ;;
    c9)
      { head_lines c9 "例が使わない macro の UNMARKED_EXPANSIONS の行は、表の検が「行を外す」と名指して赤(S2(a) の失敗例)"
        val_row
        pytest_run -k "unmarked"; } > "$OUT/checks/c9-stale-row-red.txt" 2>&1
      restore ;;
    c10)
      { head_lines c10 "doeff-hy が隠す名の接頭辞 _dh_ を hidden_name に足しても、例の検は変えずに全部緑(S3 の正常例)"
        sed -i 's/^_HIDDEN_PREFIXES = ("_hy_", "__doeff_", "_doeff_")$/_HIDDEN_PREFIXES = ("_hy_", "__doeff_", "_doeff_", "_dh_")/' "$STUB"
        (cd "$W" && git diff | grep '^[-+]_HIDDEN')
        pytest_run; } > "$OUT/checks/c10-new-prefix-check-unchanged-green.txt" 2>&1
      restore ;;
    c11)
      { head_lines c11 "新しい接頭辞の名 _dh_helper を持つ写しは、hidden_name に接頭辞を足す前は印でなく、足した後は検を変えずに名指される(S3 の失敗例 — 検は hidden_name を呼ぶ)"
        echo "## 足す前: $(unplaceable_in '(_dh_helper (_doeff_do x))') / $(unplaceable_in '(_dh_helper x)')"
        sed -i 's/^_HIDDEN_PREFIXES = ("_hy_", "__doeff_", "_doeff_")$/_HIDDEN_PREFIXES = ("_hy_", "__doeff_", "_doeff_", "_dh_")/' "$STUB"
        echo "## 足した後: $(unplaceable_in '(_dh_helper x)')"; } > "$OUT/checks/c11-new-prefix-named.txt" 2>&1
      restore ;;
    *) echo "知らない確かめ: $check" >&2 ;;
  esac
done
restore
echo "作業木の git status(空なら戻した):"
git -C "$W" status --short
