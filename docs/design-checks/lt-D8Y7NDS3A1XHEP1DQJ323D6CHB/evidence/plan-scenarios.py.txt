# 送り先の決め(coordinator_log_plan)だけで S-store と S-dist を確かめる(code を変えずに env の値だけを変える)。
from doeff_cluster.worker.entry.line_stamp import coordinator_log_plan, JobLogTarget, JobLogRefused
POD = {"POD_NAME": "coordinator-x", "POD_NAMESPACE": "agent-worker"}
# S-store 正: 置き場を替える = 値 DOEFF_JOB_LOG_OTLP だけを替える。決めの他の欄は同じ。
a = coordinator_log_plan({**POD, "DOEFF_JOB_LOG_OTLP": "http://otel-collector.claude-telemetry.svc.cluster.local:4318"}, "1")
b = coordinator_log_plan({**POD, "DOEFF_JOB_LOG_OTLP": "https://other-store.example:4318/"}, "1")
print("S-store positive:", a.endpoint, "->", b.endpoint, "| resource 同じ:", a.resource == b.resource)
assert isinstance(a, JobLogTarget) and isinstance(b, JobLogTarget) and a.resource == b.resource
# S-store 負: 使えない値(http(s) の絶対の URL でない)は送らずに理由を返す(値そのものは書かない)。
c = coordinator_log_plan({**POD, "DOEFF_JOB_LOG_OTLP": "otel-collector:4318"}, "1")
print("S-store negative:", c)
assert isinstance(c, JobLogRefused) and "otel-collector" not in c.reason
# S-dist 正: 別の namespace の coordinator は job の名が namespace で分かれる。
d = coordinator_log_plan({"POD_NAME": "coordinator-y", "POD_NAMESPACE": "doeff-worker-lab", "DOEFF_JOB_LOG_OTLP": "http://c:4318"}, "1")
print("S-dist positive:", dict((x.key, x.value) for x in d.resource)["doeff.job"])
assert dict((x.key, x.value) for x in d.resource)["doeff.job"] == "coordinator/doeff-worker-lab"
# S-dist 負: 名乗りが欠ければ共有の名で混ぜずに送らない。
e = coordinator_log_plan({"POD_NAME": "coordinator-y", "DOEFF_JOB_LOG_OTLP": "http://c:4318"}, "1")
print("S-dist negative:", e)
assert e == JobLogRefused("coordinator の名乗りの環境変数が無い: POD_NAMESPACE")
print("ok")
