"""提示の pyright の検と同じ検体を doeff-hy-check(strict)に掛け、赤の中身を出す(実測用)。"""
import contextlib, io, json, sys
from pathlib import Path

sys.path.insert(0, "/home/ubuntu/.worktrees/doeff-wt-blind-b-ki-1618/packages/doeff-hy/tests")
import test_static_check_defhandler as t  # noqa: E402
from doeff_hy.static_check import main  # noqa: E402

root = Path(sys.argv[1])
root.mkdir(parents=True, exist_ok=True)
(root / "probe_effects.hy").write_text(t.NAMED_EFFECTS, encoding="utf-8")
(root / "probe.hy").write_text(t.NAMED_PROBE, encoding="utf-8")
out = io.StringIO()
with contextlib.redirect_stdout(out):
    main(["--root", str(root), "--json", "--no-cache", "--strict", str(root / "probe.hy")])
for d in json.loads(out.getvalue() or "[]"):
    if d["severity"] == "error":
        print(d.get("line"), d.get("rule"), d.get("message"))
