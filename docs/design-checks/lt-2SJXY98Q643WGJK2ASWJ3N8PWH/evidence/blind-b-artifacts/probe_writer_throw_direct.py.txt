"""書き手の `throw` を節の本体に直に書いた形(入れ子の関数なし)の答え(実測用)。本文は ValueError を受けて "rejected" を返す。"""
import sys
import types

import doeff_hy  # noqa: F401
import hy
from doeff import run

print("doeff_hy from:", doeff_hy.__file__)

HEADER = """
(require doeff-hy.macros [defk <- defhandler])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Charge [EffectBase] #^ int amount)
(defn throw [error] (raise error))
(setv log [])

(defhandler payments
  (Charge [amount]
    (when (< amount 0) (throw (ValueError (+ "negative: " (str amount)))))
    (.append log amount)
    (resume amount)))

(defk checkout [amount] {:pre [(: amount int)] :post [(: % (| int str))]}
  (try
    (do (<- r (Charge amount)) r)
    (except [e ValueError] "rejected")))

(defn pay [amount] (payments (checkout amount)))
"""

mod = types.ModuleType("writer_throw_direct")
sys.modules[mod.__name__] = mod
for form in hy.read_many(HEADER):
    hy.eval(form, mod.__dict__, module=mod)

for amount in (5, -5):
    del mod.log[:]
    try:
        answer = run(mod.pay(amount))
        print(f"pay({amount}) -> {answer!r}; handler log {mod.log}")
    except BaseException as error:  # noqa: BLE001
        print(f"pay({amount}) -> raised {type(error).__name__}: {str(error).splitlines()[-1][:160]}; handler log {mod.log}")
