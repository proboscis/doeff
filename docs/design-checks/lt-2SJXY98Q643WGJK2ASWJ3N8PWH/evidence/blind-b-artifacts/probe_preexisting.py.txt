"""既存の版(変更なし)で見る 2 点(実測用):
1. 欄の名が `effect` の効果(包みの効果)の節で :when が外れた時、外の handler へ渡るのは何か。
2. module の段の handle・defhandler の後、module の globals に増える展開の名。
"""
import sys
import types

import doeff_hy  # noqa: F401
import hy
from doeff import run
from doeff_core_effects import state
from doeff_core_effects.scheduler import scheduled

HEADER = """
(require doeff-hy.macros [defk <- handle defhandler])
(import dataclasses [dataclass])
(import doeff [EffectBase])
(defclass [(dataclass :frozen True)] Inner [EffectBase] #^ int x)
(defclass [(dataclass :frozen True)] Traced [EffectBase] #^ object effect #^ bool on)
"""

def ev(name, *sources):
    mod = types.ModuleType(name); sys.modules[name] = mod
    for src in sources:
        for f in hy.read_many(src):
            hy.eval(f, mod.__dict__, module=mod)
    return mod

m = ev("pre_traced", HEADER, """
(setv program
  (handle
    (handle (Traced (Inner 1) False)
      (Traced [on effect] :when on (resume "inner handler answered")))
    (Traced [effect on] (resume "outer saw Traced"))
    (Inner [x] (resume "outer saw Inner"))))
""")
print("1. :when 外れ ->", repr(run(m.program)), "(期待: 'outer saw Traced')")

before = set(vars(ev("pre_globals", HEADER)))
m2 = ev("pre_globals2", HEADER, """
(setv program (handle (Inner 1) (Inner [x] (resume x))))
(defhandler h (session var calls 0) (Inner [x] (:= calls (+ calls 1)) (resume calls)))
""")
added = sorted(n for n in set(vars(m2)) - before if n not in {"program", "h"})
print("2. module の globals に増えた名 ->", added)
