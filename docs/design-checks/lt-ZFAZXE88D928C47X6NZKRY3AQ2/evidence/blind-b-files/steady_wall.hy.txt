;;; worker の process の中の壁の時計(doeff-time の GetTime)を、起動の刻 + 単調な時計の経過で答える答え手 — 機体の壁の時計の
;;; step(NTP の修正・手での変更)を、worker の時間の比べ(子の停止の猶予 TERM → KILL・準備の作り直し・起こし直しの間・準備の停滞の
;;; 止め・途絶の柵と heartbeat の期限)へ入れない。壁の時計が進んだだけで沈黙が伸びて job を止める偽の自己停止も起きない。
;;; 入口(entry/main の production-handlers)が、機体の時計の答え手のすぐ内側に並べる。単調な時計(GetMonotonic)と待ち(Delay)は
;;; 外側の機体の時計のまま通す。
(require doeff-hy.macros [defhandler <- val])
(val MODULE-TAGS {:context "worker" :role "protocol"})
(import doeff_time [GetMonotonic GetTimeEffect])
(import doeff_cluster.shared.core.clock [datetime-of-epoch-ms monotonic-ms-of])


(defhandler steady-wall [#^ int started-ms #^ int started-mono-ms]
  ;; 引数に残す理由: 起動の刻の対(壁の時計の epoch ms と、同じ所で読んだ単調な時計の ms)は process ごとの値(入口が 1 度読む)。
  (GetTimeEffect []
    (<- mono float (GetMonotonic))
    (resume (datetime-of-epoch-ms (+ started-ms (- (monotonic-ms-of mono) started-mono-ms))))))
