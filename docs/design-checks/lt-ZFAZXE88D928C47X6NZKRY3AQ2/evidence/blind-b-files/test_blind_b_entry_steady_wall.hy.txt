;; 盲検 B(ki-069f)の失敗例の確かめ — 入口(worker/entry/main の production-handlers)が機体の時計のすぐ内側に並べる
;; steady-wall(worker/protocol/steady_wall.hy — GetTime を「起動の刻 + 単調な時計の経過」で答える)を、契約の検
;; tests/test_silence_wall_clock_step.hy と同じ台本・同じ偽の coordinator・同じ本物の口の下に、入口と同じ位置で並べる。
;;   対照(契約の検と同じ組 — steady-wall なし): 機体の suspend(単調は進まず壁だけ進む)の後、戻った最初の周で自己停止を決める。
;;   入口の組(steady-wall あり): 壁の差 ≡ 単調な差になり、suspend を数えない — 自己停止を決めない・fence の分だけ遅れる。
;; どの検も「観測した振る舞い」を assert する(入口の組の検は、責務違反が在る時に緑になる)。
(require doeff-hy.macros [deftest defk <- val])
(val MODULE-TAGS {:context "doeff-cluster-test" :role "test"})
(import doeff [with-handlers])
(import doeff_core_effects.handlers [slog-handler state])
(import doeff_core_effects.os_file [os-file-handler])
(import doeff_time [GetMonotonic SimClock sim-time-handler])
(import doeff_cluster.shared.core.clock [now-epoch-ms monotonic-ms-of])
(import doeff_cluster.worker.protocol.tick_pauses [tick-pauses])
(import doeff_cluster.worker.protocol.worker_wakes [no-wakes])
(import doeff_cluster.worker.protocol.coordinator_link [LinkState coordinator-link RESEND-AFTER-MS])
(import doeff_cluster.worker.protocol.steady_wall [steady-wall])
(import tests.link_rig [LINK-ROUTE])
(import tests.stop_fixtures [stop-signal-never-comes])
(import tests.transport_http [route-cell])
(import tests.test_silence_wall_clock_step [Script Seen WORKER T script-clock wall-clock-stepped scripted-coordinator cut-off-noted
                                            worker-then-seen cut-with-step silence-at-cut-off])

;; 機体が眠っていた秒(台本は壁の時計だけをこの分進める — 単調な時計は進まない)。移し替えの 60 秒より長い。
(val SLEPT-SECONDS 70.0)


(defk booted-then-seen [script task-dir entry-clock]
  {:pre [(: script Script) (: task-dir str) (: entry-clock bool)] :post [(: % Seen)] :tags {:context "doeff-cluster-test" :role "program"}}
  "入口(main)と同じ順で組むため: 機体の時計の下で起動の刻の対(壁・単調)を読み、その対で口の最初の起点を置き、entry-clock が真なら
   同じ対の steady-wall を機体の時計のすぐ内側(偽の coordinator・待ち・口の外側)に並べて、本物の調整ループを回す。"
  (<- started-ms int (now-epoch-ms))
  (<- mono float (GetMonotonic))
  (val started-mono-ms (monotonic-ms-of mono))
  (val link (LinkState WORKER #("cpu") 1 0 T.fence-ms task-dir "boot" started-ms started-ms started-mono-ms :watch True))
  (<- seen Seen (with-handlers (+ (if entry-clock [(steady-wall started-ms started-mono-ms)] [])
                                  [(scripted-coordinator script) tick-pauses
                                   (coordinator-link link (route-cell "http://coord") LINK-ROUTE (route-cell "http://coord"))
                                   cut-off-noted])
                  (worker-then-seen)))
  seen)


(defk seen-on [script task-dir entry-clock]
  {:pre [(: script Script) (: task-dir str) (: entry-clock bool)] :post [(: % Seen)] :tags {:context "doeff-cluster-test" :role "entry"}}
  "台本 script を、機体の時計(仮想の時計と、壁の時計の代役 wall-clock-stepped)の下で回すため(契約の検の seen-on と同じ外側の組)。"
  (<- seen Seen (with-handlers [(state) (sim-time-handler :clock (SimClock)) os-file-handler slog-handler no-wakes stop-signal-never-comes
                                script-clock wall-clock-stepped]
                  (booted-then-seen script task-dir entry-clock)))
  seen)


;; --- 台本 A: 契約の検 test-a-worker-resumed-from-a-suspend-past-the-fence-… と同じ(届いている worker が 30 秒眠る)-------------

(val SUSPEND-IN-TOUCH (Script :cut-ms None :cut-back-seconds 0.0 :step-after-beats 3 :beats-back-seconds -30.0 :until-ms 20000))


(deftest test-control-the-contract-stack-decides-the-self-stop-on-resume [tmp-path]
  ;; 対照: 契約の検と同じ組(steady-wall なし)では、戻った最初の周で自己停止を決める(契約の検と同じ assert)。
  (<- seen Seen (seen-on SUSPEND-IN-TOUCH (str (/ tmp-path "tasks")) False))
  (assert (is-not seen.stepped-ms None) seen)
  (assert (is-not seen.cut-off-ms None) seen)
  (assert (<= seen.stepped-ms seen.cut-off-ms (+ seen.stepped-ms (// T.lease-ms 2))) seen)
  (print "BLIND-B A/control: stepped-ms" seen.stepped-ms "cut-off-ms" seen.cut-off-ms "delivered" (len seen.delivered)))


(deftest test-violation-the-entry-stack-never-decides-the-self-stop-after-the-same-suspend [tmp-path]
  ;; 責務違反の観測: 入口と同じ位置に steady-wall を並べると、同じ台本で自己停止を 1 度も決めない(契約の検の assert
  ;; 「cut-off-ms is not None」が赤になる形)。眠りは起きている(stepped-ms は在る)。
  (<- seen Seen (seen-on SUSPEND-IN-TOUCH (str (/ tmp-path "tasks")) True))
  (assert (is-not seen.stepped-ms None) seen)
  (assert (is seen.cut-off-ms None) seen)
  (print "BLIND-B A/entry: stepped-ms" seen.stepped-ms "cut-off-ms" seen.cut-off-ms "delivered" (len seen.delivered)))


;; --- 台本 B: 届かなくなった時に機体が 70 秒眠り、戻っても届かない(移し替えの 60 秒は眠っている間に過ぎている)--------------

(deftest test-control-the-contract-stack-stops-at-once-when-it-wakes-cut-off [tmp-path]
  ;; 対照: 届かなかった最初の heartbeat の時に壁が 70 秒進む(= 眠っていた)。沈黙は 70 秒と読まれ、その送りの失敗の枝で
  ;; すぐ自己停止を決める — 起きてから動かし続ける時間は heartbeat の間隔と送り直しの間の内。
  (<- script Script (cut-with-step (- SLEPT-SECONDS)))
  (<- seen Seen (seen-on script (str (/ tmp-path "tasks")) False))
  (<- awake int (silence-at-cut-off seen))
  (assert (< awake (+ (// T.lease-ms 4) (* 2 RESEND-AFTER-MS))) #(awake seen))
  (print "BLIND-B B/control: last-delivered" (get seen.delivered -1) "stepped-ms" seen.stepped-ms "cut-off-ms" seen.cut-off-ms "awake-ms" awake))


(deftest test-violation-the-entry-stack-keeps-the-job-a-whole-fence-after-waking-cut-off [tmp-path]
  ;; 責務違反の観測: 入口の組では眠りを数えず、起きてから fence(20 秒)の分だけ動かし続けてから自己停止を決める。実際に過ぎた
  ;; 時間(眠り 70 秒 + 起きていた時間)は移し替えの 60 秒を越えている — coordinator が他へ置いた job と 2 か所で走る時間帯。
  (<- script Script (cut-with-step (- SLEPT-SECONDS)))
  (<- seen Seen (seen-on script (str (/ tmp-path "tasks")) True))
  (<- awake int (silence-at-cut-off seen))
  (assert (< T.fence-ms awake (+ T.fence-ms (* 2 RESEND-AFTER-MS))) #(awake seen))
  (assert (> (+ (int (* 1000 SLEPT-SECONDS)) awake) T.reassign-after-ms) #(awake seen))
  (print "BLIND-B B/entry: last-delivered" (get seen.delivered -1) "stepped-ms" seen.stepped-ms "cut-off-ms" seen.cut-off-ms "awake-ms" awake
         "real-silence-ms" (+ (int (* 1000 SLEPT-SECONDS)) awake)))
