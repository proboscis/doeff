;; 本番と同じ時間の設定(lease 10 秒・fence 20 秒・移し替え 60 秒)。
(val T (ClusterTiming))
;; suspend の長さ(fence 20 秒の後)。
(val SUSPEND-SECONDS 30.0)


(defhandler suspended-then-cut [#^ bool inside-reply #^ list noted]
  "口と偽の coordinator の間に立つ代役: 3 回目に届いた heartbeat の拍で壁の時計だけを SUSPEND-SECONDS 進め(単調な時計は動かさない —
   suspend の形)、その後の要求は全部 接続の失敗にする(戻ったら網が無い機体)。進める所は 2 通り:
     inside-reply 真  3 回目の ReadEnvironment(= 返事の処理の途中 — 口はもう返事の刻を読んでいる・調整ループはまだ壁の時計を読んでいない)
     inside-reply 偽  その拍の最初の ObserveWorld(調整ループが壁の時計を読んだ後 — 既に在る suspend の台本と同じ所・対照)
   noted に進めた刻(台本の時計 ms)を残す。"
  {:tags {:context "doeff-cluster-test" :role "foundation"}}
  ;; 引数に残す理由: 検ごとに進める所を変え、進めた刻を検が読む(Ask で区別できない)。
  (session var asked 0)
  (session var armed False)
  (ReadEnvironment [names]
    (:= asked (+ asked 1))
    (when (= asked 3)
      (if inside-reply
          (do (<- at int (ScriptClockMs))
              (.append noted at)
              (<- (StepWallClock (- SUSPEND-SECONDS))))
          (:= armed True)))
    (<- found effect)
    (resume found))
  (ObserveWorld []
    (when armed
      (:= armed False)
      (<- at int (ScriptClockMs))
      (.append noted at)
      (<- (StepWallClock (- SUSPEND-SECONDS))))
    (<- view effect)
    (resume view))
  (HttpRequest [url]
    (when noted
      (return (resume (HttpFailed :url url :detail "ConnectError: refused" :kind HttpFailureKind.CONNECT-FAILED))))
    (<- answer effect)
    (resume answer)))


(defk seen-under [inside-reply beat-beside task-dir noted]
  {:pre [(: inside-reply bool) (: beat-beside bool) (: task-dir str) (: noted list)] :post [(: % Seen)]
   :tags {:context "doeff-cluster-test" :role "entry"}}
  "台本(台本の側では途絶えず壁の時計も動かさない・300 秒動かす)を、元の検と同じ組の口のすぐ外側に代役 suspended-then-cut を足して回すため。"
  (val script (Script :cut-ms None :cut-back-seconds 0.0 :step-after-beats None :beats-back-seconds 0.0 :until-ms 300000))
  (val link (LinkState WORKER #("cpu") 1 0 T.fence-ms task-dir "boot" 0 0 0 :watch True :beat-beside beat-beside))
  (<- seen Seen (with-handlers [(state) (sim-time-handler :clock (SimClock)) os-file-handler slog-handler no-wakes stop-signal-never-comes
                                script-clock wall-clock-stepped (scripted-coordinator script) (suspended-then-cut inside-reply noted)
                                tick-pauses
                                (coordinator-link link (route-cell "http://coord") LINK-ROUTE (route-cell "http://coord"))
                                cut-off-noted]
                  (worker-then-seen)))
  seen)


(deftest test-control-a-suspend-after-the-tick-read-its-clock-stops-at-the-first-tick-after-the-resume [tmp-path]
  ;; 対照(既に在る suspend の台本と同じ所で進める): 戻った最初の周で自己停止を決める。
  (val noted [])
  (<- seen Seen (seen-under False True (str (/ tmp-path "tasks")) noted))
  (print "CONTROL noted" noted "delivered" seen.delivered "cut-off-ms" seen.cut-off-ms)
  (assert (is-not seen.cut-off-ms None) seen)
  (assert (<= (get noted 0) seen.cut-off-ms (+ (get noted 0) (// T.lease-ms 2))) #(noted seen)))


(deftest test-a-suspend-inside-the-reply-handling-stops-at-the-first-tick-after-the-resume [tmp-path]
  ;; 同じ主張(#2806 — 戻った最初の周で、返事を待たずに自己停止を決める)を、suspend の刻だけ替えて確かめる。
