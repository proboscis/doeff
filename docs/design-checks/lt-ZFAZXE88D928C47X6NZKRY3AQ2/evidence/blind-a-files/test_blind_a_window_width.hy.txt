(defclass Probe []
  "計りの入れ物: phase = 局面(None・\"wakes\" = WorkerWakes を出した後・\"await\" = AwaitNextTick を出した後)・link-read = 口が壁の時計を
   読んだ実時間・widths = 窓の実時間(秒)の列。"
  (defn #^ None __init__ [self]
    (setv self.phase None self.link-read None self.widths [])))


(defhandler phase-marked [#^ Probe probe]
  "調整ループのすぐ外に立つ代役: WorkerWakes と AwaitNextTick を出した事を局面に立てて通す。"
  {:tags {:context "doeff-cluster-test" :role "foundation"}}
  ;; 引数に残す理由: 検が計りの入れ物を読む(Ask で区別できない)。
  (WorkerWakes [began began-mono-ms]
    (setv probe.phase "wakes" probe.link-read None)
    (<- wakes effect)
    (resume wakes))
  (AwaitNextTick [policy changed wakes stopping]
    (setv probe.phase "await")
    (<- waited effect)
    (resume waited)))


(defhandler wall-reads-timed [#^ Probe probe]
  "口と tick-pauses より外側に立つ代役: 壁の時計の読み(GetTime)を見て、局面ごとの最初の読みの実時間を残す。"
  {:tags {:context "doeff-cluster-test" :role "foundation"}}
  ;; 引数に残す理由: 検が計りの入れ物を読む(Ask で区別できない)。
  (GetTimeEffect []
    (when (and (= probe.phase "wakes") (is probe.link-read None))
      (setv probe.link-read (time.perf-counter)))
    (when (and (= probe.phase "await") (is-not probe.link-read None))
      (.append probe.widths (- (time.perf-counter) probe.link-read))
      (setv probe.phase None probe.link-read None))
    (<- now effect)
    (resume now)))


(defk seen-with-probe [task-dir probe]
  {:pre [(: task-dir str) (: probe Probe)] :post [(: % Seen)] :tags {:context "doeff-cluster-test" :role "entry"}}
  "途絶の台本(10 秒目から届かない・110 秒動かす)を、元の検と同じ組に計りの代役 2 つを足して回すため。"
  (val script (Script :cut-ms 10000 :cut-back-seconds 0.0 :step-after-beats None :beats-back-seconds 0.0 :until-ms 110000))
  (val link (LinkState WORKER #("cpu") 1 0 T.fence-ms task-dir "boot" 0 0 0 :watch True))
  (<- seen Seen (with-handlers [(state) (sim-time-handler :clock (SimClock)) os-file-handler slog-handler no-wakes stop-signal-never-comes
                                script-clock wall-clock-stepped (scripted-coordinator script) (wall-reads-timed probe) tick-pauses
                                (coordinator-link link (route-cell "http://coord") LINK-ROUTE (route-cell "http://coord"))
                                cut-off-noted (phase-marked probe)]
                  (worker-then-seen)))
  seen)


(deftest test-measure-the-window-between-the-two-wall-clock-reads [tmp-path]
  (val probe (Probe))
  (<- seen Seen (seen-with-probe (str (/ tmp-path "tasks")) probe))
  (val ms (sorted (lfor w probe.widths (* 1000 w))))
  (print "WINDOW ticks" (len ms) "min-ms" (round (get ms 0) 3) "median-ms" (round (statistics.median ms) 3)
         "p95-ms" (round (get ms (int (* 0.95 (len ms)))) 3) "max-ms" (round (get ms -1) 3))
  (assert (> (len ms) 100) (len ms)))
