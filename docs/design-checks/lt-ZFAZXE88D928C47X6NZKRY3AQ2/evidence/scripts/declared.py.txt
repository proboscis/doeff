"""agora-controllers の歩数の宣言(declare-step-budget)の一覧を作り、束に分ける / 測りの log と突き合わせる(card ki-069f4a5c07f7)。

使い方:
  uv run --no-project python declared.py list <agora-controllers の作業木> <出力 dir> <束の大きさ>   → <出力 dir>/batch-NN に検の id の列
  uv run --no-project python declared.py compare <agora-controllers の作業木> <log の glob>          → 宣言と走りの歩数の表(超えた物に印)
"""
import glob
import re
import sys
from pathlib import Path

DECLARED = re.compile(r'^\(declare-step-budget (test-[a-z0-9-]+) (\d+) "', re.MULTILINE)
LINE = re.compile(r"^  (?P<file>controllers/\S+?)::(?P<test>test_[a-z0-9_]+): .*doeff-vm の歩数 (?P<all>\d+).世界の走り (?P<run>\d+)")


def declarations(root: Path) -> list[tuple[str, str, int]]:
    rows = []
    for path in sorted((root / "controllers").rglob("*.hy")):
        text = path.read_text(encoding="utf-8")
        for match in DECLARED.finditer(text):
            rows.append((str(path.relative_to(root)), match[1].replace("-", "_"), int(match[2])))
    return rows


def main(argv: list[str]) -> int:
    mode, root = argv[0], Path(argv[1])
    rows = declarations(root)
    if mode == "list":
        out, size = Path(argv[2]), int(argv[3])
        ids = [f"{file}::{test}" for file, test, _ in rows]
        for index in range(0, len(ids), size):
            (out / f"batch-{index // size:02d}").write_text("\n".join(ids[index : index + size]) + "\n", encoding="utf-8")
        print(len(ids), "ids", (len(ids) + size - 1) // size, "batches")
        return 0
    measured: dict[tuple[str, str], int] = {}
    for path in sorted(glob.glob(argv[2])):
        for line in Path(path).read_text(encoding="utf-8").splitlines():
            match = LINE.match(line)
            if match:
                measured[(match["file"], match["test"])] = int(match["run"])
    over = missing = 0
    for file, test, declared in rows:
        run = measured.get((file, test))
        if run is None:
            missing += 1
            print(f"未測 {file}::{test} 宣言 {declared}")
        elif run > declared:
            over += 1
            print(f"超え {file}::{test} 宣言 {declared} 走り {run} (+{run - declared})")
    print(f"宣言 {len(rows)} 本・測れた {len(rows) - missing} 本・超え {over} 本・未測 {missing} 本")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
