(val T (ClusterTiming))


(defhandler boottime-monotonic
  "眠りを数える単調な時計の機体の代役(入口が並べる時計の答え手の位置 — worker の側の全部の外側・偽の coordinator の時計の内側):
   機体の眠り(台本が壁の時計を進める step — 負の秒)の分だけ GetMonotonic の答えも進める。壁の時計を戻す step(NTP の修正 — 正の秒)
   では動かさない。"
  {:tags {:context "doeff-cluster-test" :role "foundation"}}
  (session var slept 0.0)
  (StepWallClock [seconds]
    (when (< seconds 0)
      (:= slept (- slept seconds)))
    (<- effect)
    (resume None))
  (GetMonotonicEffect []
    (<- mono float effect)
    (resume (+ mono slept))))


(defk seen-on-a-boottime-machine [script task-dir]
  {:pre [(: script Script) (: task-dir str)] :post [(: % Seen)] :tags {:context "doeff-cluster-test" :role "entry"}}
  "台本 script を、元の検の seen-on と同じ組に boottime-monotonic を 1 つ足して回すため(壁の時計の代役のすぐ内側)。"
  (val link (LinkState WORKER #("cpu") 1 0 T.fence-ms task-dir "boot" 0 0 0 :watch True))
  (<- seen Seen (with-handlers [(state) (sim-time-handler :clock (SimClock)) os-file-handler slog-handler no-wakes stop-signal-never-comes
                                script-clock wall-clock-stepped boottime-monotonic (scripted-coordinator script) tick-pauses
                                (coordinator-link link (route-cell "http://coord") LINK-ROUTE (route-cell "http://coord"))
                                cut-off-noted]
                  (worker-then-seen)))
  seen)


(deftest test-s1-a-suspend-on-a-boottime-machine-decides-the-self-stop-at-the-first-tick [tmp-path]
  ;; 届いている worker が 3 回目の heartbeat の後に 30 秒眠る(壁も単調も 30 秒進む)。戻った最初の周で自己停止を決める。
  (val script (Script :cut-ms None :cut-back-seconds 0.0 :step-after-beats 3 :beats-back-seconds -30.0 :until-ms 20000))
  (<- seen Seen (seen-on-a-boottime-machine script (str (/ tmp-path "tasks"))))
  (print "S1 suspend: stepped-ms" seen.stepped-ms "cut-off-ms" seen.cut-off-ms "delivered" seen.delivered)
  (assert (is-not seen.stepped-ms None) seen)
  (assert (is-not seen.cut-off-ms None) seen)
  (assert (<= seen.stepped-ms seen.cut-off-ms (+ seen.stepped-ms (// T.lease-ms 2))) seen))


(deftest test-s1-a-wall-clock-stepped-back-45-seconds-on-a-boottime-machine-does-not-delay-the-self-stop [tmp-path]
  ;; 途絶の最中に壁の時計が 45 秒戻る(単調な時計は動かない)。対照と同じ刻(fence の所)に自己停止を決める。
  (<- control-script Script (cut-with-step 0.0))
  (<- control Seen (seen-on-a-boottime-machine control-script (str (/ tmp-path "control"))))
  (<- script Script (cut-with-step 45.0))
  (<- seen Seen (seen-on-a-boottime-machine script (str (/ tmp-path "stepped"))))
  (<- silence int (silence-at-cut-off seen))
  (<- control-silence int (silence-at-cut-off control))
  (print "S1 back-45: silence-at-cut-off-ms" silence "control" control-silence)
  (assert (= silence control-silence) #(silence control-silence seen))
  (assert (< T.fence-ms silence (+ T.fence-ms (* 2 RESEND-AFTER-MS))) #(silence seen)))


(deftest test-s1-a-wall-clock-stepped-back-70-seconds-on-a-boottime-machine-does-not-hold-the-heartbeats-back [tmp-path]
  ;; 届いている worker の壁の時計が 70 秒戻る。heartbeat の間はどれも lease の半分以下で、自己停止は決めない。
  (val script (Script :cut-ms None :cut-back-seconds 0.0 :step-after-beats 3 :beats-back-seconds 70.0 :until-ms 100000))
  (<- seen Seen (seen-on-a-boottime-machine script (str (/ tmp-path "tasks"))))
  (val gaps (tuple (gfor #(earlier later) (zip seen.delivered (cut seen.delivered 1 None)) (- later earlier))))
  (print "S1 back-70: max-gap-ms" (max gaps) "cut-off-ms" seen.cut-off-ms)
  (assert (<= (max gaps) (// T.lease-ms 2)) #((max gaps) seen))
  (assert (is seen.cut-off-ms None) seen))
