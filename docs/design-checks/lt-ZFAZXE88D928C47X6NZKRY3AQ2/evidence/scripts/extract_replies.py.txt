"""会話の記録から、盲検 A・B の subagent の返答(Agent tool の結果)を加工せずに取り出す。harness が付けた行頭の字下げ 2 字と、枠の頭の 1 行・末尾の agentId / usage の行だけを分ける。"""
import json
import sys
from pathlib import Path

transcript, out = Path(sys.argv[1]), Path(sys.argv[2])
found = []
for line in transcript.read_text(encoding="utf-8").splitlines():
    if "Subagent hand-back" not in line:
        continue
    try:
        row = json.loads(line)
    except ValueError:
        continue
    content = row.get("message", {}).get("content")
    if not isinstance(content, list):
        continue
    for part in content:
        if part.get("type") != "tool_result":
            continue
        body = part.get("content")
        text = "".join(p.get("text", "") for p in body) if isinstance(body, list) else str(body)
        if "Subagent hand-back" in text and ("blind-a-ki069f" in text or "blind-b-ki069f" in text):
            found.append(text)
print(len(found))
for text in found:
    which = "a" if "/tmp/blind-a-ki069f" in text else "b"
    head, _, rest = text.partition("The report follows:\n")
    reply, _, tail = rest.partition("\nagentId:")
    lines = [l[2:] if l.startswith("  ") else l for l in reply.split("\n")]
    (out / f"blind-{which}-reply.md").write_text("\n".join(lines).rstrip("\n") + "\n", encoding="utf-8")
    print(which, len(lines), "agentId:" + tail.replace("\n", " | ")[:300])
