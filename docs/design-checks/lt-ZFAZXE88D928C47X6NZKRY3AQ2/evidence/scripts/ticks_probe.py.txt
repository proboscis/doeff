"""途絶の間の worker の周の数と doeff-vm の歩数を測る(card ki-069f4a5c07f7 — 沈黙の計りの変更が周の数を変えていないかを見る)。

使い方: doeff の作業木の packages/doeff-cluster で `uv run python <この file> <途絶の秒>`。
"""
import sys

import hy  # noqa: F401 — .hy の import を有効にする
from doeff import do, run

import doeff_vm.doeff_vm as ext
from doeff_events import MemoryBroker
from doeff_time import Delay

sys.path.insert(0, ".")
from doeff_cluster.shared.intent.protocol import ClusterTiming  # noqa: E402
from doeff_cluster.sim.local import CutWorker, HostTruthOf, SimWorker, sim_cluster  # noqa: E402
from tests.fixtures.envs import sim_foundation  # noqa: E402
from tests.fixtures.sim_programs import pulses  # noqa: E402

cut_seconds = float(sys.argv[1])
net = frozenset(["cluster-net"])
workers = (SimWorker(name="w1", provides=net, task_reserve=0),)


@do
def scenario():
    yield Delay(8.0)
    before = yield HostTruthOf("w1")
    yield CutWorker("w1", cut_seconds)
    yield Delay(cut_seconds)
    after = yield HostTruthOf("w1")
    return (before.ticks, after.ticks, after.beats)


start = ext.vm_work_counts()
ticks = run(sim_cluster(pulses(sim_foundation), scenario(), notice_broker=MemoryBroker(), timing=ClusterTiming(), workers=workers))
end = ext.vm_work_counts()
print(f"cut={cut_seconds} ticks_before_cut={ticks[0]} ticks_after_cut={ticks[1]} beats={ticks[2]} steps={end[0] - start[0]}")
