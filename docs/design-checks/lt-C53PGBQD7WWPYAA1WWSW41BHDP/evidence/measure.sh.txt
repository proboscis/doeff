run_check() {  # 例の検の全部を走らせ、各検の合否と要約を出す
  echo "HEAD=$(git log -1 --format=%H)"
  echo "木の差分(docs/ の外):"; git status --short | grep -v ' docs/' || echo "(なし)"
  echo "\$ uv run pytest $F -q -p no:randomly -rA --tb=line $*"
  uv run pytest "$F" -q -p no:randomly -rA --tb=line "$@" 2>&1 | grep -E '^(PASSED|FAILED|ERROR|SKIPPED)|^E? *AssertionError|^/.*AssertionError|passed|failed' | cut -c1-400
  echo "exit=${PIPESTATUS[0]}"
}
