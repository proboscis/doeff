;;; Example 10: for/do の読み方 — From 1 つの for/do が何に展開されるかを見て、同じ for/do を走らせる。
;;;
;;; for/do は「件ごとの関数」と items を Traverse の effect 1 つに包む。件をどう走らせるか(順に・並べて)と、
;;; 件の失敗をどう扱うかは、外から積む handler が決める。

(require doeff-hy.macros [defk <- for/do])
(import doeff [do :as _doeff-do])
(import doeff [run])

(import doeff_core_effects [try-handler :as try_handler])
(import doeff_core_effects.scheduler [scheduled])

(import doeff_traverse [Traverse :as _doeff_traverse_Traverse])
(import doeff_traverse.collection [Collection])
(import doeff_traverse.handlers [sequential fail-handler :as fail_handler])

(print "=== for/do の展開 ===")
(print (hy.repr (hy.macroexpand '(for/do (<- x (From items :label "double")) (* x 2)))))
(print)

(defk doubled [items]
  {:pre [(: items list)]
   :post [(: % Collection)]}
  (<- result (for/do (<- x (From items :label "double")) (* x 2)))
  result)

(print "=== 走らせた答え ===")
(setv out (run (scheduled ((sequential) (fail_handler (try_handler (doubled [1 2 3])))))))
(print "Valid:" (. out valid_values))
