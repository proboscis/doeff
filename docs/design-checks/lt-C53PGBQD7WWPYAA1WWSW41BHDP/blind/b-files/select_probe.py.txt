"""変えた path の組ごとに、本物の表(pyproject の [[tool.doeff.contract-tests]])が選ぶ検を印字する(入口と同じ select_targets)。"""
import importlib.util
import sys
from pathlib import Path

repo = Path(sys.argv[1])
spec = importlib.util.spec_from_file_location("run_changed_tests", repo / "scripts" / "run_changed_tests.py")
module = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = module
spec.loader.exec_module(module)

table = module.read_contract_table(repo)
patterns = module.collector_patterns(repo)
for files in (
    ("packages/doeff-traverse/examples/ex04_traverse_macro.hy",
     "packages/doeff-traverse/examples/expected/ex04_traverse_macro.txt"),
    ("packages/doeff-hy/src/doeff_hy/macros.hy",),
    ("uv.lock",),
):
    selection = module.select_targets(table, module.ChangeSet(base="x", files=files), patterns, repo)
    print("changed:", files)
    for t in selection.targets:
        print("   ", t.arg, "|", t.why)
