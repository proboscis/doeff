;; 試し(設計検証 lt-6N1Z・S2/S6): 焼きの道具の入口の usable-cpus(M2)を、cgroup の file の読み(ReadText)に答える偽の答え手の下で回し、
;; 本物の file を読まずに既定の並列数を確かめる。affinity は本物の値(この Pod は 36)。
(require doeff-hy.macros [defhandler defk <- val])
(import os)
(import doeff [run with-handlers])
(import doeff_core_effects.file_effects [ReadText FileFailed])
(import doeff_cluster.worker.entry.code_prepare [usable-cpus])

(defhandler cpu-max-file [answer]
  ;; 引数に残す理由: 試しごとに cgroup の file の中身(か読めない失敗)を変える。
  (ReadText [path]
    :when (= path "/sys/fs/cgroup/cpu.max")
    (resume answer)))

(val affinity (len (os.sched-getaffinity 0)))
(print "affinity" affinity)
(for [#(label answer) [#("上限なし(zeus)" "max 100000\n")
                       #("pod の上限 2(k3s)" "200000 100000\n")
                       #("pod の上限 8(atlas)" "800000 100000\n")
                       #("file が無い(cgroup v1 の node)" (FileFailed :path "/sys/fs/cgroup/cpu.max" :detail "No such file"))
                       #("読めない中身" "garbage\n")]]
  (val jobs (run (with-handlers [(cpu-max-file answer)] (usable-cpus))))
  (print label "→" jobs)
  (assert (<= 1 jobs 8) #(label jobs)))
(print "ok: どの答えでも 1 ≤ 既定 ≤ 8")
