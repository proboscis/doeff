+class SetTaskPriority(EffectBase[None]):
+    """Change a live task's priority; the requester resumes immediately.
+
+    Spawn fixes a task's priority when it is spawned. A long-lived task whose
+    work turns into background work (a turn waiting for its user) can be
+    lowered — e.g. to ``PRIORITY_IDLE`` — and raised again later.
+
+    - Every later wake of the task uses the new priority (as if it had been
+      spawned with it, #504).
+    - Entries the task already has in the ready heap move to the new
+      priority; FIFO order among equal priorities is kept.
+    - Changing the priority of a terminal task is a no-op.
+    """
+    def __init__(self, task: "Task[Any]", priority: int) -> None:
+        super().__init__()
+        self.task = task
+        self.priority = priority
+

+# ---------------------------------------------------------------------------
+# SetTaskPriority — change a live task's priority

+class TestSetTaskPriority:
+    def test_lowered_parked_task_wakes_behind_normal_work(self):
+        """A task lowered while parked wakes at its new priority."""
+        from doeff_core_effects.scheduler import SetTaskPriority
+
+        order: list[str] = []
+
+        @do
+        def worker(gate: Any, name: str):
+            yield Wait(gate.future)
+            order.append(name)
+
+        @do
+        def body():
+            gate = yield CreatePromise()
+            lowered = yield Spawn(worker(gate, "lowered"))
+            normal = yield Spawn(worker(gate, "normal"))
+            yield SetTaskPriority(lowered, PRIORITY_IDLE)
+            yield CompletePromise(gate, None)
+            yield Gather(lowered, normal)
+            return order
+
+        assert doeff_run(scheduled(body())) == ["normal", "lowered"]
+
+    def test_lowered_queued_task_moves_behind_normal_work(self):
+        """A resume already queued moves to the new priority."""
+        from doeff_core_effects.scheduler import SetTaskPriority
+
+        order: list[str] = []
+
+        @do
+        def worker(gate: Any, name: str):
+            yield Wait(gate.future)
+            order.append(name)
+
+        @do
+        def body():
+            gate = yield CreatePromise()
+            first = yield Spawn(worker(gate, "first"))
+            second = yield Spawn(worker(gate, "second"))
+            # Both resumes are queued at NORMAL (first ahead); the body keeps its turn.
+            yield CompletePromise(gate, None, yield_to_woken=False)
+            yield SetTaskPriority(first, PRIORITY_IDLE)
+            yield Gather(first, second)
+            return order
+
+        assert doeff_run(scheduled(body())) == ["second", "first"]
+
+    def test_raised_queued_task_runs_first(self):

+        @do
