"""card ki-1294af4fa525: scheduler の普段の歩(待つ task が少ない時)の CPU が、索引と数えの記帳で重くなっていないかを両版で比べるため。

使い方: uv run --project <doeff の worktree> python step_cost.py
形 3 つの scheduler の thread の CPU(thread_time)を 5 回測り、中央値を出す:
  spawn-wait  = 中身の無い task の Spawn と Wait を 20,000 回
  threads-100 = test_100_threads_concurrent と同じ形(100 本が thread の 0.1 秒の後に外の約束で起きる)
  timed-wait  = 本番の層 2 の呼び鈴の形(WaitWithin park=True)を、ほかに待つ task 16 本の下で 2,000 回
"""
import statistics
import sys
import threading
import time

from doeff import do, run, with_handlers
from doeff_core_effects.handlers import await_handler
from doeff_core_effects.scheduler import Cancel, CreateExternalPromise, Gather, Spawn, Wait, scheduled
from doeff_time import WaitWithin, async_time_handler


@do
def noop():
    return None


@do
def spawn_wait(count):
    for _ in range(count):
        task = yield Spawn(noop())
        yield Wait(task)


@do
def sleep_task(i):
    ep = yield CreateExternalPromise()

    def worker():
        time.sleep(0.1)
        ep.complete(i)

    threading.Thread(target=worker, daemon=True).start()
    return (yield Wait(ep.future))


@do
def threads_100():
    tasks = ()
    for i in range(100):
        tasks = (*tasks, (yield Spawn(sleep_task(i))))
    return (yield Gather(*tasks))


def ringer(bells):
    """呼び鈴を少し後に鳴らす(本番の CLI の行の代わり)。"""
    while (bell := bells.get()) is not None:
        time.sleep(0.0002)
        bell.complete(True)


@do
def parked_waiter():
    bell = yield CreateExternalPromise()
    yield WaitWithin(bell.future, 3600.0, park=True)


@do
def timed_waits(count, parked, bells):
    tasks = ()
    for _ in range(parked):
        tasks = (*tasks, (yield Spawn(parked_waiter(), daemon=True)))
    for _ in range(count):
        bell = yield CreateExternalPromise()
        bells.put(bell)
        yield WaitWithin(bell.future, 3600.0, park=True)
    for task in tasks:
        yield Cancel(task)


def cpu_of(make):
    t0 = time.thread_time()
    run(scheduled(make(), implementation="python"))
    return time.thread_time() - t0


def main():
    import queue

    bells = queue.SimpleQueue()
    threading.Thread(target=ringer, args=(bells,), daemon=True).start()
    shapes = {
        "spawn-wait 20,000": lambda: spawn_wait(20_000),
        "threads-100": threads_100,
        "timed-wait 2,000 (parked 16)": lambda: with_handlers(
            [await_handler(), async_time_handler()], timed_waits(2_000, 16, bells)
        ),
    }
    import doeff_core_effects.scheduler as s
    print("scheduler:", s.__file__)
    for name, make in shapes.items():
        if sys.argv[1:] and not any(name.startswith(prefix) for prefix in sys.argv[1:]):
            continue
        cpus = [cpu_of(make) for _ in range(5)]
        print(f"{name}: CPU 中央値 {statistics.median(cpus) * 1000:.0f} ms(5 回 {[round(c * 1000) for c in cpus]})", flush=True)
    bells.put(None)


if __name__ == "__main__":
    sys.exit(main())
