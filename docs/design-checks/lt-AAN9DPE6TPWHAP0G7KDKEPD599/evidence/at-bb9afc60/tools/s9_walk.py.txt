"""card ki-1294af4fa525 の設計検証 S9: 同じ WaitWithin の形(本番の層 2 の呼び鈴・park=True)を、ほかに待つ task 16 本と 256 本の下で
Python と Rust の scheduler で回し、scheduler の thread の CPU を比べる。索引を入れたのは Python だけ(Rust は今も表をなめる)と
いう §2 の記述を、数で確かめるため(repo の外の一時の道具)。使い方: uv run python s9_walk.py [回数]"""
import gc
import os
import queue
import statistics
import sys
import threading
import time

from doeff import do, run, with_handlers
from doeff_core_effects.handlers import await_handler
from doeff_core_effects.scheduler import Cancel, CreateExternalPromise, Spawn, scheduled
from doeff_time import WaitWithin, async_time_handler


# env S9_GC=off で測る間 gc を止める(free-threaded の Python の gc は生きた object の全部をなめ、待つ task の数で増える — scheduler の外)。
GC = os.environ.get("S9_GC", "on") != "off"
# 比べる待つ task の数(env S9_PARKED=16,1024)と繰り返し(env S9_REPEAT)。
PARKED = tuple(int(n) for n in os.environ.get("S9_PARKED", "16,256").split(","))
REPEAT = int(os.environ.get("S9_REPEAT", "3"))


def ringer(bells):
    while (bell := bells.get()) is not None:
        time.sleep(0.0002)
        bell.complete(True)


@do
def parked_waiter():
    bell = yield CreateExternalPromise()
    yield WaitWithin(bell.future, 3600.0, park=True)


@do
def timed_waits(count, parked, bells):
    tasks = ()
    for _ in range(parked):
        tasks = (*tasks, (yield Spawn(parked_waiter(), daemon=True)))
    for _ in range(count):
        bell = yield CreateExternalPromise()
        bells.put(bell)
        yield WaitWithin(bell.future, 3600.0, park=True)
    for task in tasks:
        yield Cancel(task)


def cpu_total_s(implementation, parked, count, bells):
    gc.collect()
    if not GC:
        gc.disable()
    t0 = time.thread_time()
    run(scheduled(with_handlers([await_handler(), async_time_handler()], timed_waits(count, parked, bells)),
                  implementation=implementation))
    spent = time.thread_time() - t0
    gc.enable()
    return spent


def cpu_per_wait_us(implementation, parked, bells, few=100, many=600):
    """WaitWithin を few 回と many 回 回した run の CPU の差 ÷ 回数の差 — 待つ task の Spawn と Cancel の費用を引いた 1 回の増分。"""
    return (cpu_total_s(implementation, parked, many, bells) - cpu_total_s(implementation, parked, few, bells)) / (many - few) * 1e6


def main():

    bells = queue.SimpleQueue()
    threading.Thread(target=ringer, args=(bells,), daemon=True).start()
    import doeff_core_effects.scheduler as s
    print("scheduler:", s.__file__, "gc:", "on" if GC else "off(測る間)", "python:", sys.version.split()[0], sys._is_gil_enabled() if hasattr(sys, "_is_gil_enabled") else "")
    for implementation in ("python", "rust"):
        medians = {}
        for parked in PARKED:
            runs = [cpu_per_wait_us(implementation, parked, bells) for _ in range(REPEAT)]
            medians[parked] = statistics.median(runs)
            print(f"{implementation} parked={parked}: WaitWithin 1 回の CPU 中央値 {medians[parked]:.0f} µs({REPEAT} 回 {[round(r) for r in runs]}・600 回と 100 回の差 ÷ 500)", flush=True)
        print(f"{implementation}: {PARKED[1]} 本 ÷ {PARKED[0]} 本 = {medians[PARKED[1]] / medians[PARKED[0]]:.2f}", flush=True)
    bells.put(None)


if __name__ == "__main__":
    main()
